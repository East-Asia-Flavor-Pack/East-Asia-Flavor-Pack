je_boshin_war = {
	activation_description = {
		always = yes
	}

	icon = "gfx/interface/icons/ig_icons/armed_forces.dds"
	group = je_group_global_international_situations
	# Only the two civil-war opening events activate this shared journal.
	is_shown_when_inactive = { always = no }
	should_be_involved = { eafp_boshin_country_involved = yes }
	should_show_when_not_involved = {
		country_rank >= rank_value:major_power
		has_interest_marker_in_region = sr:region_northeast_asia
	}
	immediate = {
		global_var:eafp_boshin_sabaku = { save_scope_as = eafp_boshin_sabaku }
		global_var:eafp_boshin_tobaku = { save_scope_as = eafp_boshin_tobaku }
	}
	outcome = {
		header = je_boshin_war_sabaku_victory_header
		effect_header = eafp_journal_on_resolution_header
		color = "gold_positive"
		conclusion_desc = eafp_je_boshin_war_complete_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = eafp_boshin_sabaku_victory_tt
				eafp_boshin_sabaku_alive = yes
				eafp_boshin_tobaku_alive = no
			}
		}
		effect = {
			scope:eafp_boshin_sabaku = {
				trigger_event = { id = boshin_war.9 }
				change_tag = JAP
			}
			eafp_boshin_cleanup = yes
		}
	}

	outcome = {
		header = je_boshin_war_tobaku_victory_header
		effect_header = eafp_journal_on_resolution_header
		color = "blue_negative"
		conclusion_desc = eafp_je_boshin_war_fail_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = eafp_boshin_tobaku_victory_tt
				eafp_boshin_tobaku_alive = yes
				eafp_boshin_sabaku_alive = no
			}
		}
		effect = { eafp_boshin_cleanup = yes }
	}

	outcome = {
		conclusion_desc = eafp_journal_invalid_conclusion_desc
		flags = { hidden invalid_outcome }
		trigger = {
			OR = {
				NOR = {
					eafp_boshin_sabaku_alive = yes
					eafp_boshin_tobaku_alive = yes
				}
				AND = {
					has_global_variable = eafp_boshin_checked
					eafp_boshin_sabaku_alive = yes
					eafp_boshin_tobaku_alive = yes
					eafp_boshin_conflict_ongoing = no
				}
			}
		}
		effect = { eafp_boshin_cleanup = yes }
	}

	on_weekly_pulse = {
		effect = {
			eafp_boshin_sync_participants = yes
			global_var:eafp_boshin_sabaku ?= {
				random_scope_war = {
					limit = {
						is_warleader = global_var:eafp_boshin_sabaku
						is_warleader = global_var:eafp_boshin_tobaku
					}
					add_war_war_support = { target = global_var:eafp_boshin_sabaku value = 5 }
					add_war_war_support = { target = global_var:eafp_boshin_tobaku value = 5 }
				}
			}
			set_global_variable = eafp_boshin_checked
		}
	}
	weight = 20000
	should_be_pinned_by_default_uninvolved_or_context = yes
}
