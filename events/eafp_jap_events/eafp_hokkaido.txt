	 event_image = { 
		video = "asia_confucianism_shinto"
	}
	
	duration = 3
	
	immediate = {}

	trigger = {
		has_journal_entry = je_taming_the_north
		has_variable = eafp_jap_hokkaido_castle_chain_started
		NOT = { has_variable = eafp_jap_hokkaido_castle_chain_completed }
	}
	
	option = { # 후쿠야마칸 개축
		default_option = yes

			value = 1

		trigger_event = { id = hokkaido.3 }
	}

	option = { # 새로 축성
		default_option = yes
		name = hokkaido.2.b
		set_variable = {
			name = hokkaido2
			value = 2
		}
		trigger_event = { id = hokkaido.3 }
	}
}

hokkaido.3 = { # 축성 부지 확보
	type = country_event
	placement = s:STATE_HOKKAIDO.region_state:JAP

	title = hokkaido.3.t
	desc = hokkaido.3.d
	flavor = hokkaido.3.f

	icon = "gfx/interface/icons/event_icons/event_military.dds"

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	 event_image = { 
		video = "asia_confucianism_shinto"
	}
	
	duration = 3
	
	immediate = {}

	trigger = {
		has_variable = eafp_jap_hokkaido_castle_chain_started
		NOT = { has_variable = eafp_jap_taming_north_failed }
		has_state_in_state_region = STATE_HOKKAIDO
		exists = var:hokkaido2
	}
	
	option = { # 비용 대기
		default_option = yes
		name = hokkaido.3.a
		if = {
			limit = { var:hokkaido2 = 1 }
			s:STATE_HOKKAIDO.region_state:JAP = {
				add_modifier = {
					name = fukuyama_castle_construction_han_modifier
					days = normal_modifier_time
				}
			}
			add_modifier = {
				name = fukuyama_castle_construction_bakufu_modifier
				days = normal_modifier_time
			}
		}
		else = {
			s:STATE_HOKKAIDO.region_state:JAP = {
				add_modifier = {
					name = fukuyama_castle_construction_han_modifier
					days = normal_modifier_time
					multiplier = 2
				}
			}
			add_modifier = {
				name = fukuyama_castle_construction_bakufu_modifier
				days = normal_modifier_time
				multiplier = 2
			}
		}
	}

	option = { # 알아서 하라 해
		default_option = yes
		name = hokkaido.3.b
		if = {
			limit = { var:hokkaido2 = 1 }
			s:STATE_HOKKAIDO.region_state:JAP = {
				add_modifier = {
					name = fukuyama_castle_construction_han_modifier
					days = normal_modifier_time
					multiplier = 2
				}
			}
		}
		else = {
			s:STATE_HOKKAIDO.region_state:JAP = {
				add_modifier = {
					name = fukuyama_castle_construction_han_modifier
					days = normal_modifier_time
					multiplier = 4
				}
			}
			custom_tooltip = hokkaido.3.ttb2
		}
	}

	after = {
		trigger_event = {
			id = hokkaido.4
			days = normal_modifier_time
		}
	}
}

hokkaido.5 = { # 아이누의 영내 습격
	type = country_event
	placement = s:STATE_HOKKAIDO.region_state:JAP

	title = hokkaido.5.t
	desc = hokkaido.5.d
	flavor = hokkaido.5.f

	icon = "gfx/interface/icons/event_icons/event_protest.dds"

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	event_image = { 
		video = "unspecific_vandalized_storefront"
	}
	
	duration = 3

	cooldown = { days = normal_modifier_time }
	
	immediate = {}

	trigger = {
		has_journal_entry = je_taming_the_north
		OR = {
			NOT = { exists = var:eafp_jap_ainu_friendship }
			var:eafp_jap_ainu_friendship < 1
		}
	}
	
	option = {
		default_option = yes
		name = hokkaido.5.a
		change_variable = {
			name = eafp_jap_ainu_friendship
			add = -1
		}

		s:STATE_HOKKAIDO.region_state:JAP ?= {
			add_devastation = 10
		}
	}

	option = {
		name = hokkaido.5.b
		change_variable = {
			name = eafp_jap_ainu_friendship
			add = 1
		}

		s:STATE_HOKKAIDO.region_state:JAP = {
			add_modifier = {
				name = ainu_protest_modifier
				days = short_modifier_time
			}
		}
		s:STATE_HOKKAIDO.region_state:JAP ?= {
			add_devastation = 5
		}
	}
}

hokkaido.6 = { # 아이누 부족의 자발적인 합류
	type = country_event
	placement = s:STATE_HOKKAIDO.region_state:JAP

	title = hokkaido.6.t
	desc = hokkaido.6.d
	flavor = hokkaido.6.f

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	event_image = { 
		video = "asia_confucianism_shinto"
	}
	
	duration = 3

	cooldown = { days = normal_modifier_time }
	
	immediate = {}

	trigger = {
		has_journal_entry = je_taming_the_north
		exists = var:eafp_jap_ainu_friendship
		var:eafp_jap_ainu_friendship >= 1
	}
	
	option = {
		default_option = yes
		name = hokkaido.6.a
		change_variable = {
			name = eafp_jap_ainu_friendship
			add = 1
		}

		s:STATE_HOKKAIDO.region_state:JAP = {
			every_scope_pop = {
				limit = {
					has_pop_culture = ainu
				}
				change_pop_culture = { target = cu:japanese value = 0.05 }
			}
		}
		add_modifier = {
			name = ainu_assimilation_modifier
			days = short_modifier_time
		}
	}
}
