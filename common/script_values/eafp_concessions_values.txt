# Building income is deliberately weekly_profit, as approved by the user.
# Add all urban profits first, including losses; clamp only the state total.
eafp_concession_state_profit = {
	value = 0
	every_scope_building = {
		limit = { eafp_concession_is_urban_building = yes }
		add = weekly_profit
	}
}
eafp_concession_state_share = {
	value = eafp_concession_state_profit
	min = 0
	multiply = 0.05
}
eafp_concession_active_holder_count = {
	value = 0
	if = {
		limit = { has_variable_list = concession_owner_varlist eafp_concession_eligible_state = yes }
		every_in_list = {
			variable = concession_owner_varlist
			limit = {
				is_country_alive = yes
				has_company = company_type:company_eafp_concession
				NOT = { this = prev.owner }
				prestige >= prev.owner.prestige
			}
			add = 1
		}
	}
}
eafp_concession_host_debit = {
	value = 0
	every_scope_state = {
		limit = { has_variable_list = concession_owner_varlist eafp_concession_eligible_state = yes }
		add = eafp_concession_state_debit
	}
}
eafp_concession_state_debit = {
	value = eafp_concession_state_share
	multiply = eafp_concession_active_holder_count
}
eafp_concession_holder_credit = {
	value = 0
	every_in_global_list = {
		variable = concession_candidate_states_gvarlist
		limit = { eafp_concession_active = { HOLDER = prev } }
		add = eafp_concession_state_share
	}
}

# Read actual building levels; never cache the cap or scale in persistent variables.
eafp_concession_total_levels = {
	value = 0
	every_scope_building = {
		limit = { is_building_type = building_eafp_concession }
		add = level
	}
}
eafp_concession_urban_cap = {
	value = 0
	every_scope_building = {
		limit = { is_building_type = building_urban_center }
		add = level
	}
}
eafp_concession_free_capacity = {
	value = eafp_concession_urban_cap
	subtract = eafp_concession_total_levels
	min = 0
}
eafp_concession_over_capacity = {
	value = eafp_concession_total_levels
	subtract = eafp_concession_urban_cap
	min = 0
}
eafp_concession_trade_equivalent = {
	value = eafp_concession_total_levels
	multiply = 5
}
# Building scope; current average profitability, not a forecast of the next level.
eafp_concession_industry_score = {
	value = weekly_profit
	divide = { value = level min = 1 }
}

eafp_concession_expansion_cost = { value = 50000 }
# State scope, evaluated only in the return effect with its pre-removal level snapshot.
eafp_concession_loss_relations = {
	value = scope:concession_loss_levels
	multiply = 2
	add = {
		value = scope:concession_loss_levels
		divide = { value = eafp_concession_urban_cap min = 1 }
		multiply = 10
	}
	max = 50
}
eafp_concession_loss_prestige = {
	value = scope:concession_loss_levels
	add = {
		value = scope:concession_loss_levels
		divide = { value = eafp_concession_urban_cap min = 1 }
		multiply = 5
	}
	max = 30
}

# Presentation only: cap can fall below existing levels, including zero.
eafp_concession_cap_usage = {
	value = eafp_concession_total_levels
	divide = { value = eafp_concession_urban_cap min = 1 }
	min = 0
	max = 1
}
