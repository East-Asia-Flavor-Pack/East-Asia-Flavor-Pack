is_eafp_mgl_heartland_state = {
	OR = {
		state_region = s:STATE_URGA
		state_region = s:STATE_ULIASTAI
		state_region = s:STATE_TUVA
		state_region = s:STATE_ALTAI
	}
}

is_eafp_mgl_united_industrial_state = {
	OR = {
		is_eafp_mgl_heartland_state = yes
		state_region = s:STATE_BURYATIA
		state_region = s:STATE_HINGGAN
		state_region = s:STATE_ALXA
		state_region = s:STATE_DZUNGARIA
		state_region = s:STATE_QINGHAI
	}
}

is_eafp_mgl_united_industry_partner = {
	is_country_alive = yes
	OR = {
		c:MGL ?= this
		top_overlord = c:MGL
		has_treaty_foreign_investment_rights_with = { TARGET = c:MGL }
	}
}

is_eafp_mgl_steppe_representative_homeland = {
	OR = {
		AND = { owner = { country_has_primary_culture = cu:mongol } any_scope_culture = { cu:mongol ?= this has_homeland = prev } }
		AND = { owner = { country_has_primary_culture = cu:buryat } any_scope_culture = { cu:buryat ?= this has_homeland = prev } }
		AND = { owner = { country_has_primary_culture = cu:tuvan } any_scope_culture = { cu:tuvan ?= this has_homeland = prev } }
		AND = { owner = { country_has_primary_culture = cu:kazak } any_scope_culture = { cu:kazak ?= this has_homeland = prev } }
		AND = { owner = { country_has_primary_culture = cu:kirgiz } any_scope_culture = { cu:kirgiz ?= this has_homeland = prev } }
		AND = { owner = { country_has_primary_culture = cu:uzbek } any_scope_culture = { cu:uzbek ?= this has_homeland = prev } }
		AND = { owner = { country_has_primary_culture = cu:turkmen } any_scope_culture = { cu:turkmen ?= this has_homeland = prev } }
	}
}

eafp_mgl_heartland_industrial_foundation_complete = {
	eafp_mgl_heartland_state_count >= 2
	any_scope_state = {
		filter = { is_eafp_mgl_heartland_state = yes }
		percent >= 0.75
		devastation < 0.1
		market_access >= 0.85
		tax_capacity >= tax_capacity_usage
	}
	eafp_mgl_heartland_construction_levels >= 2
}

eafp_mgl_heartland_raw_materials_complete = {
	eafp_mgl_heartland_raw_material_types >= 2
	eafp_mgl_heartland_raw_material_levels >= 8
	any_scope_building = {
		filter = {
			state = { is_eafp_mgl_heartland_state = yes }
			OR = { is_building_type = building_coal_mine is_building_type = building_iron_mine is_building_type = building_logging_camp }
		}
		percent >= 0.75
		cash_reserves_ratio > 0.25
	}
}

eafp_mgl_collectivization_crisis_conditions = {
	has_variable = eafp_mgl_forced_collectivization_active
	any_scope_state = {
		filter = {
			is_eafp_mgl_heartland_state = yes
			calc_true_if = {
				amount >= 2
				turmoil >= 0.2
				devastation >= 0.1
				market_access < 0.75
				tax_capacity < tax_capacity_usage
			}
		}
		count >= 2
	}
	any_scope_building = {
		filter = {
			state = { is_eafp_mgl_heartland_state = yes }
			OR = { is_building_type = building_livestock_ranch is_building_type = building_food_industry is_building_type = building_textile_mill }
		}
		percent >= 0.25
		cash_reserves_ratio <= 0.25
	}
}

	is_country_alive = yes
	is_revolutionary = no
	NOT = { c:MGL ?= this }
	NOT = { has_variable = eafp_mgl_steppe_league_member }
	NOT = { has_variable = eafp_mgl_steppe_league_immediate_rejector }
	OR = {
		country_has_primary_culture = cu:mongol
		country_has_primary_culture = cu:buryat
		country_has_primary_culture = cu:tuvan
		country_has_primary_culture = cu:kazak
		country_has_primary_culture = cu:kirgiz
		country_has_primary_culture = cu:uzbek
		country_has_primary_culture = cu:turkmen
		c:BRY ?= this
		c:TUV ?= this
		c:ALI ?= this
		c:KAZ ?= this
		c:KZH ?= this
		c:UZH ?= this
		c:OZH ?= this
		c:KYR ?= this
		c:KOK ?= this
		c:BUK ?= this
		c:KHI ?= this
		c:TRM ?= this
		c:TRC ?= this
		c:TRH ?= this
	}
	any_scope_state = {
		OR = {
			is_eafp_mgl_united_industrial_state = yes
			state_region = s:STATE_KHIVA
			state_region = s:STATE_KIRGHIZIA
			state_region = s:STATE_TURKMENIA
			state_region = s:STATE_SEMIRECHE
			state_region = s:STATE_JETISY

		is_eafp_mgl_steppe_representative_homeland = yes
	}
