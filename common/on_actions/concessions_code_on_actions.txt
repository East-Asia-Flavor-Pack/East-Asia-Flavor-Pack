# Rights are in-force treaty articles tied to one actual state plus building ownership,
# so nothing is copied between states on a split, merge or civil war. Ownership changes
# only re-derive the candidate list and company lifecycle.
eafp_concessions_state_changed = {
	trigger = {
		OR = {
			is_in_geographic_region = geographic_region_china_proper
			has_building = building_eafp_concession
		}
	}
	effect = {
		eafp_concession_refresh_effect = yes
	}
}

# ROOT is the old country and target the newly formed revolt.
eafp_concessions_civil_war_start = {
	effect = {
		eafp_concession_refresh_effect = yes
	}
}
