# Building income is deliberately weekly_profit, as approved by the user.
# Add all urban profits first, including losses; clamp only the state total.
eafp_concession_state_profit = {
	value = 0
	every_scope_building = {
		limit = { eafp_concession_is_urban_building = yes }
		add = weekly_profit
	}
}
# Incident compensation only (paid by the holder to the host in events).
# There is no recurring treasury transfer from the Chinese owner to holders.
eafp_concession_state_share = {
	value = eafp_concession_state_profit
	min = 0
	multiply = 0.05
}
# Requires scope:concession_state (saved by effects; added via AddScope in the GUI).
# Counts countries with an in-force concession article for the state, a company and
# prestige at least equal to the owner's. Treaties are bilateral, so a non-owner bound
# by such an article is its target. Ownership is checked by eafp_concession_active,
# not here (a value cannot carry the country into building scope).
eafp_concession_active_holder_count = {
	value = 0
	if = {
		limit = {
			exists = scope:concession_state
			scope:concession_state = { eafp_concession_eligible_state = yes }
		}
		every_country = {
			limit = {
				is_country_alive = yes
				has_company = company_type:company_eafp_concession
				NOT = { this = scope:concession_state.owner }
				prestige >= scope:concession_state.owner.prestige
				any_scope_treaty = {
					any_scope_article = {
						has_type = eafp_grant_concession
						input_state = scope:concession_state
					}
				}
			}
			add = 1
		}
	}
}

# Read actual building levels; never cache the cap or scale in persistent variables.
eafp_concession_total_levels = {
	value = 0
	every_scope_building = {
		limit = { is_building_type = building_eafp_concession }
		add = level
	}
}
eafp_concession_urban_levels = {
	value = 0
	every_scope_building = {
		limit = { is_building_type = building_urban_center }
		add = level
	}
}
# Actual levels plus levels already waiting in the construction queue.
eafp_concession_total_levels_after_queue = {
	value = 0
	every_scope_building = {
		limit = { is_building_type = building_eafp_concession }
		add = level_after_queued_constructions
	}
}
eafp_concession_queued_levels = {
	value = eafp_concession_total_levels_after_queue
	subtract = eafp_concession_total_levels
	min = 0
}
# New grants consume five levels; journal expansions queue one or five.
eafp_concession_urban_cap = {
	value = eafp_concession_urban_levels
	multiply = 5
}
# Queued levels already reserve cap room, so they are subtracted as well.
eafp_concession_free_capacity = {
	value = eafp_concession_urban_cap
	subtract = eafp_concession_total_levels_after_queue
	min = 0
}
eafp_concession_over_capacity = {
	value = eafp_concession_total_levels
	subtract = eafp_concession_urban_cap
	min = 0
}
eafp_concession_trade_equivalent = {
	value = eafp_concession_total_levels
}
# Building scope; current average profitability, not a forecast of the next level.
eafp_concession_industry_score = {
	value = weekly_profit
	divide = { value = level min = 1 }
}

eafp_concession_expansion_cost = { value = 50000 }
# Per-button totals; the suffix matches the LEVELS parameter of the expand trigger/effect.
eafp_concession_expansion_cost_levels_1 = { value = eafp_concession_expansion_cost }
eafp_concession_expansion_cost_levels_5 = {
	value = eafp_concession_expansion_cost
	multiply = 5
}
# State scope, evaluated only in the return effect with its pre-removal level snapshot.
eafp_concession_loss_relations = {
	value = scope:concession_loss_levels
	multiply = 2
	add = {
		value = scope:concession_loss_levels
		divide = { value = eafp_concession_urban_levels min = 1 }
		multiply = 10
	}
	max = 50
}
eafp_concession_loss_prestige = {
	value = scope:concession_loss_levels
	add = {
		value = scope:concession_loss_levels
		divide = { value = eafp_concession_urban_levels min = 1 }
		multiply = 5
	}
	max = 30
}

# Presentation only: cap can fall below existing levels, including zero.
eafp_concession_cap_usage = {
	value = eafp_concession_total_levels
	divide = { value = eafp_concession_urban_cap min = 1 }
	min = 0
	max = 1
}
