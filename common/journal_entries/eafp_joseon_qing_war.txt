je_eafp_joseon_qing_war = {
	icon = "gfx/interface/icons/event_icons/event_military.dds"

	group = je_group_crises

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_eafp_joseon_qing_war_status
				trigger = {
					has_variable = eafp_var_jqw_current_war_support
				}
			}
			triggered_desc = {
				desc = je_eafp_joseon_qing_war_status_pending
				trigger = {
					always = yes
				}
			}
		}
	}

	progress_desc = je_eafp_joseon_qing_war_progress

	is_shown_when_inactive = {
		c:KOR ?= THIS
		exists = c:CHI
		is_subject_of = c:CHI
	}

	possible = {
		c:KOR ?= THIS
		exists = c:CHI
		any_scope_war = {
			eafp_jqw_is_joseon_independence_war_trigger = yes
		}
	}

	immediate = {
		c:CHI = {
			save_scope_as = qing_enemy
		}
		if = {
			limit = {
				any_scope_war = {
					eafp_jqw_is_joseon_independence_war_trigger = yes
				}
			}
			random_scope_war = {
				limit = {
					eafp_jqw_is_joseon_independence_war_trigger = yes
				}
				save_scope_as = joseon_qing_war
			}
		}
		trigger_event = { id = eafp_joseon_qing_war_events.1 popup = yes }
	}

	complete = {
		NOT = {
			any_scope_war = {
				this = scope:joseon_qing_war
			}
		}
		OR = {
			NOT = { exists = c:CHI }
			AND = {
				exists = c:CHI
				NOT = { is_subject_of = c:CHI }
			}
		}
	}

	on_complete = {
		trigger_event = { id = eafp_joseon_qing_war_events.2 popup = yes }
	}

	event_outcome_completed_effect_desc = {
		header = eafp_joseon_qing_war_events.2.t
		effect = {
			if = {
				limit = {
					eafp_jqw_war_support_high_trigger = yes
				}
				eafp_jqw_victory_high_reward_effect = yes
			}
			else_if = {
				limit = {
					eafp_jqw_war_support_medium_trigger = yes
				}
				eafp_jqw_victory_medium_reward_effect = yes
			}
			else = {
				eafp_jqw_victory_low_reward_effect = yes
			}
		}
	}

	fail = {
		NOT = {
			any_scope_war = {
				this = scope:joseon_qing_war
			}
		}
		exists = c:CHI
		is_subject_of = c:CHI
	}

	on_fail = {
		trigger_event = { id = eafp_joseon_qing_war_events.3 popup = yes }
	}

	event_outcome_failed_effect_desc = {
		header = eafp_joseon_qing_war_events.3.t
		effect = {
			add_modifier = {
				name = eafp_modifier_joseon_qing_war_defeat
				years = normal_modifier_time
			}
			add_radicals = {
				value = medium_radicals
			}
		}
	}

	invalid = {
		NOT = { c:KOR ?= THIS }
	}

	on_invalid = {
		remove_variable = eafp_var_jqw_current_war_support
	}

	on_weekly_pulse = {
		effect = {
			if = {
				limit = {
					any_scope_war = {
						this = scope:joseon_qing_war
					}
				}
				set_variable = {
					name = eafp_var_jqw_current_war_support
					value = "scope:joseon_qing_war.has_war_support(root)"
				}
			}
		}
	}

	weight = 7000
	can_deactivate = no

	should_be_pinned_by_default_uninvolved_or_context = yes
}
