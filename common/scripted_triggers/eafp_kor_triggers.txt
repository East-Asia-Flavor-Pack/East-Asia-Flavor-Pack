	OR = {
		NOT = { country_has_monarchy_law = yes }
		has_law = law_type:law_census_voting
		has_law = law_type:law_universal_suffrage
		has_law = law_type:law_anarchy
	}
}

	is_warleader = c:KOR
	is_warleader = c:CHI
	c:KOR = {
		war_participant_has_war_goal_of_type_against = {
			type = independence
			target = c:CHI
		}
	}
}

eafp_jqw_war_support_high_trigger = {
	var:eafp_var_jqw_current_war_support >= 50
}

eafp_jqw_war_support_medium_trigger = {
	OR = {
		NOT = { has_variable = eafp_var_jqw_current_war_support }
		AND = {
			var:eafp_var_jqw_current_war_support >= 0
			var:eafp_var_jqw_current_war_support < 50
		}
	}
}

eafp_jqw_war_support_low_trigger = {
	var:eafp_var_jqw_current_war_support < 0
}

eafp_jqw_war_support_below_high_trigger = {
	OR = {
		NOT = { has_variable = eafp_var_jqw_current_war_support }
		var:eafp_var_jqw_current_war_support < 50
	}
}
