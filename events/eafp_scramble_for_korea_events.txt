    type = country_event
    placement = ROOT
    title = scramble_for_korea_events.7.t
    desc = { first_valid = { triggered_desc = { trigger = { scope:scramble_kind = 4 } desc = scramble_for_korea_events.7.d4 } triggered_desc = { trigger = { scope:scramble_kind = 5 } desc = scramble_for_korea_events.7.d5 } triggered_desc = { trigger = { scope:scramble_kind = 6 } desc = scramble_for_korea_events.7.d6 } triggered_desc = { trigger = { scope:scramble_kind = 10 } desc = scramble_for_korea_events.7.d10 } triggered_desc = { trigger = { scope:scramble_kind = 12 } desc = scramble_for_korea_events.7.d12 } } }

    icon = "gfx/interface/icons/event_icons/event_map.dds"
    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
    duration = 1

    option = {
        name = scramble_for_korea_events.7.a

        trigger = { scramble_for_korea_response_valid = yes }

        scope:scramble_actor = { scramble_for_korea_change_influence = { AMOUNT = 3 } }
        if = {
            limit = { scope:scramble_kind = 4 }
            scramble_for_korea_change_influence = { AMOUNT = 2 }
            if = {
                limit = { NOT = { any_political_movement = { is_political_movement_type = movement_modernizer } } }
                create_political_movement = { type = movement_modernizer }
            }
            every_political_movement = {
                limit = { is_political_movement_type = movement_modernizer }
                add_modifier = { name = modifier_enlightenment_support_from_foreign days = 730 }
            }
        }
        else_if = {
            limit = { scope:scramble_kind = 5 }
            scope:scramble_building = { add_modifier = { name = modifier_kor_improved_labor_conditions days = 730 } }
            scope:scramble_actor = { ig:ig_industrialists ?= { add_modifier = { name = modifier_curtailed_kor_labor_abuses days = 730 } } }
            add_treasury = 5000
        }
        else_if = {
            limit = { scope:scramble_kind = 6 }
            scope:scramble_actor = { add_modifier = { name = modifier_korea_scramble_advisers days = 730 } }
        }
        else_if = {
            limit = { scope:scramble_kind = 10 }
            add_treasury = 5000
            scope:scramble_actor = { change_relations = { country = c:KOR value = 5 } }
        }
        else_if = {
            limit = { scope:scramble_kind = 12 }
            if = {
                limit = { any_political_lobby = { is_political_lobby_type = lobby_pro_country target = scope:scramble_actor } }
                random_political_lobby = {
                    limit = { is_political_lobby_type = lobby_pro_country target = scope:scramble_actor }
                    add_lobby_member = scope:scramble_ig
                }
            }
            else = { create_political_lobby = { type = lobby_pro_country target = scope:scramble_actor add_interest_group = scope:scramble_ig } }
            scope:scramble_ig = { add_modifier = { name = modifier_korea_enlightenment_support days = 730 } }
        }
        if = { limit = { scope:scramble_kind = 6 } scramble_for_korea_rival_response = yes }
        else = { scramble_for_korea_close_offer = yes }

        ai_chance = { base = 50 }
    }
    option = {
        name = scramble_for_korea_events.7.b
        default_option = yes
        trigger = { scramble_for_korea_response_valid = yes }
        scramble_for_korea_change_influence = { AMOUNT = 3 } scope:scramble_actor = { scramble_for_korea_change_influence = { AMOUNT = -2 } } scramble_for_korea_close_offer = yes
        ai_chance = { base = 25 }
    }
    option = {
        name = scramble_for_korea_events.7.c

        trigger = { scramble_for_korea_response_valid = yes }
        scramble_for_korea_change_influence = { AMOUNT = 1 } scope:scramble_actor = { scramble_for_korea_change_influence = { AMOUNT = 1 } } scramble_for_korea_close_offer = yes
        ai_chance = { base = 25 }
    }

        name = scramble_for_korea_events.7.cancel
        default_option = yes
        trigger = { NOT = { scramble_for_korea_response_valid = yes } }
