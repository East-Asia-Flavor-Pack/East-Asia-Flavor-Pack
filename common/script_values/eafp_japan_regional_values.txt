# Country scope throughout. L/I are stored independently; no character average.
# Legacy monthly drift retained, with guarded GDP and tax-capacity denominators.
eafp_japan_regional_tax_cap = 0.25

eafp_japan_loyalty_TOHOKU_value = {
    value = 50
    if = { limit = { has_variable = eafp_japan_region_initialized_TOHOKU } value = var:eafp_japan_loyalty_TOHOKU }
}
eafp_japan_loyalty_TOHOKU_monthly = {
    value = 50
    subtract = eafp_japan_loyalty_TOHOKU_value
    divide = 100
    s:STATE_TOHOKU.region_state:JAP ?= {
        subtract = radical_fraction
        add = loyalist_fraction
        add = modifier:state_bakuhantaisei_loyalty_progress_bar_monthly_add
    }
    # 다이묘 영지 감독 / 충성심 재확인: 원래 주간 임무 modifier의 월간 효과.
    if = { limit = { has_modifier = modifier_oversee_daimyo_domains_TOHOKU } subtract = 0.1 }
    if = { limit = { has_modifier = modifier_reaffirm_daimyos_loyalty_TOHOKU } add = 0.5 }
}
eafp_japan_independency_TOHOKU_monthly = {
    value = 50
    subtract = eafp_japan_loyalty_TOHOKU_value
    divide = 200
    add = { value = 50 subtract = var:eafp_japan_independency_TOHOKU divide = 200 }
    if = {
        limit = { gdp > 0 exists = s:STATE_TOHOKU.region_state:JAP }
        add = { value = s:STATE_TOHOKU.region_state:JAP.gdp divide = gdp min = 0 max = 1 }
    }
    s:STATE_TOHOKU.region_state:JAP ?= {
        if = {
            limit = { tax_capacity_usage > 0 }
            add = { value = tax_capacity_usage subtract = tax_capacity divide = tax_capacity_usage min = 0 max = 1 divide = 2 }
        }
        add = modifier:state_bakuhantaisei_independency_progress_bar_monthly_add
    }
    if = { limit = { has_modifier = modifier_oversee_daimyo_domains_TOHOKU } subtract = 0.5 }
}
eafp_japan_tax_loss_TOHOKU = { value = var:eafp_japan_independency_TOHOKU divide = 100 multiply = eafp_japan_regional_tax_cap }
