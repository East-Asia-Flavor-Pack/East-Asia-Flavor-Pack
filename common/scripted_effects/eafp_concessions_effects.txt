# All callers provide concession_state (state), concession_host and concession_holder (countries).
# Called from the monthly incident and the police-conflict option.
eafp_concession_start_return_war = {
	if = {
		limit = { scope:concession_host = { eafp_concession_can_start_return_war = yes } }
		scope:concession_host = {
			create_diplomatic_play = {
				name = dp_eafp_concession_incident
				type = dp_eafp_concession_incident
				target_country = scope:concession_holder
				war = yes
				add_war_goal = {
					holder = scope:concession_host
					type = eafp_concession_incident_return
					target_country = scope:concession_holder
					target_state = scope:concession_state
					primary_demand = yes
				}
			}
		}
		if = {
			limit = { scope:concession_host = { has_war_with = scope:concession_holder } }
			scope:concession_host = { trigger_event = { id = concessions_in_china.205 } }
		}
	}
}

# State scope; used only for an identified civil-war split or a same-owner merge.
# Inheritance never calls grant, so it cannot award another concession level.
eafp_concession_inherit_state_rights = {
	every_in_list = {
		variable = concession_owner_varlist
		$TARGET$ = {
			if = {
				limit = { NOT = { is_target_in_variable_list = { name = concession_owner_varlist target = prev } } }
				add_to_variable_list = { name = concession_owner_varlist target = prev }
			}
		}
	}
}

# Building scope. Binary search for COUNTRY's levels in this building through
# levels_owned_by_country comparisons (engine levels are integers; 32 steps suffice).
# Result: scope:concession_level_low. COUNTRY must be a saved scope.
eafp_concession_measure_country_levels = {
	save_temporary_scope_value_as = { name = concession_level_low value = 0 }
	save_temporary_scope_value_as = { name = concession_level_high value = level }
	while = {
		count = 32
		save_temporary_scope_value_as = {
			name = concession_level_mid
			value = { value = scope:concession_level_low add = scope:concession_level_high divide = 2 ceiling = yes }
		}
		if = {
			limit = { levels_owned_by_country = { target = $COUNTRY$ value >= scope:concession_level_mid } }
			save_temporary_scope_value_as = { name = concession_level_low value = scope:concession_level_mid }
		}
		else = {
			save_temporary_scope_value_as = { name = concession_level_high value = { value = scope:concession_level_mid subtract = 1 } }
		}
	}
}

# State scope. Adds VALUE to KEY's entry in eafp_concession_rebuild_map (creates it if absent).
# KEY is a country scope; VALUE is a saved scope value.
eafp_concession_rebuild_map_add = {
	if = {
		limit = { is_key_in_variable_map = { name = eafp_concession_rebuild_map target = $KEY$ } }
		save_temporary_scope_value_as = {
			name = eafp_rebuild_merged
			value = { value = "variable_map(eafp_concession_rebuild_map|$KEY$)" add = $VALUE$ }
		}
		remove_from_variable_map = { name = eafp_concession_rebuild_map key = $KEY$ }
		add_to_variable_map = { name = eafp_concession_rebuild_map key = $KEY$ value = scope:eafp_rebuild_merged }
	}
	else = {
		add_to_variable_map = { name = eafp_concession_rebuild_map key = $KEY$ value = $VALUE$ }
	}
}

# State scope. One create_building call giving LEVELS to scope:eafp_rebuild_owner.
# The owner's concession company receives the levels; an owner without one (e.g. a
# returned holder whose company was dissolved) keeps them as country ownership.
# On an existing building the engine keeps max(existing, scripted) reserves, so the
# follow-up calls pass RESERVES = 0 and never lower the restored reserves.
eafp_concession_rebuild_create = {
	if = {
		limit = { scope:eafp_rebuild_owner = { has_company = company_type:company_eafp_concession } }
		create_building = {
			building = building_eafp_concession
			activate_production_methods = { pm_eafp_concession pm_trade_center_trade_quantity_normal }
			reserves = $RESERVES$
			add_ownership = { company = { type = company_eafp_concession country = scope:eafp_rebuild_owner levels = $LEVELS$ } }
		}
	}
	else = {
		create_building = {
			building = building_eafp_concession
			activate_production_methods = { pm_eafp_concession pm_trade_center_trade_quantity_normal }
			reserves = $RESERVES$
			add_ownership = { country = { country = scope:eafp_rebuild_owner levels = $LEVELS$ } }
		}
	}
}

# State scope. ONE create_building call giving exactly LEVELS (a multiple of 5) to
# scope:eafp_rebuild_owner. create_building levels do not add to an existing building's
# level, so an owner's whole stored amount is created in a single call, never in pieces.
# The first owner founds the building and restores flag:cash_reserve (nearest 0.1).
eafp_concession_rebuild_chunk = {
	if = {
		limit = { has_building = building_eafp_concession }
		eafp_concession_rebuild_create = { LEVELS = $LEVELS$ RESERVES = 0 }
	}
	else = {
		save_temporary_scope_value_as = {
			name = eafp_rebuild_reserve
			value = "variable_map(eafp_concession_rebuild_map|flag:cash_reserve)"
		}
		if = {
			limit = { scope:eafp_rebuild_reserve >= 0.95 }
			eafp_concession_rebuild_create = { LEVELS = $LEVELS$ RESERVES = 1 }
		}
		else_if = {
			limit = { scope:eafp_rebuild_reserve >= 0.85 }
			eafp_concession_rebuild_create = { LEVELS = $LEVELS$ RESERVES = 0.9 }
		}
		else_if = {
			limit = { scope:eafp_rebuild_reserve >= 0.75 }
			eafp_concession_rebuild_create = { LEVELS = $LEVELS$ RESERVES = 0.8 }
		}
		else_if = {
			limit = { scope:eafp_rebuild_reserve >= 0.65 }
			eafp_concession_rebuild_create = { LEVELS = $LEVELS$ RESERVES = 0.7 }
		}
		else_if = {
			limit = { scope:eafp_rebuild_reserve >= 0.55 }
			eafp_concession_rebuild_create = { LEVELS = $LEVELS$ RESERVES = 0.6 }
		}
		else_if = {
			limit = { scope:eafp_rebuild_reserve >= 0.45 }
			eafp_concession_rebuild_create = { LEVELS = $LEVELS$ RESERVES = 0.5 }
		}
		else_if = {
			limit = { scope:eafp_rebuild_reserve >= 0.35 }
			eafp_concession_rebuild_create = { LEVELS = $LEVELS$ RESERVES = 0.4 }
		}
		else_if = {
			limit = { scope:eafp_rebuild_reserve >= 0.25 }
			eafp_concession_rebuild_create = { LEVELS = $LEVELS$ RESERVES = 0.3 }
		}
		else_if = {
			limit = { scope:eafp_rebuild_reserve >= 0.15 }
			eafp_concession_rebuild_create = { LEVELS = $LEVELS$ RESERVES = 0.2 }
		}
		else_if = {
			limit = { scope:eafp_rebuild_reserve >= 0.05 }
			eafp_concession_rebuild_create = { LEVELS = $LEVELS$ RESERVES = 0.1 }
		}
		else = {
			eafp_concession_rebuild_create = { LEVELS = $LEVELS$ RESERVES = 0 }
		}
	}
}

# State scope. LEVELS is 5 for both a new right and paid expansion (the +5 button).
# Rebuild instead of adding ownership to the live building:
#   1. snapshot into the state's variable map eafp_concession_rebuild_map
#        key = owner country -> levels it owns (levels_owned_by_country binary search)
#        key = flag:cash_reserve -> cash_reserves_ratio (0..1)
#      plus the trade-quantity PM and subsidy in temporary scope values;
#   2. add LEVELS to scope:concession_holder's entry;
#   3. remove_building, then ONE create_building per owner with its stored levels;
#   4. restore PM/subsidy and clear the map.
# The engine has no effect that staffs a building, so the new building re-hires from the
# state's (now unemployed) workers at the normal hiring rate.
eafp_concession_add_company_level = {
	save_temporary_scope_as = eafp_rebuild_state
	owner = { save_temporary_scope_as = eafp_rebuild_host }
	clear_variable_map = eafp_concession_rebuild_map
	save_temporary_scope_value_as = { name = eafp_rebuild_old_level value = 0 }
	save_temporary_scope_value_as = { name = eafp_rebuild_owned_sum value = 0 }
	# 1 limited, 2 normal (default for a new building), 3 high, 4 very high.
	save_temporary_scope_value_as = { name = eafp_rebuild_trade_pm value = 2 }
	save_temporary_scope_value_as = { name = eafp_rebuild_subsidized value = 0 }
	# A brand-new building starts with full reserves, as before.
	save_temporary_scope_value_as = { name = eafp_rebuild_reserve_low value = 1 }

	# 1. Snapshot the existing building.
	if = {
		limit = { has_building = building_eafp_concession }
		random_scope_building = {
			limit = { is_building_type = building_eafp_concession }
			save_temporary_scope_as = eafp_rebuild_building
			save_temporary_scope_value_as = { name = eafp_rebuild_old_level value = level }
			every_country = {
				save_temporary_scope_as = eafp_rebuild_owner
				scope:eafp_rebuild_building = {
					if = {
						limit = { levels_owned_by_country = { target = scope:eafp_rebuild_owner value >= 1 } }
						eafp_concession_measure_country_levels = { COUNTRY = scope:eafp_rebuild_owner }
						save_temporary_scope_value_as = {
							name = eafp_rebuild_owned_sum
							value = { value = scope:eafp_rebuild_owned_sum add = scope:concession_level_low }
						}
						scope:eafp_rebuild_state = {
							eafp_concession_rebuild_map_add = { KEY = scope:eafp_rebuild_owner VALUE = scope:concession_level_low }
						}
					}
				}
			}
			# Cash reserve ratio by bisection on [0, 1] (precision 1/1024).
			save_temporary_scope_value_as = { name = eafp_rebuild_reserve_low value = 0 }
			save_temporary_scope_value_as = { name = eafp_rebuild_reserve_high value = 1 }
			while = {
				count = 10
				save_temporary_scope_value_as = {
					name = eafp_rebuild_reserve_mid
					value = { value = scope:eafp_rebuild_reserve_low add = scope:eafp_rebuild_reserve_high divide = 2 }
				}
				if = {
					limit = { cash_reserves_ratio >= scope:eafp_rebuild_reserve_mid }
					save_temporary_scope_value_as = { name = eafp_rebuild_reserve_low value = scope:eafp_rebuild_reserve_mid }
				}
				else = {
					save_temporary_scope_value_as = { name = eafp_rebuild_reserve_high value = scope:eafp_rebuild_reserve_mid }
				}
			}
			if = {
				limit = { has_active_production_method = pm_trade_center_trade_quantity_limited }
				save_temporary_scope_value_as = { name = eafp_rebuild_trade_pm value = 1 }
			}
			else_if = {
				limit = { has_active_production_method = pm_trade_center_trade_quantity_high }
				save_temporary_scope_value_as = { name = eafp_rebuild_trade_pm value = 3 }
			}
			else_if = {
				limit = { has_active_production_method = pm_trade_center_trade_quantity_very_high }
				save_temporary_scope_value_as = { name = eafp_rebuild_trade_pm value = 4 }
			}
			if = {
				limit = { is_subsidized = yes }
				save_temporary_scope_value_as = { name = eafp_rebuild_subsidized value = 1 }
			}
		}
		# Levels no country comparison could attribute stay with the state owner, so the
		# rebuild never silently destroys levels.
		save_temporary_scope_value_as = {
			name = eafp_rebuild_unattributed
			value = { value = scope:eafp_rebuild_old_level subtract = scope:eafp_rebuild_owned_sum }
		}
		if = {
			limit = { scope:eafp_rebuild_unattributed > 0 }
			debug_log = "EAFP concession rebuild: unattributed levels kept as state-owner ownership"
			eafp_concession_rebuild_map_add = { KEY = scope:eafp_rebuild_host VALUE = scope:eafp_rebuild_unattributed }
		}
	}
	add_to_variable_map = { name = eafp_concession_rebuild_map key = flag:cash_reserve value = scope:eafp_rebuild_reserve_low }

	# 2. The requested increase goes to the holder's entry.
	save_temporary_scope_value_as = { name = eafp_rebuild_added value = $LEVELS$ }
	eafp_concession_rebuild_map_add = { KEY = scope:concession_holder VALUE = scope:eafp_rebuild_added }

	# 3. Remove the old building and recreate every owner's levels.
	if = {
		limit = { has_building = building_eafp_concession }
		remove_building = building_eafp_concession
	}
	every_country = {
		save_temporary_scope_as = eafp_rebuild_owner
		scope:eafp_rebuild_state = {
			if = {
				limit = { is_key_in_variable_map = { name = eafp_concession_rebuild_map target = scope:eafp_rebuild_owner } }
				save_temporary_scope_value_as = {
					name = eafp_rebuild_remaining
					value = "variable_map(eafp_concession_rebuild_map|scope:eafp_rebuild_owner)"
				}
				eafp_concession_rebuild_owner_levels = yes
			}
		}
	}

	# 4. Restore the snapshotted trade-quantity PM and subsidy, then drop the snapshot.
	if = {
		limit = { scope:eafp_rebuild_trade_pm = 1 }
		activate_production_method = { building_type = building_eafp_concession production_method = pm_trade_center_trade_quantity_limited }
	}
	else_if = {
		limit = { scope:eafp_rebuild_trade_pm = 3 }
		activate_production_method = { building_type = building_eafp_concession production_method = pm_trade_center_trade_quantity_high }
	}
	else_if = {
		limit = { scope:eafp_rebuild_trade_pm = 4 }
		activate_production_method = { building_type = building_eafp_concession production_method = pm_trade_center_trade_quantity_very_high }
	}
	if = {
		limit = { scope:eafp_rebuild_subsidized = 1 }
		random_scope_building = {
			limit = { is_building_type = building_eafp_concession }
			set_subsidized = yes
		}
	}
	save_temporary_scope_value_as = {
		name = eafp_rebuild_expected
		value = { value = scope:eafp_rebuild_old_level add = scope:eafp_rebuild_added }
	}
	if = {
		limit = { NOT = { eafp_concession_total_levels = scope:eafp_rebuild_expected } }
		debug_log = "EAFP concession rebuild: recreated level differs from snapshot + increase"
	}
	clear_variable_map = eafp_concession_rebuild_map
}

# Rights live on actual states, never state regions or province anchors.
eafp_concession_grant_effect = {
	if = {
		limit = {
			owner = scope:concession_host
			eafp_concession_can_grant = { HOLDER = scope:concession_holder }
		}
		scope:concession_holder = {
			if = {
				limit = { NOT = { has_company = company_type:company_eafp_concession } }
				add_modifier = { name = eafp_concession_company_slot }
				add_company = company_type:company_eafp_concession
				if = {
					limit = { has_company = company_type:company_eafp_concession }
					capital.state_region = { save_temporary_scope_as = concession_company_hq }
					company:company_eafp_concession = { set_company_state_region = scope:concession_company_hq }
				}
			}
		}
		if = {
			limit = { scope:concession_holder = { has_company = company_type:company_eafp_concession } }
			save_temporary_scope_value_as = { name = concession_previous_levels value = eafp_concession_total_levels }
			eafp_concession_add_company_level = { LEVELS = 5 }
			if = {
				limit = {
					eafp_concession_total_levels > scope:concession_previous_levels
					any_scope_building = {
						is_building_type = building_eafp_concession
						levels_owned_by_country = { target = scope:concession_holder value >= 5 }
					}
				}
				add_to_variable_list = { name = concession_owner_varlist target = scope:concession_holder }
			}
			else = { debug_log = "EAFP concession grant: company ownership creation failed; right not registered" }
		}
		else = { debug_log = "EAFP concession grant: company establishment failed; right not registered" }
		if = {
			limit = { NOT = { is_target_in_variable_list = { name = concession_owner_varlist target = scope:concession_holder } } }
			scope:concession_holder = {
				send_interface_message = {
					title = eafp_concession_grant_failed_title
					desc = eafp_concession_grant_failed_desc
					icon = "gfx/interface/icons/building_icons/building_trade_center.dds"
				}
			}
		}
		# Also removes an empty newly created company after a failed grant.
		eafp_concession_refresh_effect = yes
	}
}

# Shared by the player GUI and monthly AI. Expansion is in steps of 5 levels only.
# No treasury charge when growth fails.
eafp_concession_expand_effect = {
	if = {
		limit = { eafp_concession_can_expand = { HOLDER = scope:concession_holder } }
		save_temporary_scope_value_as = { name = concession_previous_levels value = eafp_concession_total_levels }
		eafp_concession_add_company_level = { LEVELS = 5 }
		if = {
			limit = { eafp_concession_total_levels > scope:concession_previous_levels }
			scope:concession_holder = { add_treasury = { value = eafp_concession_expansion_cost multiply = -1 } }
		}
	}
}

# FORCED is explicit at every caller; no dependence on treaty callback ordering.
eafp_concession_return_effect = {
	if = {
		limit = {
			owner = scope:concession_host
			eafp_concession_has_holder = { HOLDER = scope:concession_holder }
		}
		# Derive the holder's actual levels through comparisons, not a persistent ledger.
		# Binary search needs at most 32 steps for engine integer building levels.
		save_temporary_scope_value_as = { name = concession_loss_levels value = 0 }
		every_scope_building = {
			limit = { is_building_type = building_eafp_concession }
			eafp_concession_measure_country_levels = { COUNTRY = scope:concession_holder }
			save_temporary_scope_value_as = { name = concession_loss_levels value = scope:concession_level_low }
		}
		# Snapshot before removing the right/building or dissolving the company.
		save_temporary_scope_value_as = { name = concession_loss_relation value = eafp_concession_loss_relations }
		save_temporary_scope_value_as = { name = concession_loss_prestige value = eafp_concession_loss_prestige }
		remove_list_variable = { name = concession_owner_varlist target = scope:concession_holder }
		if = {
			limit = {
				has_building = building_eafp_concession
				NOT = { any_in_list = { variable = concession_owner_varlist always = yes } }
			}
			remove_building = building_eafp_concession
		}
		else_if = {
			limit = { has_building = building_eafp_concession }
			# Engine exposes whole-building removal, not company-specific level removal.
			# Do not destroy other holders' assets or invent negative ownership support.
			debug_log = "EAFP concession return: rights revoked; shared ownership removal unavailable, residual assets retained"
		}
		if = {
			limit = { always = $FORCED$ eafp_concession_loss_prestige > 0 }
			scope:concession_host = {
				change_relations = { country = scope:concession_holder value = { value = scope:concession_loss_relation multiply = -1 } }
			}
			scope:concession_holder = {
				# The previous loss cannot be reconstructed after rights/buildings are removed.
				# Retain the strongest unexpired shock; a smaller loss must not erase it.
				if = {
					limit = { NOT = { has_variable = concession_loss_prestige_snapshot } }
					set_variable = { name = concession_loss_prestige_snapshot value = 0 days = 730 }
				}
				if = {
					limit = {
						OR = {
							NOT = { has_modifier = eafp_concession_forced_loss }
							var:concession_loss_prestige_snapshot < scope:concession_loss_prestige
						}
					}
					set_variable = { name = concession_loss_prestige_snapshot value = scope:concession_loss_prestige days = 730 }
					remove_modifier = eafp_concession_forced_loss
					add_modifier = { name = eafp_concession_forced_loss multiplier = scope:concession_loss_prestige days = 730 }
				}
			}
		}
		eafp_concession_refresh_effect = yes
	}
}

# One global refresh, used by the international JE and immediate right/ownership changes.
# No money is moved here and this effect never schedules a timer or creates country journal entries.
eafp_concession_refresh_effect = {
	clear_global_variable_list = concession_candidate_states_gvarlist
	every_state = {
		limit = { OR = { has_variable_list = concession_owner_varlist has_building = building_eafp_concession } }
		save_temporary_scope_as = eafp_refresh_state
		every_in_list = {
			variable = concession_owner_varlist
			save_temporary_scope_as = eafp_refresh_holder
			if = {
				limit = {
					OR = {
						is_country_alive = no
						this = scope:eafp_refresh_state.owner
						NOT = { has_company = company_type:company_eafp_concession }
						scope:eafp_refresh_state = {
							NOT = {
								any_scope_building = {
									is_building_type = building_eafp_concession
									levels_owned_by_country = { target = scope:eafp_refresh_holder value >= 1 }
								}
							}
						}
					}
				}
				scope:eafp_refresh_state = {
					remove_list_variable = { name = concession_owner_varlist target = prev }
				}
			}
		}
		if = {
			limit = { any_in_list = { variable = concession_owner_varlist always = yes } }
			add_to_global_variable_list = { name = concession_candidate_states_gvarlist target = this }
		}
		else_if = {
			limit = { has_building = building_eafp_concession }
			remove_building = building_eafp_concession
		}
	}
	every_country = {
		# Rights are already reconciled above. Removing a company never removes ordinary factories.
		if = {
			limit = {
				has_company = company_type:company_eafp_concession
				NOT = { eafp_concession_holder_has_rights = yes }
			}
			remove_company = company_type:company_eafp_concession
		}
		if = {
			limit = { has_variable = concession_loss_prestige_snapshot NOT = { has_modifier = eafp_concession_forced_loss } }
			remove_variable = concession_loss_prestige_snapshot
		}
		# The modifier follows the actual company, not prosperity or the right count.
		if = {
			limit = { has_company = company_type:company_eafp_concession }
			if = {
				limit = { NOT = { has_modifier = eafp_concession_company_slot } }
				add_modifier = { name = eafp_concession_company_slot }
			}
		}
		else = { remove_modifier = eafp_concession_company_slot }
	}
}
