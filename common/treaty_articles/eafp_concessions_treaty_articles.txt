eafp_grant_concession = {
	kind = directed
	# Influence cost, same as vanilla treaty_port (paid as maintenance by the target).
	cost = 50
	icon = "gfx/interface/icons/diplomatic_treaties_articles_icons/treaty_port_treaties.dds"
	# An executed, non-renegotiable article must not repeat its one-time effect.
	# hostile / ai_consider_threaten_naval_hostilities as on vanilla treaty_port, so the AI
	# treats it as a coercive demand it may back with naval threats.
	flags = { can_be_enforced hostile ai_consider_threaten_naval_hostilities }
	usage_limit = once_per_side_with_same_inputs
	maintenance_paid_by = target_country
	required_inputs = { state }

	# Conditions follow vanilla treaty_port (14_treaty_port.txt) minus its treaty-port-only
	# parts (coastal state, is_potential_treaty_port, treaty port province control, one
	# treaty port article per state). The state must be in China Proper with an
	# unrecognized owner (eafp_concession_eligible_state).
	unlocked_by_technologies = {
		colonization
	}

	visible = {
		is_revolutionary = no
	}

	possible = {
		any_scope_state = { eafp_concession_eligible_state = yes }
		scope:other_country = {
			NOT = { is_subject_of = root }
		}
	}

	requirement_to_maintain = {
		trigger = {
			exists = scope:article.input_state
		}
	}

	# The granting country must keep owning the state. After an ownership change the new
	# owner re-grants the right (eafp_concession_transfer_treaties_effect); the old owner's
	# article then counts as not fulfilled at once and the old owner withdraws from that
	# treaty like a breach (break penalties apply while it is binding). on_withdrawal does
	# not return the right, because the article's source is no longer the state's owner;
	# the treaty's other concession articles that are still valid get treaties of their own.
	non_fulfillment = {
		consequences = withdraw
		max_consecutive_contraventions = 0
		conditions = {
			weekly = {
				NOT = { scope:article.input_state ?= { owner = root } }
			}
		}
	}

	state_valid_trigger = {
		scope:input = {
			owner = root
			eafp_concession_can_receive = { HOLDER = scope:other_country }
		}
	}
	can_ratify = {
		scope:source_country = {
			country_rank <= scope:target_country.country_rank
		}
		custom_tooltip = {
			text = eafp_grant_concession_valid_tt
			OR = {
				# Keep the already executed article valid during treaty renegotiation.
				AND = {
					exists = scope:treaty.amended_treaty
					exists = scope:article.input_state
					scope:treaty.amended_treaty = {
						any_scope_article = {
							has_type = eafp_grant_concession
							source_country = scope:source_country
							target_country = scope:target_country
							input_state = scope:article.input_state
						}
					}
				}
				AND = {
					exists = scope:article.input_state
					scope:article.input_state = {
						owner = scope:source_country
						eafp_concession_can_receive = { HOLDER = scope:target_country }
					}
				}
			}
		}
	}
	# Validate the complete article list when proposing/answering the treaty.
	can_propose = {
		custom_tooltip = {
			text = eafp_concession_duplicate_article_tt
			trigger_if = {
				limit = { exists = scope:treaty exists = scope:article.input_state }
				NOT = {
					scope:treaty = {
						any_scope_article_option = {
							count > 1
							has_type = eafp_grant_concession
							source_country = scope:source_country
							target_country = scope:target_country
							input_state = scope:article.input_state
						}
					}
				}
			}
		}
	}
	# Same withdrawal rule as vanilla treaty_port (14_treaty_port.txt): the granting
	# state owner may withdraw only while it outranks the holder, and an unrecognized
	# owner cannot withdraw from a recognized holder. The holder may always withdraw.
	can_withdraw = {
		trigger_if = {
			limit = {
				scope:withdrawing_country = scope:source_country
			}
			scope:withdrawing_country = {
				country_rank > scope:non_withdrawing_country.country_rank
				OR = {
					NOT = { is_country_type = unrecognized }
					scope:target_country = { is_country_type = unrecognized }
				}
			}
		}
	}
	on_entry_into_force = {
		scope:article_options = {
			source_country = { save_scope_as = concession_host }
			target_country = { save_scope_as = concession_holder }
			input_state ?= {
				save_scope_as = concession_state
				eafp_concession_grant_effect = yes
			}
		}
		scope:concession_state = {
			if = {
				limit = {
					NOT = {
						has_building = building_trade_center
					}
				}
				create_building = {
					building = building_trade_center
					add_ownership = { company = { type = company_eafp_concession country = scope:concession_holder levels = 1 } }
				}
			}
		}
		# The concession JE activates by itself once this treaty is in force (its possible).
	}
	# Enforcement already executes on_entry_into_force.
	on_enforced = {}
	# Breaking a binding treaty also calls on_withdrawal; clean up only here.
	# Withdrawal exposes the existing article through scope:article.
	on_withdrawal = {
		scope:article = {
			target_country = { save_scope_as = concession_holder }
			input_state ?= {
				save_scope_as = concession_state
				# The state may have changed hands since the article entered into force.
				owner = { save_scope_as = concession_host }
				# After an ownership change the right is backed by a new treaty with the new
				# owner (eafp_concession_transfer_treaties_effect); the previous owner's
				# treaty no longer controls it, so withdrawing from that one returns nothing.
				if = {
					limit = { scope:article.source_country = scope:concession_host }
					# Another concession article of this treaty lapsed (its source lost its
					# state), so the treaty ends through non_fulfillment, not by choice. This
					# article is still valid: keep the right and back it with its own treaty.
					if = {
						limit = {
							eafp_concession_has_holder = { HOLDER = scope:concession_holder }
							scope:article.treaty ?= {
								any_scope_article = {
									has_type = eafp_grant_concession
									source_country = scope:concession_host
									NOT = { input_state ?= { owner = scope:concession_host } }
								}
							}
						}
						eafp_concession_create_treaty_effect = {
							HOST = scope:concession_host
							HOLDER = scope:concession_holder
							STATE = scope:concession_state
						}
					}
					else_if = {
						limit = {
							exists = scope:withdrawing_country
							scope:withdrawing_country = scope:concession_host
							scope:article.treaty.remaining_binding_period > 0
						}
						eafp_concession_return_effect = { FORCED = yes }
					}
					else = { eafp_concession_return_effect = { FORCED = no } }
				}
			}
		}
	}
	on_break = {}
	ai = {
		treaty_categories = { economy trade expansion }
		article_ai_usage = { request offer }
		# Great powers (the only possible receivers) consider concessions more often.
		evaluation_chance = {
			value = 0.1
			if = {
				limit = { country_rank >= rank_value:great_power }
				value = 0.3
			}
		}
		state_input_filter = {
			scope:input = {
				owner = root
				eafp_concession_can_grant = { HOLDER = scope:other_country }
			}
		}
		inherent_accept_score = {
			value = 0
			if = {
				limit = { exists = scope:article.input_state }
				add = { value = 30 desc = "EAFP_CONCESSION_AI_RIGHT" }
				add = { value = 20 desc = "EAFP_CONCESSION_AI_TRADE_CENTER" }
				# Bigger cities are worth more to the receiver and cost the giver more.
				add = {
					value = scope:article.input_state.eafp_concession_urban_levels
					multiply = 2
					max = 40
					desc = "EAFP_CONCESSION_AI_CITY"
				}
				if = {
					limit = { this = scope:source_country }
					multiply = -1
				}
			}
		}
		# How eagerly the AI picks this article as a war goal / play demand.
		wargoal_score_multiplier = {
			value = 1
			if = {
				limit = {
					this = scope:target_country
					country_rank >= rank_value:great_power
				}
				add = 1
				# Joining or leading the Second Opium War against the giver.
				if = {
					limit = {
						is_diplomatic_play_enemy_of = scope:source_country
						any_diplomatic_play = { is_diplomatic_play_type = dp_second_opium_war }
					}
					add = 2
				}
				# Britain should go for Tianjin in the Second Opium War (cf. vanilla Hong Kong).
				if = {
					limit = {
						c:GBR ?= this
						scope:article.input_state ?= { state_region = s:STATE_BEIJING }
						any_diplomatic_play = { is_diplomatic_play_type = dp_second_opium_war }
					}
					add = 3
				}
			}
		}
	}
	wargoal = {
		execution_priority = 21
		contestion_type = control_target_state
		# Cheaper to add than before (10) so AI great powers fit it into plays and sway offers.
		maneuvers = { value = 5 }
		infamy = { value = 5 }
	}
}
