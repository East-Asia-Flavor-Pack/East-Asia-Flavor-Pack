# 중chi_war_dynamism 버튼 공용 효과
# STATE: STATE_ 접두사를 뺀 주 지역 이름 (예: GUIZHOU)
eafp_chi_war_dynamism_button_effect = {
	custom_tooltip = {
		text = chi_war_dynamism_$STATE$_tt
		s:STATE_$STATE$ = {
			random_scope_state = {
				limit = { eafp_chi_war_dynamism_seizable_state_trigger = yes }
				add_modifier = {
					name = chi_war_dynamism_modifier
					days = 180
				}
				add_to_variable_list = { name = chi_war_dynamism_varlist target = ROOT }
				save_scope_as = target_state
			}
		}
		trigger_event = {
			id = chi_war_dynamism.1
			days = 181
		}
	}
	custom_tooltip = chi_war_dynamism_tt2
	show_as_tooltip = {
		s:STATE_$STATE$ = {
			random_scope_state = {
				limit = {
					NOT = { owner = ROOT }
					owner = { eafp_chi_war_dynamism_target_owner_trigger = yes }
				}
				save_scope_as = target_state2
			}
		}
		scope:target_state2 = {
			add_modifier = {
				name = recently_conquered_state
				days = long_modifier_time
				is_decaying = yes
			}
		}
	}
}
