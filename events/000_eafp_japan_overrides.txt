			add_bakuhantaisei_bakufu_authority = { VALUE = -200 }
		}
		if = {
			limit = {
				has_modifier = forced_market_opening
			}
			every_scope_character = {
				limit = {
					has_ideology = ideology:ideology_shojoi
				}
				add_modifier = {
					name = shogun_forced_to_open_market_character
					days = very_long_modifier_time

				}
			}
			random_political_movement = {
				limit = {
					is_political_movement_type = movement_tokugawa_loyalist
				}
				add_modifier = {
					name = shogun_forced_to_open_market
					days = long_modifier_time
					is_decaying = yes
				}
			}
		}
		if = {
			limit = {
				owner = {
					any_political_movement = {
						is_political_movement_type = movement_tokugawa_loyalist

					limit = {
						is_political_movement_type = movement_tokugawa_loyalist

#수정: 바닐라 보신전쟁 개시 이벤트에 EAFP 진영별 저널·지원 처리를 병합.
ep2_meiji.4 = { # Boshin War - Historical
	type = country_event
	placement = root

	duration = 3

	title = ep2_meiji.4.t
	desc = ep2_meiji.4.d
	flavor = ep2_meiji.4.f

    event_image = {
		video = "ep2_japanese_artillery"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_military.dds"

	cooldown = { days = long_modifier_time }

	cancellation_trigger = {

	}

	immediate = {
		random_country = {
			limit = {
				civil_war_origin_country ?= ROOT
				has_law = law_type:law_monarchy
				var:civil_war_type_var ?= movement_type:movement_tokugawa_loyalist
			}
			save_scope_as = tokugawa_scope
			ruler = {
				save_scope_as = shogun_scope
			}
		}
		random_interest_group = {
			limit = {
				is_in_government = yes
			}
			leader = {
				save_scope_as = councilor_scope
			}
		}

		#추가: 양 진영의 보신전쟁 저널과 지원을 이벤트에서 설정한다. 군주 지정은 바닐라 처리를 따른다.
		if = {
			limit = { exists = scope:tokugawa_scope }
			scope:tokugawa_scope = {
				if = {
					limit = { NOT = { has_journal_entry = je_boshin_war_sabaku } }
					add_journal_entry = {
						type = je_boshin_war_sabaku
						target = ROOT
					}
					set_variable = boshin_war_happened
				}
			}
			ROOT = {
				if = {
					limit = { NOT = { has_journal_entry = je_boshin_war_tobaku } }
					add_journal_entry = {
						type = je_boshin_war_tobaku
						target = scope:tokugawa_scope
					}
					set_state_religion = rel:confucian
					every_scope_state = {
						create_building = {
							building = "building_port"
							add_ownership = {
								country = {
									country = ROOT
									levels = 1
								}
							}
							reserves = 1
							activate_production_methods = { "pm_basic_port" }
						}
					}
					capital = {
						create_building = {
							building = "building_trade_center"
							activate_production_methods = { "pm_trade_center_trade_quantity_normal" }
							add_ownership = {
								country = {
									country = ROOT
									levels = 2
								}
							}
							reserves = 1
						}
					}
					every_scope_character = {
						limit = { is_bakufu_politician = yes }
						remove_bakufu_politician_role = yes
					}
					add_modifier = {
						name = modifier_japan_hiring_bonus
						months = 24
					}
				}
			}
		}
	}

	option = { # Play as the Emperor
		name = ep2_meiji.4.a
		default_option = yes
	}

	option = { # Play as the Tokugawa
		name = ep2_meiji.4.b
		trigger = {
			is_player = yes
		}
		play_as = scope:tokugawa_scope
	}
}

			is_political_movement_type = movement_meiji_restorationist

		}

			CULTURE_1 = primary_culture
		}

				owner = ROOT
				any_neighbouring_state = {
					is_treaty_port = yes
				}
			}
			save_scope_as = relevant_treaty_port_state
		}
		random_in_global_list = {
			variable = great_power_country_list
			limit = {
				has_strategic_region_interest_tier = {
					strategic_region = ROOT.capital.region
					value >= 1
				}
			}
			save_scope_as = relevant_gp
			random_primary_culture = {
				save_scope_as = relevant_gp_culture
			}
		}
		scope:relevant_gp ?= {
			create_character = {

				culture = primary_culture

				interest_group = ig_industrialists
