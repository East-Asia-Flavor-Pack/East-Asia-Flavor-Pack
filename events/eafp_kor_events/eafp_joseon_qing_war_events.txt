	option = {
		name = eafp_joseon_qing_war_events.3.a
		default_option = yes

		set_variable = failed_je_eafp_joseon_qing_war
		if = {
			limit = { var:eafp_var_jqw_result_grade = 3 }
			add_modifier = {
				name = eafp_modifier_joseon_qing_war_defeat_humiliation
				years = 3
			}
			add_radicals = {
				value = small_radicals
			}
		}
		else_if = {
			limit = { var:eafp_var_jqw_result_grade = 2 }
			add_modifier = {
				name = eafp_modifier_joseon_qing_war_defeat_humiliation
				years = 4
			}
			add_radicals = {
				value = medium_radicals
			}
		}
		else = {
			add_modifier = {
				name = eafp_modifier_joseon_qing_war_defeat_humiliation
				years = 6
			}
			add_radicals = {
				value = very_large_radicals
			}
		}
		if = {
			limit = {
				exists = c:CHI
			}
			change_relations = {
				country = c:CHI
				value = 20
			}
		}
		if = {
			limit = {
				var:eafp_var_jqw_result_grade <= 2
				has_modifier = eafp_modi_joseon_qing_war_covered_defeat
			}
			add_radicals = { value = small_radicals }
		}
		remove_variable = eafp_var_joseon_qing_war_months
		remove_variable = eafp_var_jqw_result
		remove_variable = eafp_var_jqw_result_grade
		remove_variable = eafp_var_jqw_kor_enforced_goal
		remove_variable = eafp_var_jqw_qing_enforced_goal
		remove_variable = eafp_var_jqw_kor_capitulated
		remove_variable = eafp_var_jqw_bad_shells_coverup
		remove_variable = eafp_var_jqw_transport_reformed

		ai_chance = {
			base = 100
		}
	}
}
