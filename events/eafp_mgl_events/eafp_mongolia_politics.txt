eafp_mgl.408 = {
	dlc = dlc_eafp
	type = country_event
	placement = root
	title = eafp_mgl.408.t
	desc = eafp_mgl.408.d
	flavor = eafp_mgl.408.f
	duration = 3
	event_image = { video = "unspecific_signed_contract" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_default.dds"
	trigger = { has_variable = eafp_mgl_invited_to_steppe_league is_eafp_mgl_steppe_league_candidate = yes }
	cancellation_trigger = { OR = { NOT = { exists = c:MGL } NOT = { has_journal_entry = je_eafp_mgl_echoes_of_chinggis } } }
	option = {
		name = eafp_mgl.408.a
		default_option = yes
		ai_chance = {
			base = 50
			modifier = { trigger = { relations:c:MGL >= relations_threshold:cordial } add = 30 }
			modifier = { trigger = { has_treaty_alliance_with = { TARGET = c:MGL } } add = 50 }
			modifier = { trigger = { has_treaty_defensive_pact_with = { TARGET = c:MGL } } add = 30 }
			modifier = { trigger = { has_treaty_support_independence_with = { TARGET = c:MGL } } add = 40 }
			modifier = { trigger = { is_subject = yes liberty_desire >= 50 } add = 20 }
			modifier = { trigger = { is_subject = yes liberty_desire < 15 } add = -30 }
			modifier = { trigger = { c:MGL = { infamy >= infamy_value:infamous } } add = -30 }
			modifier = { trigger = { c:MGL = { is_at_war = yes } } add = -20 }
			modifier = { trigger = { c:MGL = { has_law_or_variant = law_type:law_ethnostate } NOR = { country_has_primary_culture = cu:mongol country_has_primary_culture = cu:buryat } } add = -30 }
		}
		set_variable = eafp_mgl_steppe_league_acceptor
		set_variable = eafp_mgl_steppe_league_member
		if = { limit = { NOR = { has_treaty_alliance_with = { TARGET = c:MGL } has_treaty_defensive_pact_with = { TARGET = c:MGL } } } create_treaty = { first_country = root second_country = c:MGL is_draft = no binding_period = { years = 5 } articles_to_create = { { article = defensive_pact } } } }
		every_country = { limit = { NOT = { this = root } has_variable = eafp_mgl_steppe_league_member } trigger_event = { id = eafp_mgl.411 days = 1 popup = yes } }
		c:MGL ?= { change_relations = { country = root value = 20 } generate_eafp_mgl_steppe_league_member_list = yes }
	}
	option = {
		name = eafp_mgl.408.b
		ai_chance = { base = 40 }
		set_variable = eafp_mgl_steppe_league_immediate_rejector
		c:MGL ?= { change_relations = { country = root value = -10 } generate_eafp_mgl_steppe_league_member_list = yes }
	}
}
