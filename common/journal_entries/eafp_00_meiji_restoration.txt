# P0: Vanilla owns the canonical Meiji journal entries. This dormant companion
# preserves the old EAFP progress model without replacing any vanilla key.
je_eafp_jap_legacy_meiji_restoration = {
	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	group = je_group_meiji_restoration

	#is_shown_in_lobby = {
	#	c:JAP = THIS
	#}

	is_shown_when_inactive = {
		#exists = c:JAP
		#c:JAP = ROOT
		#NOT = { has_variable = meiji_restoration_complete }
		#has_law_or_variant = law_type:law_monarchy
	}

	possible = {
		always = no # 추가
		#has_law_or_variant = law_type:law_monarchy
		#NOT = { has_modifier = declared_bankruptcy }
		#ig:ig_landowners ?= {
		#	is_in_government = no
		#}
		#OR = {
		#	government_legitimacy >= 50
		#	ig:ig_landowners ?= { is_marginal = yes }
		#	has_revolution = yes
		#}
	}

	immediate = {
		set_variable = {
			name = shogunate_var
			value = 0
		}
	}

	is_progressing = {
		meiji_completion_trigger = yes
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					meiji_completion_trigger = yes
				}
				change_variable = {
					name = shogunate_var
					add = 1
				}
			}
		}
	}

	complete = {
		# If-else clause to avoid errors because of missing variable before JE activates
		trigger_if = {
			limit = { exists = var:shogunate_var }
			OR = {
				custom_tooltip = {
					text = defeat_shogunate_civil_war_tt
					scope:journal_entry = { is_goal_complete = yes }
				}
				AND = {
					scope:journal_entry = { is_goal_complete = yes }
					meiji_completion_trigger = yes
				}
			}
		}
		trigger_else = {
			custom_tooltip = {
				text = je_eafp_jap_legacy_meiji_restoration_goal
				OR = {
					custom_tooltip = defeat_shogunate_civil_war_tt
					meiji_completion_trigger = yes
				}
			}
		}
	}

	on_complete = {
		trigger_event = {
			id = eafp_jap_meiji_legacy.1
		}
	}

	on_fail = {
		if = {
			limit = { has_modifier = shogun_ig_forced_to_open_market }
			remove_modifier = shogun_ig_forced_to_open_market
		}
		if = {
			limit = { has_law_or_variant = law_type:law_monarchy }
			add_modifier = {
				name = honorable_restoration_failed
				days = very_long_modifier_time
				is_decaying = yes
			}
		}
	}

	fail = {
		OR = {
			has_modifier = declared_bankruptcy
			ig:ig_landowners ?= {
				is_in_government = yes
			}
			NOT = { has_law_or_variant = law_type:law_monarchy }
		}
	}

	current_value = {
		value = root.var:shogunate_var
	}

	goal_add_value = {
		value = 120
	}

	progressbar = yes

	weight = 10000
	should_be_pinned_by_default_uninvolved_or_context = yes
}
