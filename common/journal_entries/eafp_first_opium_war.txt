je_first_opium_war = {
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	group = je_group_foreign_affairs
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			NOT = { has_war_with = scope:target }
			OR = {
				scope:target ?= {
					has_law_or_variant = law_type:law_free_trade
				}
				has_treaty_foreign_investment_rights_with = { TARGET = scope:target }
				any_scope_state = {
					is_treaty_port = yes
					state_region = {
						any_scope_state = {
							owner = scope:target
						}
					}
				}
			}
		}
		effect = {
			c:CHI = {
				trigger_event = {
					id = first_opium_war.153
					days = 5
				}
			}
			c:GBR = {
				trigger_event = {
					id = first_opium_war.154
					days = 5
				}
			}
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = { has_war_with = scope:target }
			NOT = {
				OR = {
					scope:target ?= {
						has_law_or_variant = law_type:law_free_trade
					}
					has_treaty_foreign_investment_rights_with = { TARGET = scope:target }
					any_scope_state = {
						is_treaty_port = yes
						state_region = {
							any_scope_state = {
								owner = scope:target
							}
						}
					}
				}
			}
			NOT = {
				any_diplomatic_play = {
					initiator = c:GBR
					target = c:CHI
					is_diplomatic_play_type = dp_first_opium_war
				}
			}
		}
		effect = {
			c:CHI = {
				trigger_event = {
					id = first_opium_war.155
					days = 5
				}
			}
			c:GBR = {
				trigger_event = {
					id = first_opium_war.156
					days = 5
				}
			}
		}
	}
	outcome = {
		conclusion_desc = eafp_journal_invalid_conclusion_desc
		flags = { hidden invalid_outcome }
		trigger = {
			NOT = { exists = scope:target }
		}
		effect = {
			remove_variable = opium_wars_aggressor
		}
	}

	weight = 10000
	should_be_pinned_by_default_uninvolved_or_context = yes
}

je_first_opium_war_legalization = {
	icon = "gfx/interface/icons/goods_icons/opium.dds"
	group = je_group_qing

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			market = { mg:opium = { market_goods_pricier <= 0 } }
			scaled_gold_reserves >= 0.2
		}
		effect = {
			trigger_event = { id = first_opium_war.251 popup = yes }
		}
	}

	outcome = {
		conclusion_desc = eafp_journal_invalid_conclusion_desc
		flags = { hidden invalid_outcome }
		trigger = {
			OR = {
				NOT = { exists = c:CHI }
				NOT = { THIS = c:CHI }
				has_variable = opium_wars_target
				has_variable = opium_wars_gave_in
				AND = {
					OR = {
						has_law = law_type:law_canton_system
						has_law = law_type:law_isolationism
					}
					active_law:lawgroup_trade_policy ?= {
						NOT = {
							has_amendment = amendment_type:amendment_first_opium_war_legalization
						}
					}
				}
			}
		}
		effect = {
			if = {
				limit = {
					NOT = { has_variable = opium_wars_target }
					NOT = { has_variable = opium_wars_gave_in }
					OR = {
						has_law = law_type:law_canton_system
						has_law = law_type:law_isolationism
					}
					active_law:lawgroup_trade_policy ?= {
						NOT = {
							has_amendment = amendment_type:amendment_first_opium_war_legalization
						}
					}
				}
				add_modifier = {
					name = eafp_first_opium_war_legalization_repealed_decay
					days = normal_modifier_time
					is_decaying = yes
				}
				trigger_event = { id = first_opium_war.201 popup = yes }
			}
		}
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		color = "blue_negative"
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		flags = { timeout }
		effect = {
			trigger_event = { id = first_opium_war.201 popup = yes }
		}
	}
	timeout = 3600
	weight = 1000
	should_be_pinned_by_default_uninvolved_or_context = yes
}
