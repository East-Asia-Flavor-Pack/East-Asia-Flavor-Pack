# Rights, building levels and cap are all read from engine state; nothing is cached.
# Final urban groups from the installed 1.13 building-group inheritance tree.
# A single OR avoids counting a building again through its parent group.
eafp_concession_is_urban_building = {
	OR = {
		is_building_group = bg_manufacturing
		is_building_group = bg_service
		is_building_group = bg_urban_facilities
		is_building_group = bg_owner_buildings
	}
}

eafp_concession_eligible_state = {
	is_in_geographic_region = geographic_region_china_proper
	is_treaty_port = no
	owner = {
		OR = {
			c:CHI ?= this
			eafp_is_chinese_central_government = yes
			eafp_is_chinese_fragment = yes
		}
	}
}

# No stored right lists. A right is read live from two engine states:
#   treaty    - an in-force eafp_grant_concession article (target = holder, input = state);
#   ownership - the holder's levels in that state's concession building.
# Treaty checks enter article scope, so HOLDER and STATE must be saved scopes
# (scope:..., or root in a state-scoped scripted GUI), never this/prev.

# Country scope; a host or holder bound by an in-force concession article.
eafp_concession_involved_country = {
	any_scope_treaty = {
		any_scope_article = { has_type = eafp_grant_concession }
	}
}

# Any scope. True while an in-force concession article grants STATE to HOLDER.
# Country-scoped treaty lists contain in-force treaties only (global ones include drafts).
eafp_concession_has_treaty = {
	exists = $HOLDER$
	exists = $STATE$
	$HOLDER$ = {
		any_scope_treaty = {
			any_scope_article = {
				has_type = eafp_grant_concession
				target_country = $HOLDER$
				input_state = $STATE$
			}
		}
	}
}

# State scope; HOLDER is a country scope token.
# Duplicate in-force grants are prevented by the article's usage_limit, not by a list.
eafp_concession_can_grant = {
	eafp_concession_eligible_state = yes
	eafp_concession_free_capacity >= 5
	exists = $HOLDER$
	NOT = { owner = $HOLDER$ }
	$HOLDER$ = {
		is_country_alive = yes
		exists = capital
		country_rank >= rank_value:great_power
		has_interest_marker_in_region = prev.region
	}
}

# State scope (STATE is the same state as a saved scope). A right = treaty + ownership.
eafp_concession_has_holder = {
	exists = $HOLDER$
	$HOLDER$ = { is_country_alive = yes }
	NOT = { owner = $HOLDER$ }
	eafp_concession_has_treaty = { HOLDER = $HOLDER$ STATE = $STATE$ }
	any_scope_building = {
		is_building_type = building_eafp_concession
		levels_owned_by_country = { target = $HOLDER$ value >= 1 }
	}
}

# State scope (STATE is the same state as a saved scope).
eafp_concession_active = {
	eafp_concession_eligible_state = yes
	$HOLDER$ = { has_company = company_type:company_eafp_concession }
	eafp_concession_has_holder = { HOLDER = $HOLDER$ STATE = $STATE$ }
	owner.prestige <= $HOLDER$.prestige
}

# Called by delayed event options: never act on a different owner or right.
eafp_concession_event_valid = {
	exists = scope:concession_state
	exists = scope:concession_host
	exists = scope:concession_holder
	scope:concession_state = {
		owner = scope:concession_host
		eafp_concession_active = { HOLDER = scope:concession_holder STATE = scope:concession_state }
	}
}

# Country scope: only the war's initiator is evaluated here.
eafp_concession_can_start_return_war = {
	eafp_concession_event_valid = yes
	aggressive_diplomatic_plays_permitted = yes
	NOT = { has_war_with = scope:concession_holder }
	NOT = { is_diplomatic_play_enemy_of = scope:concession_holder }
	NOT = { has_truce_with = scope:concession_holder }
	NOT = { is_subject_of = scope:concession_holder }
	NOT = { is_diplomatic_play_participant_with = scope:concession_holder }
	NOT = { is_in_war_together = scope:concession_holder }
	NOT = {
		any_scope_treaty = {
			OR = { first_country = scope:concession_holder second_country = scope:concession_holder }
			any_scope_article = { OR = { has_type = alliance has_type = defensive_pact } }
		}
	}
	scope:concession_holder = {
		NOT = { is_subject_of = scope:concession_host }
	}
}

# State scope (STATE is the same state as a saved scope); shared by GUI validation and the effect.
eafp_concession_can_expand = {
	custom_tooltip = {
		text = eafp_concession_expand_active_tt
		eafp_concession_active = { HOLDER = $HOLDER$ STATE = $STATE$ }
	}
	custom_tooltip = {
		text = eafp_concession_expand_treaty_tt
		eafp_concession_has_treaty = { HOLDER = $HOLDER$ STATE = $STATE$ }
	}
	# Free capacity already subtracts levels waiting in the construction queue.
	custom_tooltip = {
		text = eafp_concession_expand_capacity_$LEVELS$_tt
		eafp_concession_free_capacity >= $LEVELS$
	}
	custom_tooltip = {
		text = eafp_concession_expand_treasury_tt
		$HOLDER$ = {
			has_company = company_type:company_eafp_concession
			gold_reserves >= eafp_concession_expansion_cost_levels_$LEVELS$
		}
	}
	custom_tooltip = {
		text = eafp_concession_expand_ownership_tt
		any_scope_building = {
			is_building_type = building_eafp_concession
			levels_owned_by_country = { target = $HOLDER$ value >= 1 }
		}
	}
	custom_tooltip = {
		text = eafp_concession_expand_queue_tt
		any_scope_building = {
			is_building_type = building_eafp_concession
			can_queue_building_levels = $LEVELS$
		}
	}
}

# Any scope; HOLDER is a saved country scope. Unlike involvement, this excludes hosts:
# HOLDER must be the target of an in-force concession article.
eafp_concession_holder_has_rights = {
	exists = $HOLDER$
	$HOLDER$ = {
		any_scope_treaty = {
			any_scope_article = {
				has_type = eafp_grant_concession
				target_country = $HOLDER$
			}
		}
	}
}
