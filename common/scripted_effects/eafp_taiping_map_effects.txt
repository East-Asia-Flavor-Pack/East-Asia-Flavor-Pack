# 태평천국 분쇄(je_taiping) / 태평의 혁명(je_taiping_revolution) 맵모드 툴팁. root = state.
eafp_taiping_map_tooltip = {
	if = {
		limit = {
			exists = c:TPG
			exists = c:CHI
			OR = {
				eafp_taiping_map_counts_for_tpg = yes
				eafp_taiping_map_counts_for_chi = yes
			}
		}
		c:TPG = { save_scope_as = eafp_taiping_map_tpg }
		c:CHI = { save_scope_as = eafp_taiping_map_chi }
		if = {
			limit = { eafp_taiping_map_tpg_recent = yes }
			custom_tooltip_no_bullet = "eafp_taiping_map_tpg_recent_tt"
		}
		else_if = {
			limit = { eafp_taiping_map_tpg_occupied = yes }
			custom_tooltip_no_bullet = "eafp_taiping_map_tpg_occupied_tt"
		}
		else_if = {
			limit = { eafp_taiping_map_chi_recent = yes }
			custom_tooltip_no_bullet = "eafp_taiping_map_chi_recent_tt"
		}
		else = {
			custom_tooltip_no_bullet = "eafp_taiping_map_chi_occupied_tt"
		}
		if = {
			limit = { eafp_taiping_map_counts_for_tpg = yes }
			custom_tooltip_no_bullet = "eafp_taiping_map_counts_for_tpg_tt"
		}
		else = {
			custom_tooltip_no_bullet = "eafp_taiping_map_counts_for_chi_tt"
		}
		if = {
			limit = {
				c:TPG = {
					has_variable = num_unincorporated_states_in_china_proper
					has_variable = num_claimed_states_in_china_proper
				}
			}
			custom_tooltip_no_bullet = "eafp_taiping_map_summary_tt"
		}
	}
}
