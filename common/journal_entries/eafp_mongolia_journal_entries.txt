je_eafp_mgl_restoration_of_the_state = {
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"
	group = je_group_historical_content
	should_be_pinned_by_default_uninvolved_or_context = yes
	weight = 260
	transferable = no

	widget = {
		gui = "gui/journal_entry_widgets/eafp_mongolia_widgets.gui"
		name = "widget_je_eafp_mgl_jebtsundamba"
		container = "custom_widget_container_je_icon"
	}

	is_shown_when_inactive = { c:MGL ?= ROOT }
	possible = {
		c:MGL ?= ROOT
		NOT = { has_variable = eafp_mgl_restoration_started }
	}
	immediate = {
		set_variable = eafp_mgl_restoration_started
		# History creates the character; do not spawn it here before character history loads.
		random_scope_character = {
			limit = { has_template = eafp_mgl_fifth_jebtsundamba_character_template }
			owner = { set_variable = { name = eafp_mgl_jebtsundamba_var value = PREV } }
		}
	}
	on_weekly_pulse = {
		effect = { eafp_mgl_update_jebtsundamba = yes }
	}
	on_monthly_pulse = {
		events = { eafp_mgl.100 }
	}

	complete = { is_subject = no }
	on_complete = {
		trigger_event = { id = eafp_mgl.103 popup = yes }
		trigger_event = { id = eafp_mgl.210 days = 90 popup = yes }
	}
}

je_eafp_mgl_echoes_of_chinggis = {
	icon = "gfx/interface/icons/event_icons/event_trade.dds"
	scripted_button = je_eafp_mgl_invite_steppe_league_candidates_button
	scripted_button = je_eafp_mgl_leave_steppe_league_button
	group = je_group_historical_content
	should_be_pinned_by_default_uninvolved_or_context = yes
	weight = 240
	transferable = no
	is_shown_when_inactive = {
		c:MGL ?= ROOT
		is_player = yes
	}

	possible = {
		is_subject = no
		has_technology_researched = pan-nationalism
		bureaucracy >= 0
		country_rank >= rank_value:unrecognized_regional_power
		OR = {
			has_variable = eafp_mgl_unite_the_mongols_complete
			eafp_mgl_united_state_count >= 6
		}
		NOT = { is_at_war = yes }
		any_scope_state = { filter = { is_in_geographic_region = geographic_region_eafp_mgl_heartland } percent >= 0.75 turmoil < 0.2 }
		NOT = { has_variable = eafp_mgl_steppe_congress_started }
	}

	immediate = {
		set_variable = eafp_mgl_steppe_congress_started
		set_variable = eafp_mgl_steppe_league_member
		trigger_event = { id = eafp_mgl.500 days = 7 popup = yes }
	}

	on_weekly_pulse = { effect = { generate_eafp_mgl_steppe_league_member_list = yes } }

	fail = {
		OR = {
			custom_tooltip = {
				text = eafp_mgl_steppe_league_rejected_tt
				has_variable = eafp_mgl_steppe_league_immediate_rejector
			}
			custom_tooltip = {
				text = eafp_mgl_steppe_league_lost_eligibility_tt
				AND = { NOT = { c:MGL ?= this } has_variable = eafp_mgl_invited_to_steppe_league NOT = { has_variable = eafp_mgl_steppe_league_member } NOT = { is_eafp_mgl_steppe_league_candidate = yes } }
			}
			AND = { NOT = { c:MGL ?= this } is_subject = yes NOT = { top_overlord = c:MGL } }
			AND = { c:MGL ?= this is_subject = yes }
		}
	}
	on_fail = {
		remove_variable = eafp_mgl_invited_to_steppe_league
		remove_variable = eafp_mgl_steppe_league_acceptor
		remove_variable = eafp_mgl_steppe_league_member
		generate_eafp_mgl_steppe_league_member_list = yes
	}

	complete = {
		c:MGL ?= this
		custom_tooltip = {
			text = eafp_mgl_steppe_charter_selected_tt
			has_variable = eafp_mgl_steppe_charter_ratified
		}
		custom_tooltip = {
			text = eafp_mgl_steppe_charter_ratifiers_tt
			any_country = {
				NOT = { c:MGL ?= this }
				has_variable = eafp_mgl_steppe_league_member
				has_variable = eafp_mgl_steppe_charter_ratifier
				count >= 2
			}
		}
	}
	on_complete = {
		trigger_event = { id = eafp_mgl.409 days = 1 popup = yes }
	}
}

		eafp_mgl_united_state_count >= 2

	immediate = {

			name = eafp_mgl_united_gdp_goal

		trigger_event = { id = eafp_mgl.520 days = 7 popup = yes }
	}

			10 = eafp_mgl.511
			10 = eafp_mgl.512
			10 = eafp_mgl.513
			10 = eafp_mgl.514
			10 = eafp_mgl.521
			10 = eafp_mgl.522
			10 = eafp_mgl.523
			10 = eafp_mgl.524
			10 = eafp_mgl.525
		}
	}
	complete = {
		eafp_mgl_united_state_count >= 2
		custom_tooltip = {
			text = eafp_mgl_united_gdp_growth_goal_tt
			has_variable = eafp_mgl_united_gdp_goal
			eafp_mgl_united_gdp >= var:eafp_mgl_united_gdp_goal
		}
		custom_tooltip = {
			text = eafp_mgl_united_raw_reserves_goal_tt
			eafp_mgl_united_raw_building_count > 0
			eafp_mgl_united_healthy_raw_building_share >= 0.75
		}

			text = eafp_mgl_united_industry_reserves_goal_tt
			eafp_mgl_united_industry_building_count > 0
			eafp_mgl_united_healthy_industry_building_share >= 0.75
		}
	}
	on_complete = {
		remove_variable = eafp_mgl_united_gdp_goal
		set_variable = eafp_mgl_industrial_combine_unlocked
		add_modifier = { name = eafp_mgl_industrial_takeoff years = 10 }
		trigger_event = { id = eafp_mgl.519 days = 1 popup = yes }
		add_modifier = { name = eafp_mgl_united_industrial_market years = 10 }
		trigger_event = { id = eafp_mgl.526 days = 1 popup = yes }
	}
	on_invalid = { remove_variable = eafp_mgl_united_gdp_goal }
}
