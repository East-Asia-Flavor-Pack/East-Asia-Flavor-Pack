										owner = scope:eafp_qing_treaty_partner
										is_capital = no
									}
									owner = {
										is_subject_type = subject_type_military_headquarters_china
										is_subject_of_and_not_being_freed = scope:eafp_qing_treaty_partner
									}
								}
								# EAFP requires an undivided Amur/Outer Manchuria; Central Asian slices are transferred individually.
								OR = {
									NOR = {
										state_region = s:STATE_AMUR
										state_region = s:STATE_OUTER_MANCHURIA
									}
									is_split_state = no
								}
							}
						}
					}
				} # 추가
				country_is_valid_qing_treaty_partner = yes
			}
			save_scope_as = donor_scope
		}
		if = {
			limit = {
				scope:donor_scope = {
					owes_obligation_to = ROOT
				}
			}
			custom_tooltip = obligation_for_amur_tt
		}
		else_if = {
			limit = {
				scope:donor_scope = {
					is_subject_of = ROOT
				}
			}
			custom_tooltip = transfer_of_amur_tt
		}
		else = {
			if = {
				limit = {
					exists = scope:donor_scope
				}
				c:RUS ?= {
					show_as_tooltip = { # 수정
						create_incident = {
							country = scope:donor_scope
							value = 10
						}
					}
				}
			}
			custom_tooltip = transfer_of_amur_tt
		}
		set_variable = rus_bought_amur
		trigger_event = { id = russo_chinese.4 popup = yes }
	}
}
