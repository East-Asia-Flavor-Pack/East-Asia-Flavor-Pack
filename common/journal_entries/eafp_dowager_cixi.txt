je_dowager_cixi = { # dowager_cixi_events.1에서 트리거
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	group = je_group_qing

	is_shown_in_lobby = {
		c:CHI ?= THIS
	}

	scripted_button = remove_cixi_button
	scripted_button = reduce_cixi_power_authority_button
	scripted_button = reduce_cixi_power_bureaucracy_button
	scripted_button = reduce_cixi_power_money_button

	immediate = {
		random_scope_character = {
			limit = { has_template = eafp_chi_xingzhen_yehe_nara_template }
			save_scope_as = Xingzhen_Yehe_Nara_scope
			set_character_busy_and_immortal = yes
			trigger_event = {
				id = dowager_cixi_events.31
				months = 1 # 테스트
			}
		}
		set_variable = {
			name = cixi_coup
			value = 0 # 1이 되면 반란
		}
		# cixi_coup_d는 script_value에서.
	}
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = je_dowager_cixi_goal
				scope:Xingzhen_Yehe_Nara_scope = {
					has_variable = cixi_complete
				}
			}
		}
		effect = {
			trigger_event = dowager_cixi_events.4
			hidden_effect = {
				remove_variable = cixi_coup
			}
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = cixi_not_removed
				scope:Xingzhen_Yehe_Nara_scope = {
					has_variable = cixi_fail
				}

			}
			# A fulfilled objective wins if both ending conditions are met.
			hidden_trigger = {
				NOT = {
					AND = {
						custom_tooltip = {
							text = je_dowager_cixi_goal
							scope:Xingzhen_Yehe_Nara_scope = {
								has_variable = cixi_complete
							}
						}

					}
				}
			}
		}
		effect = {
			trigger_event = dowager_cixi_events.3
			hidden_effect = {
				remove_variable = cixi_coup
			}
		}
	}

	outcome = {
		conclusion_desc = eafp_journal_invalid_conclusion_desc
		flags = { hidden invalid_outcome }
		trigger = {
			custom_tooltip = {
				text = china_shatters_tt
				has_global_variable = china_shatters
			}
		}
		effect = {
			hidden_effect = {
				remove_variable = cixi_coup
			}
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_dowager_cixi_status
				trigger = {
					always = yes
				}
			}
		}
	}
	on_monthly_pulse = {
		effect = {
			change_variable = {
				name = cixi_coup
				add = cixi_coup_d
			}
			clamp_variable = {
				name = cixi_coup
				max = 1
				min = 0
			}
			if = {
				limit = { var:cixi_coup >= 1 }
				set_variable = {
					name = cixi_coup
					value = 0
				}
				trigger_event = { id = dowager_cixi_events.5 }
			}
			# dowager_cixi_events.6 내용을 ai 용으로 더 완화
			if = {
				limit = { is_ai = yes }
				if = {
					limit = {
						scope:Xingzhen_Yehe_Nara_scope = {
							has_ideology = ideology:ideology_cixi1
						}
						var:china_warlord_explosion_sum <= 0.4
					}
					scope:Xingzhen_Yehe_Nara_scope = {
						set_ideology = ideology:ideology_cixi2
					}
				}
				if = {
					limit = {
						scope:Xingzhen_Yehe_Nara_scope = {
							has_ideology = ideology:ideology_cixi2
						}
						var:china_warlord_explosion_sum <= 0.2
					}
					scope:Xingzhen_Yehe_Nara_scope = {
						set_ideology = ideology:ideology_cixi3
					}
				}
			}
		}
		random_events = {
			100 = 0
			# TODO
		}
		events = {
			dowager_cixi_events.6
		}
	}

	current_value = {
		value = var:cixi_coup
	}

	goal_add_value = {
		add = 1
	}

	progressbar = yes

	weight = 700

	should_be_pinned_by_default_uninvolved_or_context = yes

	transferable = no
	can_revolution_inherit = no
}

je_remove_cixi = {
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	group = je_group_qing

	immediate = {
		set_variable = {
			name = remove_cixi_progress
			value = 0
		}
		random_scope_character = {
			limit = { has_template = eafp_chi_xingzhen_yehe_nara_template }
			save_scope_as = Xingzhen_Yehe_Nara_scope
		}
	}
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			scope:journal_entry = {
				is_goal_complete = yes
			}
			scope:Xingzhen_Yehe_Nara_scope.popularity <= ruler.popularity
		}
		effect = {
			show_as_tooltip = {
				trigger_event = dowager_cixi_events.4
			}
			hidden_effect = {
				scope:Xingzhen_Yehe_Nara_scope = {
					set_variable = cixi_complete
				}
				remove_variable = remove_cixi_progress
			}
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			scope:Xingzhen_Yehe_Nara_scope.popularity > ruler.popularity
		}
		effect = {
			hidden_effect = {
				remove_variable = remove_cixi_progress
			}
		}
	}

	outcome = {
		conclusion_desc = eafp_journal_invalid_conclusion_desc
		flags = { hidden invalid_outcome }
		trigger = {
			NOT = { has_journal_entry = je_dowager_cixi }
		}
		effect = {
			hidden_effect = {
				remove_variable = remove_cixi_progress
			}
		}
	}

	on_weekly_pulse = {
		effect = {
			change_variable = {
				name = remove_cixi_progress
				add = 1
			}
		}
	}

	current_value = {
		value = var:remove_cixi_progress
	}

	goal_add_value = {
		add = 12
	}

	progressbar = yes

	weight = 699

	should_be_pinned_by_default_uninvolved_or_context = yes

	transferable = no
	can_revolution_inherit = no
}
