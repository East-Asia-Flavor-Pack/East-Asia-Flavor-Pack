je_chinese_rule = {
	activation_description = {
		always = yes
	}

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	group = je_group_historical_content

	immediate = {
		if = {
			limit = { NOT = { has_variable = eafp_china_military_headquarters_governor_term } }
			eafp_china_reset_military_headquarters_governor_term = yes
		}
	}

	on_yearly_pulse = {
		effect = {
			eafp_china_military_headquarters_governor_yearly_effect = yes
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			OR = {
				NOT = { is_subject_type = subject_type_military_headquarters_china }
				NOT = { is_subject_of = c:CHI }
			}
		}
		effect = {
			if = {
				limit = { has_law = law_type:law_eafp_military_headquarters }
				activate_law = law_type:law_autocracy
			}
			if = {
				limit = { c:MGL ?= THIS }
				if = {
					limit = {
						NOT = { has_variable = eafp_mgl_restoration_started }
						NOT = { has_journal_entry = je_eafp_mgl_restoration_of_the_state }
					}
					add_journal_entry = { type = je_eafp_mgl_restoration_of_the_state }
				}
			}
			else = {
				if = {
					limit = { country_has_primary_culture = cu:han }
					remove_primary_culture = cu:han
				}
				if = {
					limit = { country_has_primary_culture = cu:manchu }
					remove_primary_culture = cu:manchu
				}
				hidden_effect = {
					if = {
						limit = { any_scope_character = { is_heir_of_own_country = yes } }
						heir = { kill_character = { hidden = yes } }
					}
					ruler = { kill_character = { hidden = yes } }
				}
			}

			if = {
				limit = {
					is_subject_type = subject_type_military_headquarters_china
				}
				change_subject_type = subject_type_vassal
			}

			remove_variable = eafp_china_military_headquarters_governor_term
		}
	}

	weight = 500
	should_be_pinned_by_default_uninvolved_or_context = yes

	transferable = no
}
