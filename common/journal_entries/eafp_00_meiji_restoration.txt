REPLACE:je_meiji_restoration = {
	icon = "gfx/interface/icons/faction_icons/faction_japan_imperial_house.dds"

	group = je_group_meiji_restoration

	widget = {
		gui = "gui/journal_entry_widgets/ep2_japan_widgets.gui"
		name = "widget_je_meiji_restoration_emperor"
		container = "custom_widget_container_je_icon"
	}

	#수정: 바닐라의 일본 다이묘 목록을 EAFP의 번과 다이묘 위젯으로 교체.
	#widget = {
	#	gui = "gui/journal_entry_widgets/ep2_japan_widgets.gui"
	#	name = "widget_je_meiji_restoration_daimyos_list"
	#	container = "custom_widget_container_3"
	#}
	widget = {
		gui = "gui/journal_entry_widgets/eafp_je_bakuhantaisei.gui"
		name = "widget_je_bakuhantaisei_japan_regions"
		container = "custom_widget_container_3"
	}

	is_shown_in_lobby = {
		c:JAP = THIS
	}

	is_shown_when_inactive = {
		c:JAP ?= THIS
		has_law = law_type:law_monarchy
		has_law = law_type:law_bakufu
		NOT = { has_variable = japan_restoration_complete }
	}

	possible = {
		has_law = law_type:law_monarchy
		has_law = law_type:law_bakufu
		NOT = { has_law_or_variant = law_type:law_isolationism }
	}

	immediate = {
		eafp_japan_ensure_reform_journal = yes #추가: 개혁 일지가 없고 완료하지 않은 경우에만 생성.
		je_meiji_restoration_update_japanese_emperor = yes
		je_meiji_restoration_update_daimyos = yes
		country_calculate_and_cache_daimyo_loyalties_per_state = yes
		if = {
			limit = {
				c:JAP ?= this # Revolters don't get this
			}
			trigger_event = { id = ep2_meiji.1 }
		}
		if = {
			limit = {
				NOT = {
					any_political_movement = {
						is_political_movement_type = movement_meiji_restorationist
					}
				}
			}
			create_political_movement = { type = movement_meiji_restorationist }
		}
		random_political_movement = {
			limit = {
				is_political_movement_type = movement_meiji_restorationist
			}
			save_scope_as = meiji_movement_scope
		}
		ig:ig_landowners ?= {
			save_scope_as = daimyo_ig
		}
		set_variable = { name = restoration_timer_var value = 0 }
	}

	on_weekly_pulse = {
		effect = {
			country_calculate_and_cache_daimyo_loyalties_per_state = yes # We love good script practises
			je_meiji_restoration_update_japanese_emperor = yes
			je_meiji_restoration_update_daimyos = yes
		}
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					has_variable = restoration_timer_var
					ruler ?= {
						has_trait = special_character_japanese_emperor
					}
					NOT = {
						any_civil_war = {
							civil_war_progress > 0
						}
					}
					NOT = {
						has_law = law_type:law_bakufu
					}
					ig:ig_landowners ?= {
						leader ?= {
							NOT = {
								has_variable = house_tokugawa
							}
						}
					}
				}
				change_variable = { name = restoration_timer_var add = 1 }
			}
			else = {
				set_variable = { name = restoration_timer_var value = 0 }
			}
		}
		#수정: 정국 사건은 막부 개혁 일지가 한 번만 호출한다.
	}

	# Imperial Victory
	custom_completion_header = je_meiji_restoration_imperial_victory_header
	complete = { eafp_japan_imperial_victory = yes }

	custom_on_completion_header = je_meiji_restoration_on_imperial_victory_header
	on_complete = {
		#추가: 사건 효과가 현재 정국을 바꾸기 전에 천황 승리 상태를 기록.
		set_variable = japan_emperor_restored
		set_variable = japan_restoration_complete
		trigger_event = {
			id = meiji.1
		}
		if = {
			limit = {
				is_player = no
				c:EZO ?= {
					is_player = no
					is_subject_of = ROOT
					NOT = {
						has_social_hierarchy = edo_system
					}
				}
			}
			annex = c:EZO
		}
	}

	event_outcome_completed_effect_desc = {
		effect = {
			if = {
				limit = {
					any_scope_state = {
						state_region = s:STATE_KANTO
					}
				}
				set_capital = STATE_KANTO

				s:STATE_KANTO = {
					random_scope_state = {
						set_hub_name = {
							type = city
							name = HUB_NAME_STATE_KANTO_city_meiji
						}
					}
				}
			}
			ig:ig_landowners ?= {
				show_as_tooltip = {
					remove_ideology = ideology_japan_hierarchic_2
					add_ideology = ideology_hierarchic
				}
				set_interest_group_name = ig_kazoku
			}
			hidden_effect = { country_clear_daimyo = yes }
			if = {
				limit = {
					NOT = {
						any_political_movement = {
							is_political_movement_type = movement_modernizer
						}
					}
				}
				create_political_movement = { type = movement_modernizer }
			}
			japan_display_add_meiji_reforms = yes
		}
	}

	event_outcome_completed_effect_desc = {
		header = meiji.1.a
		effect = {
			ig:ig_industrialists ?= {
				add_modifier = {
					name = meiji_favored_ig
					days = very_long_modifier_time
				}
			}
		}
	}

	event_outcome_completed_effect_desc = {
		header = meiji.1.b
		effect = {
			ig:ig_intelligentsia ?= {
				add_modifier = {
					name = meiji_favored_ig
					days = very_long_modifier_time
				}
			}
		}
	}

	#수정: 결과 이벤트와 보상은 막부 개혁에서 이미 처리했다.
	custom_failure_header = je_meiji_restoration_shogunate_victory_header
	fail = { eafp_japan_shogunate_victory_completed = yes }
	custom_on_failure_header = je_meiji_restoration_on_shogunate_victory_header
	on_fail = { }

	invalid = {
		NOT = {
			country_has_monarchy_law = yes
		}
	}

	on_invalid = {
		set_variable = japan_restoration_complete
		if = {
			limit = {
				has_variable = emperor_var
			}
			remove_variable = emperor_var
		}
		if = {
			limit = {
				is_player = no
				c:EZO ?= {
					is_player = no
					is_subject_of = ROOT
					NOT = {
						has_social_hierarchy = edo_system
					}
				}
			}
			annex = c:EZO
		}
	}

	weight = 10000
	should_be_pinned_by_default_uninvolved_or_context = yes
	can_revolution_inherit = yes
}

#수정: 사전 노선 선택과 혼인 실패의 강제 전환 제거.
REPLACE:je_meiji_imperial_marriage = {
	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	group = je_group_meiji_restoration

	immediate = {
		ruler = {
			save_scope_as = imperial_marriage_ruler
		}
		random_scope_character = {
			limit = {
				has_variable = japan_imperial_princess
			}
			save_scope_as = imperial_princess_scope
		}
		ordered_scope_treaty = {
			limit = {
				binds = ROOT
				any_scope_article = {
					OR = {
						has_type = trade_privilege
						has_type = foreign_investment_rights
						has_type = law_commitment
					}
				}
			}
			order_by = {
				value = "ROOT.ai_treaty_value(scope:source_country)"
				multiply = -1
			}
			position = 0
			save_scope_as = relevant_treaty_scope
		}
		if = {
			limit = {
				exists = scope:relevant_treaty_scope
			}
			scope:journal_entry ?= {
				set_variable = japan_break_treaty_var
			}
		}
	}

	complete = {
		has_journal_entry = je_bakufu_kaikaku
		ruler = { has_variable = japan_imperial_marriage }
		trigger_if = {
			limit = {
				scope:journal_entry ?= {
					has_variable = japan_break_treaty_var
				}
			}
			trigger_if = {
				limit = {
					exists = scope:relevant_treaty_scope
				}
				custom_tooltip = {
					text = has_broken_relevant_treaty_tt
					NOT = {
						any_scope_treaty = {
							scope:relevant_treaty_scope ?= this
						}
					}
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = im_broken_applicable_treaty_tt
					NOT = {
						exists = scope:relevant_treaty_scope
					}
				}
			}
		}
		has_law_or_variant = law_type:law_closed_borders
		NOT = {
			has_law_or_variant = law_type:law_free_trade
		}
		NOT = {
			any_state_in_home_islands = {
				is_treaty_port = yes
			}
		}
	}

	on_complete = {
		custom_tooltip = {
			text = kobu_gattai_progress_tt
			set_variable = japan_completed_imperial_marriage
		}
	}

	fail = {
		OR = {
			custom_tooltip = {
				text = ruler_does_not_have_imperial_marriage_tt
				ruler = {
					NOT = {
						has_variable = japan_imperial_marriage
					}
				}
			}
			custom_tooltip = {
				text = je_has_timed_out_tt
				NOT = {
					has_journal_entry = je_meiji_imperial_marriage
				}
			}
		}
	}

	on_fail = {
		eafp_japan_clear_imperial_marriage = yes
		custom_tooltip = eafp_japan_marriage_ended_tt
	}

	on_timeout = {
		eafp_japan_clear_imperial_marriage = yes
		custom_tooltip = eafp_japan_marriage_ended_tt
	}

	#수정: 상위 정치 일지 종료는 혼인 과제만 정리한다.
	invalid = {
		OR = {
			NOT = { has_journal_entry = je_bakufu_kaikaku }
			NOT = { country_has_monarchy_law = yes }
		}
	}
	on_invalid = { eafp_japan_clear_imperial_marriage = yes }

	timeout = 1825

	weight = 10000
	should_be_pinned_by_default_uninvolved_or_context = yes
	can_revolution_inherit = no
}
