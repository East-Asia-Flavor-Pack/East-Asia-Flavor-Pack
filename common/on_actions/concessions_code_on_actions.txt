eafp_concessions_state_changed = {
	trigger = {
		OR = {
			has_variable_list = concession_owner_varlist
			is_in_geographic_region = geographic_region_china_proper
		}
	}
	effect = {
		# At ownership transfer, union only pieces now owned by the same country.
		# Both possible surviving states receive the union before an engine merge.
		# Other countries' split states and treaty ports are never copied here.
		if = {
			limit = { exists = owner is_treaty_port = no }
			owner = { save_temporary_scope_as = eafp_merge_owner }
			state_region = {
				every_scope_state = {
					limit = { owner = scope:eafp_merge_owner is_treaty_port = no }
					save_temporary_scope_as = eafp_merge_destination
					state_region = {
						every_scope_state = {
							limit = {
								owner = scope:eafp_merge_owner
								is_treaty_port = no
								NOT = { this = scope:eafp_merge_destination }
							}
							eafp_concession_inherit_state_rights = { TARGET = scope:eafp_merge_destination }
						}
					}
				}
			}
		}
		# A split can leave non-multiple-of-5 shares in this state and its sibling pieces;
		# round every owner to the nearest multiple of 5 before rights are reconciled.
		state_region = {
			every_scope_state = {
				limit = { has_building = building_eafp_concession }
				eafp_concession_round_ownership_effect = yes
			}
		}
		eafp_concession_refresh_effect = yes
		# Remaining holders get a concession treaty with the state's new owner.
		eafp_concession_transfer_treaties_effect = yes
	}
}

# ROOT is the old country and target the newly formed revolt.
# Only copy a split state's rights to this specific revolt in the same region.
eafp_concessions_civil_war_start = {
	effect = {
		every_scope_state = {
			limit = { has_variable_list = concession_owner_varlist }
			save_temporary_scope_as = eafp_split_source
			state_region = {
				every_scope_state = {
					limit = { owner = scope:target is_treaty_port = no }
					save_temporary_scope_as = eafp_split_target
					scope:eafp_split_source = {
							eafp_concession_inherit_state_rights = { TARGET = scope:eafp_split_target }
					}
				}
			}
		}
		eafp_concession_refresh_effect = yes
	}
}
