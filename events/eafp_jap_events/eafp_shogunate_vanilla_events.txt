REPLACE:shogunate.6 = { # Accession of Iemochi
	type = country_event
	placement = root

	duration = 3

	title = shogunate.6.t
	desc = shogunate.6.d
	flavor = shogunate.6.f

    gui_window = event_window_1char_tabloid

	left_icon = scope:shogun_scope

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/misc/1Character_Banner"

	dlc = dlc018

	icon = "gfx/interface/icons/faction_icons/faction_japan_tokugawa.dds"

	cooldown = { days = never_fire_again_modifier_time }

	trigger = {
		has_dlc_feature = ep2_content
		# Triggered by on_new_ruler
	}

	cancellation_trigger = {

	}

	immediate = {
		random_scope_character = {
			limit = {
				has_template = JAP_iemochi_tokugawa
			}
			save_scope_as = shogun_scope
		}
		random_scope_character = {
			limit = {
				has_trait = special_character_japanese_emperor
			}
			save_scope_as = emperor_scope
		}
		if = {
			limit = {
				scope:shogun_scope ?= {
					age < 16
				}
			}
#수정 섭정 후보 요시노부가 사망했으면 재생성하지 않음.
# 			if = {
# 				limit = {
# 					NOT = {
# 						any_scope_character = {
# 							has_template = JAP_yoshinobu_tokugawa
# 						}
# 					}
# 				}
# 				create_character = {
# 					template = JAP_yoshinobu_tokugawa
# 				}
# 			}
# 			random_scope_character = {
# 				limit = {
# 					has_template = JAP_yoshinobu_tokugawa
# 				}
# 				save_scope_as = yoshinobu_scope
# 			}
			#수정
			eafp_japan_prepare_shogun_candidate = { TEMPLATE = JAP_yoshinobu_tokugawa BIRTH_DATE = 1837.10.28 CANDIDATE = yoshinobu_scope }
		}
	}

	option = {
		name = shogunate.6.a
		default_option = yes
		trigger = {
			scope:shogun_scope ?= {
				is_adult = yes
			}
		}
		scope:shogun_scope ?= {
			set_first_name = Iemochi
		}
		show_as_tooltip = {
			scope:shogun_scope = {
				set_character_as_ruler = yes
			}
		}
	}

	option = {
		name = shogunate.6.a_regency
		default_option = yes
		trigger = {
			scope:shogun_scope ?= {
				is_adult = no
			}
		}
		scope:shogun_scope ?= {
			set_first_name = Iemochi
		}
		if = {
			limit = {
				scope:shogun_scope ?= {
					is_adult = no
				}
			}
			add_regency_modifier = yes
			set_variable = { name = hide_ruler_change_notification days = 3 }
			create_character = {
				template = JAP_yoshiyori_tokugawa
				on_created = {
					set_character_as_ruler = yes
					set_variable = { name = regency_years value = 0 }
				}
			}
			set_heir = scope:shogun_scope
		}
	}

	option = {
		name = shogunate.6.b_regency
		trigger = {
			scope:shogun_scope ?= {
				is_adult = no
			}
			#추가 살아 있는 섭정 후보가 있을 때만 선택 가능.
			exists = scope:yoshinobu_scope
			scope:yoshinobu_scope = { is_character_alive = yes }
		}
		scope:shogun_scope ?= {
			set_first_name = Iemochi
		}
		if = {
			limit = {
				scope:shogun_scope ?= {
					is_adult = no
				}
			}
			add_regency_modifier = yes
			set_variable = { name = hide_ruler_change_notification days = 3 }
			scope:yoshinobu_scope ?= {
				set_character_as_ruler = yes
				set_variable = { name = regency_years value = 0 }
			}
			set_heir = scope:shogun_scope
		}
	}

	after = {
		if = {
			limit = {
				scope:shogun_scope ?= {
					has_variable = daimyo_var
				}
			}
			scope:shogun_scope ?= {
				character_clear_daimyo_status = yes
			}
			create_character = {
				template = JAP_tokugawa_mochitsugu
			}
		}
	}
}
