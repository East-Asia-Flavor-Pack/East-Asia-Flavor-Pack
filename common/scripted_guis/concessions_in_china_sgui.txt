	}

# A holder's own action only. The international journal has no owner country.
eafp_concession_expand_gui = {
	scope = state
	is_shown = { eafp_concession_has_holder = { HOLDER = scope:actor } }
	is_valid = { eafp_concession_can_expand = { HOLDER = scope:actor } }
	effect = {
		scope:actor = { save_temporary_scope_as = concession_holder }
		# Named in the effect line below ([SCOPE.sState('concession_state')]).
		save_temporary_scope_as = concession_state
		# The overwrite steps would flood the auto tooltip; one effect line describes them.
		custom_tooltip = {
			text = eafp_concession_expand_effect_tt
			eafp_concession_expand_effect = yes
		}
		# Display only: the fee is charged inside eafp_concession_expand_effect, and only
		# after the levels were added. This renders the vanilla treasury effect line.
		show_as_tooltip = {
			scope:actor = { add_treasury = { value = eafp_concession_expansion_cost multiply = -1 } }
		}
	}
}
