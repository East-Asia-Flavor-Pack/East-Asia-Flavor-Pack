# Character scope: the historical shogun, never a regent.
eafp_japan_faction_succession_ruler = {
	is_ruler_of_own_country = yes
	owner = {
		c:JAP ?= THIS
		has_law = law_type:law_bakufu
		has_gov_regency = no
	}
	OR = {
		has_template = JAP_ieyoshi_tokugawa
		has_template = JAP_iesada_tokugawa
		has_template = JAP_iemochi_tokugawa
		has_template = JAP_yoshinobu_tokugawa
	}
}

# Country scope. A used template without a living local character is not recreated.
eafp_japan_shogun_candidate_available = {
	game_date >= $BIRTH_DATE$
	OR = {
		any_scope_character = {
			has_template = $TEMPLATE$
			is_character_alive = yes
			NOT = { is_ruler_of_own_country = yes }
		}
		NOT = { is_template_used = $TEMPLATE$ }
	}
}
