# Internal bridge for scripted monthly incidents. Normal demands use treaty articles.
eafp_concession_incident_return = {
	icon = "gfx/interface/icons/war_goals/enforce_treaty_article.dds"
	kind = custom
	target_type = country
	contestion_type = control_own_capital
	settings = { require_target_be_part_of_war }
	possible = { always = no }
	valid = {
		exists = scope:target_state
		exists = scope:target_country
		scope:target_state = {
			owner = root
			eafp_concession_eligible_state = yes
			eafp_concession_has_holder = { HOLDER = scope:target_country }
		}
	}
	maneuvers = { value = 10 }
	infamy = { value = 0 }
	on_enforced = {
		root = { save_scope_as = concession_host }
		scope:target_country = { save_scope_as = concession_holder }
		scope:target_state ?= {
			save_scope_as = concession_state
			eafp_concession_return_effect = { FORCED = yes }
		}
	}
	ai = { is_significant_demand = no }
}

# Lens demand modelled on vanilla take_treaty_port: the holder demands a concession in a
# state of the target. Enforcement (accepted demand or won play) creates a real treaty with
# the eafp_grant_concession article, so the right stays treaty-backed (can_withdraw,
# on_withdrawal return) and on_entry_into_force grants the 5 levels and opens the JE.
eafp_demand_concession = {
	icon = "gfx/interface/icons/war_goals/take_treaty_port.dds"
	kind = custom
	settings = {
		require_target_be_part_of_war
		can_add_for_other_country
		requires_interest
	}
	execution_priority = 21
	contestion_type = control_target_state
	target_type = state
	possible = {
		country_rank >= rank_value:great_power
		scope:target_state = {
			owner = scope:target_country
			eafp_concession_can_grant = { HOLDER = root }
		}
	}
	valid = {
		exists = scope:target_state
		scope:target_state = {
			owner = scope:target_country
			eafp_concession_can_grant = { HOLDER = root }
		}
	}
	maneuvers = {
		value = 0
		add = {
			desc = "MANEUVERS_BASE_VALUE"
			value = 5
		}
		multiply = {
			desc = "MANEUVERS_TARGET_STATE_POPULATION_FACTOR"
			value = scope:target_state.state_population
			divide = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_FACTOR
			multiply = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_MULTIPLIER
			max = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_MULTIPLIER_MAX_PER_STATE
		}
	}
	infamy = { value = 5 }
	on_enforced = {
		create_treaty = {
			name = eafp_concession_treaty_name
			first_country = scope:target_country
			second_country = root
			is_draft = no
			binding_period = { years = 5 }
			articles_to_create = {
				{
					article = eafp_grant_concession
					source_country = scope:target_country
					target_country = root
					inputs = {
						{ state = scope:target_state }
					}
				}
			}
		}
	}
	ai = { is_significant_demand = no }
}
