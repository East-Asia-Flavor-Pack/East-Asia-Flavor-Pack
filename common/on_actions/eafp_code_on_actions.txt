eafp_on_diplo_play_back_down = {
	trigger = {
		scope:actor ?= {
			has_journal_entry = je_eastasian_expansionism
		}
	}
	effect = {
		scope:actor ?= {
			je:je_eastasian_expansionism = {
				add_progress = {
					name = eastasian_expansionism_paranoia_progress_bar
					value = 20
				}
			}
		}
	}
}

eafp_on_diplomatic_demand_accepted = {
	trigger = {
		scope:recipient ?= {
			has_journal_entry = je_eastasian_expansionism
		}
	}
	effect = {
		scope:recipient ?= {
			je:je_eastasian_expansionism = {
				add_progress = {
					name = eastasian_expansionism_paranoia_progress_bar
					value = 20
				}
			}
		}
	}
}

eafp_on_wargoal_enforced = {
	# Final enforcement only; the occupation timer also fires before the peace settlement.
	effect = {
		if = {
			limit = {
				scope:target ?= {
					has_journal_entry = je_eastasian_expansionism
				}
			}
			scope:target ?= {
				je:je_eastasian_expansionism = {
					add_progress = {
						value = scope:wargoal_impact
						name = eastasian_expansionism_paranoia_progress_bar
					}
				}
			}
		}
		if = {
			limit = {
				scope:actor ?= {
					has_journal_entry = je_eastasian_expansionism
				}
			}
			scope:actor ?= {
				je:je_eastasian_expansionism = {
					add_progress = {
						value = {
							value = scope:wargoal_impact
							multiply = -1
						}
						name = eastasian_expansionism_paranoia_progress_bar
					}
				}
			}
		}
	}
}
