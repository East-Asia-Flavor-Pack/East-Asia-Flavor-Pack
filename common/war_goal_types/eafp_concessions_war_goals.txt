# Internal bridge for scripted monthly incidents. Normal demands use treaty articles.
eafp_concession_incident_return = {
	icon = "gfx/interface/icons/war_goals/enforce_treaty_article.dds"
	kind = custom
	target_type = country
	contestion_type = control_own_capital
	settings = { require_target_be_part_of_war }
	possible = { always = no }
	valid = {
		exists = scope:target_state
		exists = scope:target_country
		scope:target_state = {
			owner = root
			eafp_concession_eligible_state = yes
			eafp_concession_has_holder = { HOLDER = scope:target_country }
		}
	}
	maneuvers = { value = 10 }
	infamy = { value = 0 }
	on_enforced = {
		root = { save_scope_as = concession_host }
		scope:target_country = { save_scope_as = concession_holder }
		scope:target_state ?= {
			save_scope_as = concession_state
			eafp_concession_return_effect = yes
		}
	}
	ai = { is_significant_demand = no }
}
