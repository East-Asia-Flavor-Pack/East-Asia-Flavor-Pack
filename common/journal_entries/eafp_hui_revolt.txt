	group = je_group_qing

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			NOT = {
				exists = c:PNG
			}
		}
		effect = {
			trigger_event = { id = hui_revolt.6 }
		}
		event_outcome_effect_desc = {
			header = hui_revolt.6.a
			effect = {
				s:STATE_YUNNAN.region_state:CHI = {
					kill_population_percent_in_state = { percent = 0.025 }
					add_devastation = 20
				}
			}
		}
		event_outcome_effect_desc = {
			header = hui_revolt.6.b
			effect = {
				s:STATE_YUNNAN.region_state:CHI = {
					kill_population_percent_in_state = { percent = 0.05 }
					add_devastation = 20
					custom_tooltip = { text = hui_revolt.6.ttb }
				}
			}
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			exists = c:PNG
			has_truce_with = c:PNG
		}
		effect = {
			if = {
				limit = { has_journal_entry = je_warlord_china }
				post_notification = warlord_china_explosion_point_remove
				change_variable = {
					name = china_warlord_explosion
					subtract = 0.25
				}
			}
		}
	}

	weight = 400
	should_be_pinned_by_default_uninvolved_or_context = yes

	transferable = no
}
