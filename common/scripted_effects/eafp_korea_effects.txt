eafp_land_problem_add_modifier = {	#상태등급별 효과 업데이트
	eafp_land_problem_add_modifier_comp = { comp = adm }
	eafp_land_problem_add_modifier_comp = { comp = civ }

eafp_land_problem_add_modifier_comp = {
	remove_modifier = eafp_modi_land_problem_$comp$_lvl_0
	remove_modifier = eafp_modi_land_problem_$comp$_lvl_1
	remove_modifier = eafp_modi_land_problem_$comp$_lvl_2
	remove_modifier = eafp_modi_land_problem_$comp$_lvl_3
	remove_modifier = eafp_modi_land_problem_$comp$_lvl_4
	remove_modifier = eafp_modi_land_problem_$comp$_lvl_5

	if = {
		limit = {
			eafp_val_land_problem_$comp$ <= eafp_val_land_problem_threshold1
		}
		add_modifier = {
			name = eafp_modi_land_problem_$comp$_lvl_0
			months = 3
		}
	}
	else_if = {
		limit = {
			eafp_val_land_problem_$comp$ <= eafp_val_land_problem_threshold2
			eafp_val_land_problem_$comp$ >= eafp_val_land_problem_threshold1
		}
		add_modifier = {
			name = eafp_modi_land_problem_$comp$_lvl_1
			months = 3
		}
	}
	else_if = {
		limit = {
			eafp_val_land_problem_$comp$ <= eafp_val_land_problem_threshold3
			eafp_val_land_problem_$comp$ >= eafp_val_land_problem_threshold2
		}
		add_modifier = {
			name = eafp_modi_land_problem_$comp$_lvl_2
			months = 3
		}
	}
	else_if = {
		limit = {
			eafp_val_land_problem_$comp$ <= eafp_val_land_problem_threshold4
			eafp_val_land_problem_$comp$ >= eafp_val_land_problem_threshold3
		}
		add_modifier = {
			name = eafp_modi_land_problem_$comp$_lvl_3
			months = 3
		}
	}
	else_if = {
		limit = {
			eafp_val_land_problem_$comp$ <= eafp_val_land_problem_threshold5
			eafp_val_land_problem_$comp$ >= eafp_val_land_problem_threshold4
		}
		add_modifier = {
			name = eafp_modi_land_problem_$comp$_lvl_4
			months = 3
		}
	}
	else = {
		add_modifier = {
			name = eafp_modi_land_problem_$comp$_lvl_5
			months = 3
		}
	}

}

scramble_for_korea_change_influence = {
	if = {
		limit = {
			is_involved_in_journal_entry = je_scramble_for_korea
			NOT = { has_global_variable = scramble_for_korea_finished }
		}
		if = {
			limit = { NOT = { has_variable = scramble_for_korea_participants_influence } }
			set_variable = { name = scramble_for_korea_participants_influence value = 0 }
		}
		change_variable = { name = scramble_for_korea_participants_influence add = $AMOUNT$ }
		clamp_variable = { name = scramble_for_korea_participants_influence min = 0 max = 100 }
	}
}
# Called on departure and at both terminal paths; button cooldowns deliberately survive departure.
scramble_for_korea_clear_country = {
	remove_variable = scramble_for_korea_participants_influence
	if = {
		limit = { has_modifier = modifier_korea_scramble_advisers }
		remove_modifier = modifier_korea_scramble_advisers
	}
}
# Shared weekly/monthly JE synchronization. Membership is always changed first.
scramble_for_korea_sync = {
	every_country = {
		if = {
			limit = {
				is_involved_in_journal_entry = je_scramble_for_korea
				NOT = { scramble_for_korea_eligible = yes }
			}
			scope:journal_entry = { remove_involved_country = prev }
			scramble_for_korea_clear_country = yes
		}
		else_if = {
			limit = {
				NOT = { is_involved_in_journal_entry = je_scramble_for_korea }
				scramble_for_korea_eligible = yes
			}
			scope:journal_entry = { add_involved_country = prev }
		}
		if = {
			limit = { is_involved_in_journal_entry = je_scramble_for_korea }
			if = {
				limit = { NOT = { has_variable = scramble_for_korea_participants_influence } }
				set_variable = { name = scramble_for_korea_participants_influence value = 0 }
			}
		}
		else_if = {
			limit = { has_variable = scramble_for_korea_participants_influence }
			scramble_for_korea_clear_country = yes
		}
	}
	# Clear pending offers left by the removed intervention event chain.
}
