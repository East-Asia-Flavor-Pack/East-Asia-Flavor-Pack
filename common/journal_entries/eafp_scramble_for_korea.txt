je_scramble_for_korea = {
	activation_description = {
		exists = c:CHI
		NOT = { has_global_variable = china_shatters }
		OR = {
		    c:KOR ?= { is_subject_of = c:CHI }
		    c:CHI ?= { is_country_type = recognized }
		}
		exists = c:JAP
		c:JAP ?= {
		    has_variable = japan_restoration_complete
		    is_country_type = recognized
		}
		exists = c:KOR
		c:KOR ?= { is_country_type = unrecognized }
		NOT = { has_global_variable = scramble_for_korea_finished }
	}

    icon = "gfx/interface/icons/event_icons/event_map.dds"
    group = je_group_global_international_situations
    is_shown_when_inactive = { NOT = { has_global_variable = scramble_for_korea_finished } }
    possible = {
        exists = c:CHI
        NOT = { has_global_variable = china_shatters }
        OR = {
            c:KOR ?= { is_subject_of = c:CHI }
            c:CHI ?= { is_country_type = recognized }
        }
        exists = c:JAP
        c:JAP ?= {
            has_variable = japan_restoration_complete
            is_country_type = recognized
        }
        exists = c:KOR
        c:KOR ?= { is_country_type = unrecognized }
        NOT = { has_global_variable = scramble_for_korea_finished }
    }
    should_be_involved = { scramble_for_korea_eligible = yes }
    should_show_when_not_involved = {
        OR = {
            c:KOR ?= this
            c:CHI ?= this
            c:JAP ?= this
            country_rank >= rank_value:great_power
        }
    }
    widget = {
        gui = "gui/journal_entry_widgets/eafp_je_scramble_for_korea.gui"
        name = "eafp_korea_scramble_influence"
        container = "custom_widget_container_3"
    }
    immediate = {
        c:KOR = {
            save_scope_as = korea_scope
            add_modifier = { name = modifier_kor_foreign_influence }
        }
        c:KOR ?= { trigger_event = { id = scramble_for_korea_events.1 popup = yes } }
        c:CHI ?= { trigger_event = { id = scramble_for_korea_events.2 popup = yes } }
        c:JAP ?= { trigger_event = { id = scramble_for_korea_events.2 popup = yes } }
    }
    immediate_all_involved = {
        set_variable = { name = scramble_for_korea_participants_influence value = 0 }
        if = {
            limit = { c:KOR ?= this }
            # Each existing completion is imported once, only at JE activation.
            if = { limit = { has_variable = completed_je_korean_reformation_society } change_variable = { name = scramble_for_korea_participants_influence add = 5 } }
            if = { limit = { has_variable = completed_je_korean_reformation_military } change_variable = { name = scramble_for_korea_participants_influence add = 5 } }
            if = { limit = { has_variable = completed_je_korean_reformation_economy } change_variable = { name = scramble_for_korea_participants_influence add = 5 } }
            if = { limit = { has_variable = completed_je_korean_reformation_diplomacy } change_variable = { name = scramble_for_korea_participants_influence add = 5 } }
            if = { limit = { has_variable = korean_reformation_complete_var } change_variable = { name = scramble_for_korea_participants_influence add = 10 } }
        }
        else = { set_variable = { name = scramble_for_korea_participants_influence value = 10 } }
        if = {
            limit = {
                NOR = {
                    c:KOR ?= THIS
                    c:CHI ?= THIS
                    c:JAP ?= THIS
                }
            }
            trigger_event = { id = scramble_for_korea_events.3 popup = yes }
        }
    }
    on_become_involved_after_activation = {
        set_variable = { name = scramble_for_korea_participants_influence value = 0 }
        if = {
            limit = { c:KOR ?= this }
            trigger_event = { id = scramble_for_korea_events.1 popup = yes }
        }
        else_if = {
            limit = { OR = { c:CHI ?= this c:JAP ?= this } }
            trigger_event = { id = scramble_for_korea_events.2 popup = yes }
        }
        else = { trigger_event = { id = scramble_for_korea_events.3 popup = yes } }
    }
    on_no_longer_involved = { scramble_for_korea_clear_country = yes }
    on_weekly_pulse = { effect = { scramble_for_korea_sync = yes } }
    on_monthly_pulse = {
        effect = {
            scramble_for_korea_sync = yes
            # Korea reads all foreign scores before any foreign score changes.
            c:KOR = { scramble_for_korea_change_influence = { AMOUNT = scramble_for_korea_monthly_change } }
            scope:journal_entry = {
                every_scope_je_involved = {
                    limit = { NOT = { c:KOR ?= this } }
                    scramble_for_korea_change_influence = { AMOUNT = scramble_for_korea_monthly_change }
                }
            }
        }
    }
    outcome = {
		header = je_scramble_for_korea_foreign_victory_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
	        OR = {
	            scramble_for_korea_overlord_wins = yes
	            any_country = {
	                scramble_for_korea_score_winner = yes
	            }
	        }
	    }
		effect = {
	        custom_tooltip = korea_scramble_foreign_victory_effect_tt
	        show_as_tooltip = {
	            scope:journal_entry = {
	                random_scope_je_involved = { annex = c:KOR }
	            }
	        }
	        hidden_effect = {
	            if = {
	                limit = { scramble_for_korea_overlord_wins = yes }
	                c:KOR ?= { overlord = { save_scope_as = scramble_winner } }
	            }
	            else = {
	                random_country = {
	                    limit = { scramble_for_korea_score_winner = yes }
	                    save_scope_as = scramble_winner
	                }
	            }
	            set_global_variable = scramble_for_korea_finished
	            # Clear contest-only support before applying the winner's lasting rewards.
	            scramble_for_korea_cleanup = yes
	            scope:scramble_winner ?= {
	                annex = c:KOR
	            }

	            scope:journal_entry = {
	                every_scope_je_involved = { scramble_for_korea_notify_result = yes }
	            }
	            # 저널 참여 X인 국가가 종주국이 됐을 경우
	            if = {
	                limit = {
	                    scope:journal_entry = {
	                        NOT = { any_scope_je_involved = { scope:scramble_winner ?= THIS } }
	                    }
	                }
	                scope:scramble_winner ?= { scramble_for_korea_notify_result = yes }
	            }
	        }
	    }
	}

    outcome = {
		header = je_scramble_for_korea_korean_victory_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
	        c:KOR ?= { is_subject = no }
	        OR = {
	            scope:journal_entry = {
	                any_scope_je_involved = {
	                    count = all
	                    THIS = c:KOR
	                }
	            }
	            c:KOR ?= {
	                scramble_for_korea_score_winner = yes
	                is_country_type = recognized
	            }
	        }
			# A fulfilled objective wins if both ending conditions are met.
			hidden_trigger = {
				NOT = {
					AND = {
						OR = {
						    scramble_for_korea_overlord_wins = yes
						    any_country = {
						        scramble_for_korea_score_winner = yes
						    }
						}

					}
				}
			}
		}
		effect = {
	        c:KOR = { save_scope_as = scramble_winner }
	        set_global_variable = scramble_for_korea_finished
	        scramble_for_korea_cleanup = yes
	        c:KOR = { add_modifier = { name = modifier_korea_scramble_sovereignty days = 1825 } }
	        scope:journal_entry = {
	            every_scope_je_involved = { scramble_for_korea_notify_result = yes }
	        }
	    }
	}

    outcome = {
		conclusion_desc = eafp_journal_invalid_conclusion_desc
		flags = { hidden invalid_outcome }
		trigger = {
	        NOT = { exists = c:KOR }
	    }
		effect = {
	        set_global_variable = scramble_for_korea_finished
	        scramble_for_korea_cleanup = yes
	    }
	}

    progressbar = no
    weight = 200
    should_be_pinned_by_default_involved = yes
    should_be_pinned_by_default_uninvolved_or_context = no
}
