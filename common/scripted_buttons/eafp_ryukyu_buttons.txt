button_je_eafp_ryukyu_dispatch_envoys = {
	name = button_je_eafp_ryukyu_dispatch_envoys
	desc = button_je_eafp_ryukyu_dispatch_envoys_desc

	visible = {
		c:KOR ?= this
		has_journal_entry = je_eafp_ryukyu_intervention
		has_variable = eafp_ryukyu_intervention_var
	}

	cooldown = { days = 365 }

	possible = {
		exists = c:RYU
		custom_tooltip = {
			text = eafp_ryukyu_independence_progress_not_complete_tt
			has_variable = eafp_ryukyu_intervention_progress_var
			var:eafp_ryukyu_intervention_progress_var < 100
		}
		NOT = { has_law_or_variant = law_type:law_isolationism }
	}

	effect = {
		add_treasury = { value = credit multiply = -0.01 }
		change_variable = {
			name = eafp_ryukyu_intervention_progress_var
			add = 10
		}
		c:JAP ?= {
			change_relations = {
				country = ROOT
				value = -10
			}
		}
		c:CHI ?= {
			change_relations = {
				country = ROOT
				value = -5
			}
		}
		custom_tooltip = eafp_ryukyu_dispatch_envoys_effect_tt
	}

	ai_chance = {
		value = 5
	}
}
