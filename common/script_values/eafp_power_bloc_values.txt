INJECT:ai_power_bloc_identity_score = {
	### Chinese Interregnum: warlords seek membership, while other countries avoid it and leave if already admitted.
	if = {
		limit = {
			scope:actor.power_bloc ?= {
				has_identity = identity:identity_eafp_chinese_interregnum
			}
		}
		if = {
			limit = { eafp_is_chinese_fragment = yes }
			add = {
				value = 1000
				desc = "DIPLOMATIC_ACCEPTANCE_FROM_IDENTITY"
			}
		}
		else = {
			add = {
				value = -1000
				desc = "DIPLOMATIC_ACCEPTANCE_FROM_IDENTITY"
			}
		}
	}
}
