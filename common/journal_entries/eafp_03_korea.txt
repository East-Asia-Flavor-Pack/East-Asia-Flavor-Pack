	activation_description = {
		always = yes
	}

	should_be_pinned_by_default_uninvolved_or_context = no

	icon = "gfx/interface/icons/event_icons/event_fire.dds"

	group = je_group_global_international_situations

		OR = {
			c:KOR ?= this
			civil_war_origin_country ?= c:KOR
		}

	}

	should_show_when_not_involved = {
		OR = {
			has_strategic_region_interest_tier = {
				strategic_region = sr:region_northeast_asia
				value >= 3
			}
			capital ?= {
				is_in_geographic_region = geographic_region_northeast_asia
			}
		}
	}

	# Added by event

	immediate = {
		c:KOR ?= {
			save_scope_as = korea_scope
			ruler = {
				save_scope_as = korea_ruler_scope
			}
			overlord = {
				save_scope_as = china_scope
			}
		}
		c:JAP ?= {
			save_scope_as = japan_scope
		}
		random_country = {
			limit = {
				civil_war_origin_country = scope:korea_scope
			}
			save_scope_as = korea_rebellion_scope
		}
	}

	outcome = {
		header = je_korean_rebellion_government_victory
		effect_header = je_korean_rebellion_on_government_victory

		conclusion_desc = eafp_je_korean_rebellion_complete_conclusion_desc
		trigger = {
			scope:korea_scope ?= {
				has_revolution = no
				custom_tooltip = {
					text = korean_ruler_retained_tt
					scope:korea_ruler_scope = {
						is_ruler_of_own_country = yes
					}
				}
			}
		}
		effect = {
			if = {
				limit = {
					scope:japan_scope ?= {
						is_involved_in_journal_entry = je_korean_rebellion
					}
					c:KOR ?= {
						is_subject = yes
						scope:china_scope ?= overlord
					}
				}
				custom_tooltip = {
					text = japan_demands_korea_tt
					scope:japan_scope ?= {
						trigger_event = { id = gg_korea.8 days = 7 popup = yes}
					}
				}
			}

				every_scope_je_involved = {
					limit = {
						OR = {
							scope:china_scope ?= this
							scope:japan_scope ?= this
						}
						has_variable = sided_with_korean_gov
					}
					add_modifier = {
						name = modifier_victory_in_korean_intervention
						days = long_modifier_time
					}
				}
				every_scope_je_involved = {
					limit = {
						OR = {
							scope:china_scope ?= this
							scope:japan_scope ?= this
						}
						has_variable = sided_with_korean_rebels
					}
					add_modifier = {
						name = modifier_failed_korean_intervention
						days = long_modifier_time
					}
				}
			}
		}
	}

	outcome = {
		header = je_korean_rebellion_rebel_victory
		effect_header = je_korean_rebellion_on_rebel_victory
		color = "blue_negative"
		conclusion_desc = eafp_je_korean_rebellion_fail_conclusion_desc

			c:KOR ?= {
				has_revolution = no

					text = korean_ruler_deposed_tt

						NOT = { exists = scope:korea_ruler_scope }
						scope:korea_ruler_scope ?= {
							is_ruler_of_own_country = no
						}
					}

			}

								text = korean_ruler_retained_tt

			scope:journal_entry ?= {
				every_scope_je_involved = {
					limit = {
						OR = {
							scope:china_scope ?= this
							scope:japan_scope ?= this

					}

					}
				}

					limit = {
						OR = {
							scope:china_scope ?= this
							scope:japan_scope ?= this
						}
						has_variable = sided_with_korean_gov
					}
					add_modifier = {
						name = modifier_failed_korean_intervention
						days = long_modifier_time
					}
				}
