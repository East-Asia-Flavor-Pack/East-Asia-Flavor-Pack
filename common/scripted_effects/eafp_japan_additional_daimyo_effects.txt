# Character scope. Historical registration never rewinds a han's succession counter.
eafp_japan_register_han_daimyo = {
	if = {
		limit = { NOT = { has_global_variable = japan_daimyo_abolished } }
		set_variable = { name = daimyo_var value = s:STATE_$STATE$ }
		set_variable = { name = daimyo_han_var value = flag:$HAN$ }
		s:STATE_$STATE$ = {
			if = {
				limit = {
					OR = {
						NOT = { has_variable = eafp_daimyo_chain_id_$HAN$ }
						var:eafp_daimyo_chain_id_$HAN$ < $INDEX$
					}
				}
				set_variable = { name = eafp_daimyo_chain_id_$HAN$ value = $INDEX$ }
			}
		}
	}
}

# Country scope. The state-region counter is unique to this han and survives civil wars.
eafp_japan_generate_kumamoto_daimyo = {
	if = {
		limit = {
			NOT = { has_global_variable = japan_daimyo_abolished }
			any_scope_state = { state_region = s:STATE_KYUSHU }
		}
		s:STATE_KYUSHU = {
			if = {
				limit = { NOT = { has_variable = eafp_daimyo_chain_id_kumamoto } }
				set_variable = { name = eafp_daimyo_chain_id_kumamoto value = 0 }
			}
		}
		# Skip used historical people, including dead, retired and foreign characters.
		# Each iteration advances the counter; at most the historical roster's length.
		while = {
			limit = {
				OR = {
					AND = {
						s:STATE_KYUSHU = { var:eafp_daimyo_chain_id_kumamoto = 0 }
						is_template_used = eafp_jap_hosokawa_yoshikuni_kumamoto
					}
					AND = {
						s:STATE_KYUSHU = { var:eafp_daimyo_chain_id_kumamoto = 1 }
						is_template_used = eafp_jap_hosokawa_morihisa_kumamoto
					}
				}
			}
			s:STATE_KYUSHU = { change_variable = { name = eafp_daimyo_chain_id_kumamoto add = 1 } }
		}
		if = {
			limit = {
				s:STATE_KYUSHU = { var:eafp_daimyo_chain_id_kumamoto = 0 }
				game_date >= 1835.7.23
			}
			create_character = { template = eafp_jap_hosokawa_yoshikuni_kumamoto }
		}
		else_if = {
			limit = {
				s:STATE_KYUSHU = { var:eafp_daimyo_chain_id_kumamoto = 1 }
				game_date >= 1839.4.14
			}
			create_character = { template = eafp_jap_hosokawa_morihisa_kumamoto }
		}
		else = { eafp_japan_generate_random_kumamoto_daimyo = yes }
	}
}
