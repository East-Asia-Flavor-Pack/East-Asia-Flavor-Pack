# Building income is deliberately weekly_profit, as approved by the user.
# Add all urban profits first, including losses; clamp only the state total.
eafp_concession_state_profit = {
	value = 0
	every_scope_building = {
		limit = { eafp_concession_is_urban_building = yes }
		add = weekly_profit
	}
}
# Incident compensation only (paid by the holder to the host in events).
# There is no recurring treasury transfer from the Chinese owner to holders.
eafp_concession_state_share = {
	value = eafp_concession_state_profit
	min = 0
	multiply = 0.05
}

# Read actual building levels; never cache the cap or scale in persistent variables.
eafp_concession_total_levels = {
	value = 0
	every_scope_building = {
		limit = { is_building_type = building_eafp_concession }
		add = level
	}
}
eafp_concession_urban_levels = {
	value = 0
	every_scope_building = {
		limit = { is_building_type = building_urban_center }
		add = level
	}
}
# New grants and journal expansions both consume five levels.
eafp_concession_urban_cap = {
	value = eafp_concession_urban_levels
	multiply = 5
}
eafp_concession_free_capacity = {
	value = eafp_concession_urban_cap
	subtract = eafp_concession_total_levels
	min = 0
}
eafp_concession_over_capacity = {
	value = eafp_concession_total_levels
	subtract = eafp_concession_urban_cap
	min = 0
}
eafp_concession_trade_equivalent = {
	value = eafp_concession_total_levels
}
# Building scope; current average profitability, not a forecast of the next level.
eafp_concession_industry_score = {
	value = weekly_profit
	divide = { value = level min = 1 }
}

# One +5 expansion (journal button and AI): 500,000 per click (100,000 per level).
eafp_concession_expansion_cost = { value = 500000 }
# Minimum occupancy of the concession building before the AI pays for an expansion
# (same threshold vanilla uses for building expansion conditions).
eafp_concession_ai_expand_min_occupancy = { value = 0.8 }
# One level of a local industry or port bought for a concession company by the
# monthly JE investment, paid from the holder's treasury; the holder needs at least
# this much in its treasury to be chosen.
eafp_concession_industry_level_cost = { value = 100000 }
# The monthly industry investment needs the local concession building to be this staffed
# and funded. Looser than the AI expansion (occupancy 0.8); new buildings start with zero
# reserves, so investment begins once the concession has built some up.
eafp_concession_industry_min_occupancy = { value = 0.6 }
eafp_concession_industry_min_cash_reserves = { value = 0.2 }
# State scope, evaluated only in the return effect with its pre-removal level snapshot.
eafp_concession_loss_relations = {
	value = scope:concession_loss_levels
	multiply = 2
	add = {
		value = scope:concession_loss_levels
		divide = { value = eafp_concession_urban_levels min = 1 }
		multiply = 10
	}
	max = 50
}
eafp_concession_loss_prestige = {
	value = scope:concession_loss_levels
	add = {
		value = scope:concession_loss_levels
		divide = { value = eafp_concession_urban_levels min = 1 }
		multiply = 5
	}
	max = 30
}

# Presentation only: cap can fall below existing levels, including zero.
eafp_concession_cap_usage = {
	value = eafp_concession_total_levels
	divide = { value = eafp_concession_urban_cap min = 1 }
	min = 0
	max = 1
}
