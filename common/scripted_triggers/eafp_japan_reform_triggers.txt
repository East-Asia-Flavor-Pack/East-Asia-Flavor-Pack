# 국가 범위. 선행 노선 선택 없이 두 정치 목표를 독립적으로 평가한다.
eafp_japan_imperial_victory = {
	custom_tooltip = {
		text = at_least_six_months_tt
		has_variable = restoration_timer_var
		var:restoration_timer_var >= 6
	}
	custom_tooltip = {
		text = ruler_is_japanese_emperor_tt
		ruler ?= {
			has_trait = special_character_japanese_emperor
		}
	}
	custom_tooltip = {
		text = is_not_in_civil_war_tt
		NOT = {
			any_civil_war = {
				civil_war_progress > 0
			}
		}
	}
	NOT = {
		has_law = law_type:law_bakufu
	}
	ig:ig_landowners ?= {
		custom_tooltip = {
			text = landowners_leader_is_not_tokugawa_tt
			leader ?= {
				NOT = {
					has_variable = house_tokugawa
				}
			}
		}
	}
}

eafp_japan_kobu_gattai_completed = {
	custom_tooltip = {
		text = kobu_gattai_trigger_conditions_tt
		has_variable = bakufu_kaikaku_complete_var
		has_variable = japan_restoration_complete
		has_variable = kobu_gattai
		NOT = { has_variable = kogi_yoron }
	}
	custom_tooltip = {
		text = bakufu_kaikaku_complete_tt
		has_variable = bakufu_kaikaku_complete_var
	}
}
eafp_japan_kogi_yoron_completed = {
	custom_tooltip = {
		text = kogi_yoron_trigger_conditions_tt
		has_variable = bakufu_kaikaku_complete_var
		has_variable = japan_restoration_complete
		has_variable = kogi_yoron
		has_variable = japan_emperor_restored
		NOT = { has_variable = kobu_gattai }
	}
	custom_tooltip = {
		text = bakufu_kaikaku_complete_tt
		has_variable = bakufu_kaikaku_complete_var
	}
}
eafp_japan_shogunate_victory_completed = {
	custom_tooltip = {
		text = completed_meiji_restoration_with_shogunate_victory_tt
		OR = {
			eafp_japan_kobu_gattai_completed = yes
			eafp_japan_kogi_yoron_completed = yes
		}
	}
}
eafp_japan_reform_common_conditions = {
	c:JAP ?= THIS
	has_journal_entry = je_meiji_restoration
	custom_tooltip = {
		text = eafp_japan_reform_not_completed_tt
		NOT = { has_variable = japan_restoration_complete }
		NOT = { has_variable = bakufu_kaikaku_complete_var }
	}
	has_law = law_type:law_monarchy
	custom_tooltip = {
		text = is_not_in_civil_war_tt
		NOT = { any_civil_war = { civil_war_progress > 0 } }
	}
	NOT = { eafp_japan_imperial_victory = yes }
	trigger_if = {
		limit = {
			any_political_movement = { is_political_movement_type = movement_meiji_restorationist }
		}
		any_political_movement = {
			is_political_movement_type = movement_meiji_restorationist
			OR = {
				political_movement_support < 0.05
				political_movement_radicalism < 0.5
			}
		}
	}
}
eafp_japan_kobu_gattai_petition_ready = {
	eafp_japan_reform_common_conditions = yes
	legitimacy >= 75
	custom_tooltip = {
		text = ruler_is_japanese_shogun_tt
		has_law = law_type:law_bakufu
	}
	custom_tooltip = {
		text = shogun_imperial_marriage_tt
		has_variable = japan_completed_imperial_marriage
	}
}
eafp_japan_kobu_gattai_conditions = {
	eafp_japan_kobu_gattai_petition_ready = yes
	custom_tooltip = {
		text = shogun_renewed_bakufu_tt
		has_variable = je_meiji_restoration_proclaim_imperial_restoration_var
		has_variable = japan_renewed_bakufu
	}
}
eafp_japan_kogi_yoron_petition_ready = {
	eafp_japan_reform_common_conditions = yes
	legitimacy >= 75
	custom_tooltip = {
		text = is_democratic_tt
		country_has_voting_franchise = yes
	}
	ig:ig_landowners = {
		custom_tooltip = {
			text = landowners_in_government_tt
			is_in_government = yes
		}
		custom_tooltip = {
			text = landowners_leader_is_tokugawa_tt
			leader = { has_variable = house_tokugawa }
		}
	}
	custom_tooltip = {
		text = ruler_is_japanese_emperor_tt
		ruler = { has_variable = house_yamato }
	}
}
eafp_japan_kogi_yoron_conditions = {
	eafp_japan_kogi_yoron_petition_ready = yes
	custom_tooltip = {
		text = eafp_japan_kogi_petition_accepted_tt
		has_variable = je_meiji_restoration_proclaim_imperial_restoration_var
		has_variable = japan_taikunate_var
	}
	# 두 조건이 동시에 충족될 경우 공무합체 우선
	hidden_trigger = { NOT = { eafp_japan_kobu_gattai_conditions = yes } }
}
eafp_japan_reform_invalid = {
	NOT = { eafp_japan_shogunate_victory_completed = yes }
	OR = {
		NOT = { c:JAP ?= THIS }
		NOT = { country_has_monarchy_law = yes }
		eafp_japan_imperial_victory = yes
		AND = {
			has_variable = japan_restoration_complete
			has_variable = japan_emperor_restored
			NOT = { has_variable = bakufu_kaikaku_complete_var }
		}
	}
}
