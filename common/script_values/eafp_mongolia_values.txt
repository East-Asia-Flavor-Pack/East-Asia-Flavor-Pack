# fixed opening-day target. Ownership and treaty changes update this total.

	every_country = {
		limit = { is_eafp_mgl_united_industry_partner = yes }
		every_scope_state = {
			limit = { is_in_geographic_region = geographic_region_eafp_mgl_united_industrial_area }

		}
	}
}
