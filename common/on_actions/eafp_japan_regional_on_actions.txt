on_state_owner_change = {
    on_actions = { eafp_japan_on_regional_state_change }
}
on_state_created = {
    on_actions = { eafp_japan_on_regional_state_change }
}
eafp_japan_on_regional_state_change = {
    trigger = {
        OR = { has_variable = eafp_japan_loyalty_cache_managed c:JAP ?= owner }
    }
    effect = {
        eafp_japan_release_region_state = yes
        c:JAP ?= { eafp_japan_refresh_regions = yes }
    }
}

# Registered in 00_code_on_actions_definition.txt. The lifecycle record survives role expiry;
# the guarded end effect makes career-end and death callbacks idempotent.
eafp_japan_on_bakufu_politician_death = {
    trigger = {
        exists = c:JAP
        has_variable = eafp_bakufu_office_position
    }
    effect = {
        if = {
            limit = { c:JAP = { eafp_japan_regional_active = yes } }
            remove_bakufu_politician_role = yes
        }
        else = { remove_bakufu_politician_role_without_appointment = yes }
        c:JAP ?= { eafp_japan_refresh_regions = yes }
    }
}
