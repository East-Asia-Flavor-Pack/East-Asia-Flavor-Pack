# All callers provide concession_state (state), concession_host and concession_holder (countries).
# Called from the monthly incident and the police-conflict option.
eafp_concession_start_return_war = {
	if = {
		limit = { scope:concession_host = { eafp_concession_can_start_return_war = yes } }
		scope:concession_host = {
			create_diplomatic_play = {
				name = dp_eafp_concession_incident
				type = dp_eafp_concession_incident
				target_country = scope:concession_holder
				war = yes
				add_war_goal = {
					holder = scope:concession_host
					type = eafp_concession_incident_return
					target_country = scope:concession_holder
					target_state = scope:concession_state
					primary_demand = yes
				}
			}
		}
		if = {
			limit = { scope:concession_host = { has_war_with = scope:concession_holder } }
			scope:concession_host = { trigger_event = { id = concessions_in_china.205 } }
		}
	}
}

# State scope; used only for an identified civil-war split or a same-owner merge.
# Inheritance never calls grant, so it cannot award another concession level.
eafp_concession_inherit_state_rights = {
	every_in_list = {
		variable = concession_owner_varlist
		$TARGET$ = {
			if = {
				limit = { NOT = { is_target_in_variable_list = { name = concession_owner_varlist target = prev } } }
				add_to_variable_list = { name = concession_owner_varlist target = prev }
			}
		}
	}
}

# State scope. LEVELS is 5 for a new right and 1 for paid expansion.
# Never reset the production methods/reserves of an existing shared building.
eafp_concession_add_company_level = {
	if = {
		limit = { any_scope_building = { is_building_type = building_eafp_concession } }
		create_building = {
			building = building_eafp_concession
			add_ownership = { company = { type = company_eafp_concession country = scope:concession_holder levels = $LEVELS$ } }
		}
	}
	else = {
		create_building = {
			building = building_eafp_concession
			activate_production_methods = { pm_eafp_concession_trade_quantity_normal }
			reserves = 1
			add_ownership = { company = { type = company_eafp_concession country = scope:concession_holder levels = $LEVELS$ } }
		}
	}
}

# Rights live on actual states, never state regions or province anchors.
eafp_concession_grant_effect = {
	if = {
		limit = {
			owner = scope:concession_host
			eafp_concession_can_grant = { HOLDER = scope:concession_holder }
		}
		scope:concession_holder = {
			if = {
				limit = { NOT = { has_company = company_type:company_eafp_concession } }
				add_modifier = { name = eafp_concession_company_slot }
				add_company = company_type:company_eafp_concession
				if = {
					limit = { has_company = company_type:company_eafp_concession }
					capital.state_region = { save_temporary_scope_as = concession_company_hq }
					company:company_eafp_concession = { set_company_state_region = scope:concession_company_hq }
				}
			}
		}
		if = {
			limit = { scope:concession_holder = { has_company = company_type:company_eafp_concession } }
			save_temporary_scope_value_as = { name = concession_previous_levels value = eafp_concession_total_levels }
			eafp_concession_add_company_level = { LEVELS = 5 }
			if = {
				limit = {
					eafp_concession_total_levels > scope:concession_previous_levels
					any_scope_building = {
						is_building_type = building_eafp_concession
						levels_owned_by_country = { target = scope:concession_holder value >= 5 }
					}
				}
				add_to_variable_list = { name = concession_owner_varlist target = scope:concession_holder }
				# Expansion is allowed only while the granting treaty article is in force.
				if = {
					limit = { NOT = { is_target_in_variable_list = { name = concession_treaty_varlist target = scope:concession_holder } } }
					add_to_variable_list = { name = concession_treaty_varlist target = scope:concession_holder }
				}
			}
			else = { debug_log = "EAFP concession grant: company ownership creation failed; right not registered" }
		}
		else = { debug_log = "EAFP concession grant: company establishment failed; right not registered" }
		if = {
			limit = { NOT = { is_target_in_variable_list = { name = concession_owner_varlist target = scope:concession_holder } } }
			scope:concession_holder = {
				send_interface_message = {
					title = eafp_concession_grant_failed_title
					desc = eafp_concession_grant_failed_desc
					icon = "gfx/interface/icons/building_icons/building_trade_center.dds"
				}
			}
		}
		# Also removes an empty newly created company after a failed grant.
		eafp_concession_refresh_effect = yes
	}
}

# Shared by the player GUI and monthly AI. No treasury charge when growth fails.
eafp_concession_expand_effect = {
	if = {
		limit = { eafp_concession_can_expand = { HOLDER = scope:concession_holder } }
		save_temporary_scope_value_as = { name = concession_previous_levels value = eafp_concession_total_levels }
		eafp_concession_add_company_level = { LEVELS = 1 }
		if = {
			limit = { eafp_concession_total_levels > scope:concession_previous_levels }
			scope:concession_holder = { add_treasury = { value = eafp_concession_expansion_cost multiply = -1 } }
		}
	}
}

# FORCED is explicit at every caller; no dependence on treaty callback ordering.
# Every return (treaty withdrawal, incident, war) ends only the right and treaty status.
# Built concession levels always remain with their owners, whoever owns the state;
# no script path removes building_eafp_concession.
eafp_concession_return_effect = {
	if = {
		limit = {
			owner = scope:concession_host
			eafp_concession_has_holder = { HOLDER = scope:concession_holder }
		}
		# Derive the holder's actual levels through comparisons, not a persistent ledger.
		# Binary search needs at most 32 steps for engine integer building levels.
		save_temporary_scope_value_as = { name = concession_loss_levels value = 0 }
		every_scope_building = {
			limit = { is_building_type = building_eafp_concession }
			save_temporary_scope_value_as = { name = concession_level_low value = 0 }
			save_temporary_scope_value_as = { name = concession_level_high value = level }
			while = {
				count = 32
				save_temporary_scope_value_as = {
					name = concession_level_mid
					value = { value = scope:concession_level_low add = scope:concession_level_high divide = 2 ceiling = yes }
				}
				if = {
					limit = { levels_owned_by_country = { target = scope:concession_holder value >= scope:concession_level_mid } }
					save_temporary_scope_value_as = { name = concession_level_low value = scope:concession_level_mid }
				}
				else = {
					save_temporary_scope_value_as = { name = concession_level_high value = { value = scope:concession_level_mid subtract = 1 } }
				}
			}
			save_temporary_scope_value_as = { name = concession_loss_levels value = scope:concession_level_low }
		}
		# Snapshot before removing the right or dissolving the company.
		save_temporary_scope_value_as = { name = concession_loss_relation value = eafp_concession_loss_relations }
		save_temporary_scope_value_as = { name = concession_loss_prestige value = eafp_concession_loss_prestige }
		remove_list_variable = { name = concession_owner_varlist target = scope:concession_holder }
		if = {
			limit = { is_target_in_variable_list = { name = concession_treaty_varlist target = scope:concession_holder } }
			remove_list_variable = { name = concession_treaty_varlist target = scope:concession_holder }
		}
		if = {
			limit = { always = $FORCED$ eafp_concession_loss_prestige > 0 }
			scope:concession_host = {
				change_relations = { country = scope:concession_holder value = { value = scope:concession_loss_relation multiply = -1 } }
			}
			scope:concession_holder = {
				# The previous loss cannot be reconstructed after rights/buildings are removed.
				# Retain the strongest unexpired shock; a smaller loss must not erase it.
				if = {
					limit = { NOT = { has_variable = concession_loss_prestige_snapshot } }
					set_variable = { name = concession_loss_prestige_snapshot value = 0 days = 730 }
				}
				if = {
					limit = {
						OR = {
							NOT = { has_modifier = eafp_concession_forced_loss }
							var:concession_loss_prestige_snapshot < scope:concession_loss_prestige
						}
					}
					set_variable = { name = concession_loss_prestige_snapshot value = scope:concession_loss_prestige days = 730 }
					remove_modifier = eafp_concession_forced_loss
					add_modifier = { name = eafp_concession_forced_loss multiplier = scope:concession_loss_prestige days = 730 }
				}
			}
		}
		eafp_concession_refresh_effect = yes
	}
}

# One global refresh, used by the international JE and immediate right/ownership changes.
# No money is moved here and this effect never schedules a timer or creates country journal entries.
eafp_concession_refresh_effect = {
	clear_global_variable_list = concession_candidate_states_gvarlist
	every_state = {
		limit = { OR = { has_variable_list = concession_owner_varlist has_building = building_eafp_concession } }
		save_temporary_scope_as = eafp_refresh_state
		every_in_list = {
			variable = concession_owner_varlist
			save_temporary_scope_as = eafp_refresh_holder
			if = {
				limit = {
					OR = {
						is_country_alive = no
						this = scope:eafp_refresh_state.owner
						NOT = { has_company = company_type:company_eafp_concession }
						scope:eafp_refresh_state = {
							NOT = {
								any_scope_building = {
									is_building_type = building_eafp_concession
									levels_owned_by_country = { target = scope:eafp_refresh_holder value >= 1 }
								}
							}
						}
					}
				}
				scope:eafp_refresh_state = {
					remove_list_variable = { name = concession_owner_varlist target = prev }
				}
			}
			# Adopt a right whose granting article is in force but not yet recorded (older saves).
			# Inherited rights have no article and therefore never become expandable.
			else_if = {
				limit = {
					scope:eafp_refresh_state = { NOT = { is_target_in_variable_list = { name = concession_treaty_varlist target = scope:eafp_refresh_holder } } }
					any_scope_treaty = {
						any_scope_article = {
							has_type = eafp_grant_concession
							target_country = scope:eafp_refresh_holder
							input_state = scope:eafp_refresh_state
						}
					}
				}
				scope:eafp_refresh_state = {
					add_to_variable_list = { name = concession_treaty_varlist target = scope:eafp_refresh_holder }
				}
			}
		}
		# Treaty status never outlives the right it belongs to.
		every_in_list = {
			variable = concession_treaty_varlist
			save_temporary_scope_as = eafp_refresh_treaty_holder
			if = {
				limit = {
					scope:eafp_refresh_state = {
						NOT = { is_target_in_variable_list = { name = concession_owner_varlist target = scope:eafp_refresh_treaty_holder } }
					}
				}
				scope:eafp_refresh_state = {
					remove_list_variable = { name = concession_treaty_varlist target = scope:eafp_refresh_treaty_holder }
				}
			}
		}
		# A building without rights (any return, or a change of state owner) is always retained.
		if = {
			limit = { any_in_list = { variable = concession_owner_varlist always = yes } }
			add_to_global_variable_list = { name = concession_candidate_states_gvarlist target = this }
		}
	}
	every_country = {
		# Rights are already reconciled above. Removing a company never removes ordinary factories.
		# A company that still owns retained assets (e.g. after treaty withdrawal) is kept.
		if = {
			limit = {
				has_company = company_type:company_eafp_concession
				NOT = { eafp_concession_holder_has_rights = yes }
				company:company_eafp_concession = { company_owned_levels < 1 }
			}
			remove_company = company_type:company_eafp_concession
		}
		if = {
			limit = { has_variable = concession_loss_prestige_snapshot NOT = { has_modifier = eafp_concession_forced_loss } }
			remove_variable = concession_loss_prestige_snapshot
		}
		# The modifier follows the actual company, not prosperity or the right count.
		if = {
			limit = { has_company = company_type:company_eafp_concession }
			if = {
				limit = { NOT = { has_modifier = eafp_concession_company_slot } }
				add_modifier = { name = eafp_concession_company_slot }
			}
		}
		else = { remove_modifier = eafp_concession_company_slot }
	}
}
