concessions_in_china.200 = {
	type = country_event
	placement = scope:concession_state
	title = concessions_in_china.200.t
	desc = concessions_in_china.200.d
	flavor = concessions_in_china.200.f
	event_image = { video = "gfx/event_pictures/asia_farmers_market.bk2" }
	duration = 3
	trigger = { eafp_concession_event_valid = yes }
	option = {
		name = concessions_in_china.200.a
		default_option = yes
		if = {
			limit = { eafp_concession_event_valid = yes }
			scope:concession_host = { change_relations = { country = scope:concession_holder value = 5 } }
		}
		ai_chance = { base = 50 }
	}
	option = {
		name = concessions_in_china.200.b
		if = {
			limit = { eafp_concession_event_valid = yes }
			scope:concession_host = { change_relations = { country = scope:concession_holder value = -10 } }
		}
		ai_chance = { base = 20 }
	}
}

	duration = 3
	trigger = { eafp_concession_event_valid = yes }
	option = {
		name = concessions_in_china.201.a
		default_option = yes
		if = {
			limit = { eafp_concession_event_valid = yes }
			save_scope_value_as = { name = eafp_concession_compensation value = scope:concession_state.eafp_concession_state_share }
			scope:concession_holder = { add_treasury = { value = scope:eafp_concession_compensation multiply = -1 } }
			scope:concession_host = { add_treasury = scope:eafp_concession_compensation }
		}
		ai_chance = { base = 60 }
	}
	option = {
		name = concessions_in_china.201.b
		if = {
			limit = { eafp_concession_event_valid = yes }
			scope:concession_host = { change_relations = { country = scope:concession_holder value = -10 } }
		}
		ai_chance = { base = 30 }
	}
	option = {
		name = concessions_in_china.201.c
		trigger = { scope:concession_host = { eafp_concession_can_start_return_war = yes } }
		eafp_concession_start_return_war = yes
		ai_chance = { base = 5 }
	}
}
