# Country = JAP unless explicitly marked state scope. UI never owns L/I.
# .disable sources remain unchanged; only loyalty / independency are restored.

# State scope. Release only an EAFP-owned cache, then match vanilla's JAP-only
# character loop. If no vanilla daimyo exists, its accessors use neutral default.
eafp_japan_release_region_state = {
    if = {
        limit = { has_variable = eafp_japan_loyalty_cache_managed }
        # Recall missions using the actual state, including lost split-state pieces.
        save_scope_as = eafp_japan_lost_mission_state
        c:JAP ?= {
            every_scope_character = {
                limit = { var:eafp_bakuhantaisei_politician_current_mission_location ?= scope:eafp_japan_lost_mission_state }
                remove_bakufu_politician_mission = yes
            }
        }
        clear_saved_scope = eafp_japan_lost_mission_state
        if = { limit = { has_modifier = modifier_oversee_daimyo_domains } remove_modifier = modifier_oversee_daimyo_domains }
        if = { limit = { has_modifier = modifier_reaffirm_daimyos_loyalty } remove_modifier = modifier_reaffirm_daimyos_loyalty }
        # 저널 완료/무효화 및 주 상실 시 유산도 함께 정리한다.
        if = {
            limit = { has_modifier = legacy_of_sekigahara_modifier }
            remove_modifier = legacy_of_sekigahara_modifier
        }
        eafp_japan_clear_vanilla_region_modifiers = yes
        remove_variable = eafp_japan_loyalty_cache_managed
        remove_variable = cached_daimyo_loyalty
        remove_modifier = eafp_japan_regional_tax_waste
        remove_modifier = eafp_japan_regional_autonomy
        if = {
            limit = { c:JAP ?= owner }
            save_scope_as = eafp_japan_released_state
            owner = {
                every_scope_character = {
                    limit = { has_role_of_type = magnate has_variable = daimyo_var var:daimyo_var ?= scope:eafp_japan_released_state.state_region }
                    scope:eafp_japan_released_state = { set_variable = { name = cached_daimyo_loyalty value = PREV.loyalty } }
                }
            }
            clear_saved_scope = eafp_japan_released_state
        }
    }
}

# Only called in JAP scope. Foreign portions are released, never assigned JAP L.
eafp_japan_refresh_region = {
    remove_variable = eafp_japan_region_active_$STATE$
    s:STATE_$STATE$ = {
        every_scope_state = {
            if = {
                limit = { has_variable = eafp_japan_loyalty_cache_managed NOT = { c:JAP ?= owner } }
                eafp_japan_release_region_state = yes
            }
        }
    }
    if = {
        limit = { eafp_japan_regional_active = yes exists = s:STATE_$STATE$.region_state:JAP }
        if = {
            limit = { NOT = { has_variable = eafp_japan_region_initialized_$STATE$ } }
            # 막번체제 프로그레스 바 초기값: 옛 earp_jap_values.disable 자료.
            set_variable = { name = eafp_japan_loyalty_$STATE$ value = $LOYALTY$ }
            set_variable = { name = eafp_japan_independency_$STATE$ value = $INDEPENDENCY$ }
            set_variable = eafp_japan_region_initialized_$STATE$
        }
        set_variable = eafp_japan_region_active_$STATE$
        # Keep the UI linked to JAP's actual state, including split regions.
        set_variable = { name = eafp_japan_region_state_$STATE$ value = s:STATE_$STATE$.region_state:JAP }
        # Synchronize mission modifiers before evaluating the state's monthly totals.
        s:STATE_$STATE$.region_state:JAP = {
            if = {
                limit = { has_modifier = modifier_oversee_daimyo_domains }
                remove_modifier = modifier_oversee_daimyo_domains
            }
            if = {
                limit = { owner = { has_modifier = modifier_oversee_daimyo_domains_$STATE$ } }
                add_modifier = { name = modifier_oversee_daimyo_domains days = 14 }
            }
            if = {
                limit = { has_modifier = modifier_reaffirm_daimyos_loyalty }
                remove_modifier = modifier_reaffirm_daimyos_loyalty
            }
            if = {
                limit = { owner = { has_modifier = modifier_reaffirm_daimyos_loyalty_$STATE$ } }
                add_modifier = { name = modifier_reaffirm_daimyos_loyalty days = 14 }
            }
        }
        s:STATE_$STATE$.region_state:JAP = { eafp_japan_sync_vanilla_region_modifiers = yes }
        set_variable = { name = eafp_japan_loyalty_delta_$STATE$ value = eafp_japan_loyalty_$STATE$_monthly }
        set_variable = { name = eafp_japan_independency_delta_$STATE$ value = eafp_japan_independency_$STATE$_monthly }
        s:STATE_$STATE$.region_state:JAP = {
            set_variable = { name = cached_daimyo_loyalty value = owner.var:eafp_japan_loyalty_$STATE$ }
            set_variable = eafp_japan_loyalty_cache_managed
            remove_modifier = eafp_japan_regional_tax_waste
            remove_modifier = eafp_japan_regional_autonomy
            add_modifier = { name = eafp_japan_regional_tax_waste multiplier = owner.eafp_japan_tax_loss_$STATE$ days = 14 }
            add_modifier = { name = eafp_japan_regional_autonomy multiplier = { value = owner.var:eafp_japan_independency_$STATE$ divide = 100 } days = 14 }
        }
    }
    else = {
        remove_variable = eafp_japan_region_state_$STATE$
        s:STATE_$STATE$ = { every_scope_state = { eafp_japan_release_region_state = yes } }
        # A mission may still point at the lost state object. Clear its country cost
        # explicitly instead of relying on region_state:JAP resolving after loss.
        every_scope_character = {
            limit = { var:eafp_bakuhantaisei_politician_current_mission_location.state_region ?= s:STATE_$STATE$ }
            remove_bakufu_politician_mission = yes
        }
        remove_modifier = modifier_oversee_daimyo_domains_$STATE$
        remove_modifier = modifier_reaffirm_daimyos_loyalty_$STATE$
    }
}

eafp_japan_monthly_region = {
    if = {
        limit = { has_variable = eafp_japan_region_active_$STATE$ }
        # Both deltas were calculated before either state variable changes.
        set_variable = { name = eafp_japan_loyalty_$STATE$ value = { value = var:eafp_japan_loyalty_$STATE$ add = var:eafp_japan_loyalty_delta_$STATE$ min = 0 max = 100 } }
        set_variable = { name = eafp_japan_independency_$STATE$ value = { value = var:eafp_japan_independency_$STATE$ add = var:eafp_japan_independency_delta_$STATE$ min = 0 max = 100 } }
    }
}
