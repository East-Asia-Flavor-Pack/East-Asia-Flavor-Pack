# Standalone EAFP Korean intervention journal. It does not inspect or modify the
# vanilla Ryukyu Rivalry journal.

je_eafp_ryukyu_intervention = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	group = je_group_global_international_situations

	scripted_button = button_je_eafp_ryukyu_dispatch_envoys
	scripted_button = button_je_eafp_ryukyu_arm_harbors

	is_shown_in_lobby = {
		c:KOR = THIS
	}

	is_shown_when_inactive = {
		c:KOR = THIS
		has_variable = eafp_var_sam_all_clear
		NOT = { has_variable = eafp_ryukyu_intervention_declined_var }
		NOT = { has_variable = eafp_ryukyu_sidecar_finished_var }
		exists = c:JAP
		exists = c:CHI
		exists = c:RYU
	}

	should_be_involved = {
		c:KOR ?= this
	}

	should_show_when_not_involved = {
		always = no
	}

	possible = {
		c:KOR = THIS
		exists = c:JAP
		exists = c:CHI
		exists = c:RYU
	}

	immediate = {
		if = {
			limit = { NOT = { has_variable = eafp_ryukyu_intervention_progress_var } }
			set_variable = {
				name = eafp_ryukyu_intervention_progress_var
				value = 0
			}
		}
		if = {
			limit = {
				NOT = { has_variable = eafp_ryukyu_intervention_offered_var }
				NOT = { has_variable = eafp_ryukyu_intervention_var }
			}
			set_variable = eafp_ryukyu_intervention_offered_var
			trigger_event = { id = eafp_ryukyu.1 days = 7 popup = yes }
		}
	}

	complete = {
		has_variable = eafp_ryukyu_intervention_var
		c:RYU ?= { is_subject = no }
		has_variable = eafp_ryukyu_intervention_progress_var
		var:eafp_ryukyu_intervention_progress_var >= 100
	}

	on_complete = {
		if = {
			limit = {
				c:RYU ?= { is_subject = no }
				can_have_as_subject = { who = c:RYU type = subject_type_tributary }
			}
			create_diplomatic_pact = {
				country = c:RYU
				type = tributary
			}
			if = {
				limit = {
					NOT = { is_involved_in_journal_entry = je_eafp_taiwan_pioneer }
					NOT = { has_inactive_journal_entry = je_eafp_taiwan_pioneer }
					NOT = { has_variable = eafp_taiwan_pioneer_complete_var }
					NOT = { has_variable = eafp_taiwan_pioneer_failed_var }
				}
				add_contextless_journal_entry = je_eafp_taiwan_pioneer
			}
			trigger_event = { id = eafp_ryukyu.2 popup = yes }
		}
		set_variable = eafp_ryukyu_sidecar_finished_var
	}

	invalid = {
		OR = {
			NOT = { exists = c:JAP }
			NOT = { exists = c:CHI }
			NOT = { exists = c:RYU }
			has_variable = eafp_ryukyu_intervention_declined_var
		}
	}

	on_invalid = {
		set_variable = eafp_ryukyu_sidecar_finished_var
	}

	current_value = { value = root.var:eafp_ryukyu_intervention_progress_var }
	goal_add_value = { value = 100 }
	progressbar = yes
	weight = 10000
	should_be_pinned_by_default_uninvolved_or_context = yes
	can_revolution_inherit = no
}
