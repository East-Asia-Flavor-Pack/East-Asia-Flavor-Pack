# All callers provide concession_state (state), concession_host and concession_holder (countries).
# Called from the monthly incident and the police-conflict option.
eafp_concession_start_return_war = {
	if = {
		limit = { scope:concession_host = { eafp_concession_can_start_return_war = yes } }
		scope:concession_host = {
			create_diplomatic_play = {
				name = dp_eafp_concession_incident
				type = dp_eafp_concession_incident
				target_country = scope:concession_holder
				war = yes
				add_war_goal = {
					holder = scope:concession_host
					type = eafp_concession_incident_return
					target_country = scope:concession_holder
					target_state = scope:concession_state
					primary_demand = yes
				}
			}
		}
	}
}

# State scope; used only for an identified civil-war split or a same-owner merge.
# Inheritance never calls grant, so it cannot award another concession level.
eafp_concession_inherit_state_rights = {
	every_in_list = {
		variable = concession_owner_varlist
		$TARGET$ = {
			if = {
				limit = { NOT = { is_target_in_variable_list = { name = concession_owner_varlist target = prev } } }
				add_to_variable_list = { name = concession_owner_varlist target = prev }
			}
		}
	}
}

# Building scope. Binary search for COUNTRY's levels in this building through
# levels_owned_by_country comparisons (engine levels are integers; 32 steps suffice).
# Result: scope:concession_level_low. COUNTRY must be a saved scope.
eafp_concession_measure_country_levels = {
	save_temporary_scope_value_as = { name = concession_level_low value = 0 }
	save_temporary_scope_value_as = { name = concession_level_high value = level }
	while = {
		count = 32
		save_temporary_scope_value_as = {
			name = concession_level_mid
			value = { value = scope:concession_level_low add = scope:concession_level_high divide = 2 ceiling = yes }
		}
		if = {
			limit = { levels_owned_by_country = { target = $COUNTRY$ value >= scope:concession_level_mid } }
			save_temporary_scope_value_as = { name = concession_level_low value = scope:concession_level_mid }
		}
		else = {
			save_temporary_scope_value_as = { name = concession_level_high value = { value = scope:concession_level_mid subtract = 1 } }
		}
	}
}

# State scope. Adds VALUE to KEY's entry in eafp_concession_rebuild_map (creates it if absent).
# KEY is a country scope; VALUE is a saved scope value.
eafp_concession_rebuild_map_add = {
	if = {
		limit = { is_key_in_variable_map = { name = eafp_concession_rebuild_map target = $KEY$ } }
		save_temporary_scope_value_as = {
			name = eafp_rebuild_merged
			value = { value = "variable_map(eafp_concession_rebuild_map|$KEY$)" add = $VALUE$ }
		}
		remove_from_variable_map = { name = eafp_concession_rebuild_map key = $KEY$ }
		add_to_variable_map = { name = eafp_concession_rebuild_map key = $KEY$ value = scope:eafp_rebuild_merged }
	}
	else = {
		add_to_variable_map = { name = eafp_concession_rebuild_map key = $KEY$ value = $VALUE$ }
	}
}

# Country scope. Creates the country's concession company if it has none, exactly as a
# concession treaty grant does: company slot modifier, add_company, HQ in the capital's
# state region. Shared by the grant effect and every create_building of concession levels.
eafp_concession_ensure_company = {
	if = {
		limit = { NOT = { has_company = company_type:company_eafp_concession } }
		add_modifier = { name = eafp_concession_company_slot }
		add_company = company_type:company_eafp_concession
		if = {
			limit = { has_company = company_type:company_eafp_concession }
			capital.state_region = { save_temporary_scope_as = concession_company_hq }
			company:company_eafp_concession = { set_company_state_region = scope:concession_company_hq }
		}
	}
}

# State scope. One create_building call giving LEVELS to scope:eafp_rebuild_owner's
# concession company. Concession levels are only ever company-owned: an owner without a
# company first gets one (eafp_concession_ensure_company). If the engine still refuses
# the company, nothing is created rather than falling back to country ownership.
# On an existing building the engine keeps max(existing, scripted) reserves, so
# RESERVES = 0 there leaves the live building's reserves untouched.
eafp_concession_rebuild_create = {
	scope:eafp_rebuild_owner = { eafp_concession_ensure_company = yes }
	if = {
		limit = { scope:eafp_rebuild_owner = { has_company = company_type:company_eafp_concession } }
		create_building = {
			building = building_eafp_concession
			activate_production_methods = { pm_eafp_concession pm_trade_center_trade_quantity_normal }
			reserves = $RESERVES$
			add_ownership = { company = { type = company_eafp_concession country = scope:eafp_rebuild_owner levels = $LEVELS$ } }
		}
	}
	else = { debug_log = "EAFP concession: owner has no concession company and none could be founded; levels not created" }
}

# State scope. ONE create_building call giving exactly LEVELS to
# scope:eafp_rebuild_owner. On the live building this overwrites that owner's levels
# (create_building levels do not add to what is already there), so the owner's whole
# new total goes in a single call. A newly founded concession building starts with zero
# cash reserves; on a live building reserves = 0 leaves its current reserves untouched.
eafp_concession_rebuild_chunk = {
	eafp_concession_rebuild_create = { LEVELS = $LEVELS$ RESERVES = 0 }
}

# State scope. LEVELS is 5 for a new right (grant, treaty-backed restore) and 1 for the
# monthly investment's concession growth. No remove_building: the live building keeps its other owners, workforce, cash reserves,
# production methods and subsidy. create_building levels do not add to an existing
# building, so the holder's ownership is overwritten with its new total in one call:
#   1. read the holder's current levels into the state's variable map
#      eafp_concession_rebuild_map (key = holder country);
#   2. add LEVELS to that entry;
#   3. run eafp_concession_rebuild_owner_levels (level switch -> one create_building).
# Callers judge success with scope:concession_expected_levels (holder's new total).
eafp_concession_add_company_level = {
	save_temporary_scope_as = eafp_rebuild_state
	clear_variable_map = eafp_concession_rebuild_map
	save_temporary_scope_value_as = { name = eafp_rebuild_added value = $LEVELS$ }

	# 1. The holder's current levels (0 for a new holder or a new building).
	if = {
		limit = { has_building = building_eafp_concession }
		random_scope_building = {
			limit = { is_building_type = building_eafp_concession }
			if = {
				limit = { levels_owned_by_country = { target = scope:concession_holder value >= 1 } }
				eafp_concession_measure_country_levels = { COUNTRY = scope:concession_holder }
				scope:eafp_rebuild_state = {
					eafp_concession_rebuild_map_add = { KEY = scope:concession_holder VALUE = scope:concession_level_low }
				}
			}
		}
	}

	# 2. The requested increase goes to the holder's entry.
	eafp_concession_rebuild_map_add = { KEY = scope:concession_holder VALUE = scope:eafp_rebuild_added }
	save_temporary_scope_value_as = {
		name = concession_expected_levels
		value = "variable_map(eafp_concession_rebuild_map|scope:concession_holder)"
	}

	# 3. Overwrite the holder's ownership on the live building (or found it).
	scope:concession_holder = { save_temporary_scope_as = eafp_rebuild_owner }
	save_temporary_scope_value_as = { name = eafp_rebuild_remaining value = scope:concession_expected_levels }
	eafp_concession_rebuild_owner_levels = yes
	if = {
		limit = {
			NOT = {
				any_scope_building = {
					is_building_type = building_eafp_concession
					levels_owned_by_country = { target = scope:concession_holder value >= scope:concession_expected_levels }
				}
			}
		}
		debug_log = "EAFP concession overwrite: holder levels did not reach the expected total"
	}
	clear_variable_map = eafp_concession_rebuild_map
}

# Rights live on actual states, never state regions or province anchors.
eafp_concession_grant_effect = {
	if = {
		limit = {
			owner = scope:concession_host
			eafp_concession_can_receive = { HOLDER = scope:concession_holder }
		}
		scope:concession_holder = { eafp_concession_ensure_company = yes }
		if = {
			limit = { scope:concession_holder = { has_company = company_type:company_eafp_concession } }
			eafp_concession_add_company_level = { LEVELS = 5 }
			if = {
				limit = {
					any_scope_building = {
						is_building_type = building_eafp_concession
						levels_owned_by_country = { target = scope:concession_holder value >= scope:concession_expected_levels }
					}
				}
				add_to_variable_list = { name = concession_owner_varlist target = scope:concession_holder }
				# A concluded concession gives the holder +1000 involvement in the state's strategic region.
				region = { save_temporary_scope_as = concession_region }
				scope:concession_holder = {
					add_involvement = { strategic_region = scope:concession_region value = 1000 }
				}
				# The international concession JE activates through its possible block
				# (an in-force concession treaty exists), not from here.
			}
			else = { debug_log = "EAFP concession grant: company ownership creation failed; right not registered" }
		}
		else = { debug_log = "EAFP concession grant: company establishment failed; right not registered" }
		if = {
			limit = { NOT = { is_target_in_variable_list = { name = concession_owner_varlist target = scope:concession_holder } } }
			scope:concession_holder = {
				send_interface_message = {
					title = eafp_concession_grant_failed_title
					desc = eafp_concession_grant_failed_desc
					icon = "gfx/interface/icons/building_icons/building_trade_center.dds"
				}
			}
		}
		# Also removes an empty newly created company after a failed grant.
		eafp_concession_refresh_effect = yes
	}
}

# State scope; scope:concession_holder is the country whose right just ended while other
# rights remain in the state. The engine has no per-owner level removal, so:
#   1. store every OTHER country's levels in eafp_concession_rebuild_map (key = country);
#   2. remove_building;
#   3. recreate each stored owner with one create_building (level switch).
# The returning holder is never stored, so its share is gone; other owners keep their
# levels. The rebuilt building starts fresh (workers re-hire, zero reserves, default PMs).
eafp_concession_remove_holder_share = {
	save_temporary_scope_as = eafp_rebuild_state
	clear_variable_map = eafp_concession_rebuild_map
	random_scope_building = {
		limit = { is_building_type = building_eafp_concession }
		save_temporary_scope_as = eafp_rebuild_building
		every_country = {
			limit = { NOT = { this = scope:concession_holder } }
			save_temporary_scope_as = eafp_rebuild_owner
			scope:eafp_rebuild_building = {
				if = {
					limit = { levels_owned_by_country = { target = scope:eafp_rebuild_owner value >= 1 } }
					eafp_concession_measure_country_levels = { COUNTRY = scope:eafp_rebuild_owner }
					scope:eafp_rebuild_state = {
						eafp_concession_rebuild_map_add = { KEY = scope:eafp_rebuild_owner VALUE = scope:concession_level_low }
					}
				}
			}
		}
	}
	remove_building = building_eafp_concession
	every_country = {
		save_temporary_scope_as = eafp_rebuild_owner
		scope:eafp_rebuild_state = {
			if = {
				limit = { is_key_in_variable_map = { name = eafp_concession_rebuild_map target = scope:eafp_rebuild_owner } }
				save_temporary_scope_value_as = {
					name = eafp_rebuild_remaining
					value = "variable_map(eafp_concession_rebuild_map|scope:eafp_rebuild_owner)"
				}
				eafp_concession_rebuild_owner_levels = yes
			}
		}
	}
	clear_variable_map = eafp_concession_rebuild_map
}

# State scope. Records in the state variable map eafp_concession_urban_map, for every
# concession holder here, the levels it owns in this state's urban buildings that the
# concession company can be given (eafp_concession_urban_investment_building: the
# industries, the port and the trade center: every building a concession company is given
# besides the concession building, so "used" = the holder's levels in this state minus its
# concession levels). Used as "used" against the cap (the holder's concession levels
# in this state) by the monthly investment and shown in the journal widget.
eafp_concession_urban_usage_refresh = {
	save_temporary_scope_as = eafp_usage_state
	clear_variable_map = eafp_concession_urban_map
	every_in_list = {
		variable = concession_owner_varlist
		save_temporary_scope_as = eafp_usage_holder
		save_temporary_scope_value_as = { name = eafp_usage_sum value = 0 }
		scope:eafp_usage_state = {
			every_scope_building = {
				limit = {
					eafp_concession_urban_investment_building = yes
					levels_owned_by_country = { target = scope:eafp_usage_holder value >= 1 }
				}
				eafp_concession_measure_country_levels = { COUNTRY = scope:eafp_usage_holder }
				save_temporary_scope_value_as = {
					name = eafp_usage_sum
					value = { value = scope:eafp_usage_sum add = scope:concession_level_low }
				}
			}
			add_to_variable_map = { name = eafp_concession_urban_map key = scope:eafp_usage_holder value = scope:eafp_usage_sum }
		}
	}
}

# State scope; a state with a concession right and concession building. Monthly from the
# international JE. Replaces the old privately funded construction queue and the old paid
# +5 concession expansion:
#   1. the investing company is the concession company that owns levels of this state's
#      concession building (largest share wins) and whose country can pay
#      eafp_concession_industry_level_cost; a holder without levels here never invests here.
#      It must have room for something: urban levels here (eafp_concession_urban_map) below
#      its concession levels here, or a free level under the state's concession cap;
#   2. the building is the one with a positive weekly profit and the best profit per level
#      among the concession building itself (while the cap has a free level) and the
#      existing local industries, port and trade center that can take one more level (while the holder's
#      urban levels are below its concession levels);
#   3. that holder's company share in it becomes current + 1 at once (create_building; for
#      the concession building eafp_concession_add_company_level with LEVELS 1), and the
#      cost is taken from the holder's treasury only if the share really grew.
eafp_concession_industry_investment_effect = {
	save_temporary_scope_as = eafp_industry_state
	# Current urban levels per holder (the cap check below reads this map).
	eafp_concession_urban_usage_refresh = yes
	# Starts at 0: only a company that owns concession levels in THIS state can be chosen.
	save_temporary_scope_value_as = { name = eafp_industry_best value = 0 }
	random_scope_building = {
		limit = { is_building_type = building_eafp_concession }
		save_temporary_scope_as = eafp_industry_concession
	}
	every_in_list = {
		variable = concession_owner_varlist
		limit = {
			is_country_alive = yes
			has_company = company_type:company_eafp_concession
			gold_reserves >= eafp_concession_industry_level_cost
		}
		save_temporary_scope_as = eafp_industry_candidate
		scope:eafp_industry_concession = { eafp_concession_measure_country_levels = { COUNTRY = scope:eafp_industry_candidate } }
		# Cap: urban levels a company may hold in this state = its concession levels here.
		save_temporary_scope_value_as = {
			name = eafp_industry_used
			value = "scope:eafp_industry_state.variable_map(eafp_concession_urban_map|scope:eafp_industry_candidate)"
		}
		if = {
			limit = {
				scope:concession_level_low > scope:eafp_industry_best
				OR = {
					scope:eafp_industry_used < scope:concession_level_low
					scope:eafp_industry_state = { eafp_concession_free_capacity >= 1 }
				}
			}
			save_temporary_scope_value_as = { name = eafp_industry_best value = scope:concession_level_low }
			save_temporary_scope_value_as = { name = eafp_industry_holder_used value = scope:eafp_industry_used }
			save_temporary_scope_as = eafp_industry_holder
		}
	}
	if = {
		limit = { exists = scope:eafp_industry_holder }
		ordered_scope_building = {
			limit = {
				level > 0
				weekly_profit > 0
				OR = {
					# Concession growth: one more level while the urban cap has room.
					AND = {
						is_building_type = building_eafp_concession
						scope:eafp_industry_state = { eafp_concession_free_capacity >= 1 }
					}
					# Industry/port/trade center: while the holder's urban levels are below its concession levels.
					AND = {
						scope:eafp_industry_holder_used < scope:eafp_industry_best
						can_queue_building_levels = 1
						OR = {
							AND = { is_building_type = building_food_industry scope:eafp_industry_state = { can_construct_building = building_food_industry } }
							AND = { is_building_type = building_textile_mill scope:eafp_industry_state = { can_construct_building = building_textile_mill } }
							AND = { is_building_type = building_furniture_manufactory scope:eafp_industry_state = { can_construct_building = building_furniture_manufactory } }
							AND = { is_building_type = building_glassworks scope:eafp_industry_state = { can_construct_building = building_glassworks } }
							AND = { is_building_type = building_tooling_workshop scope:eafp_industry_state = { can_construct_building = building_tooling_workshop } }
							AND = { is_building_type = building_paper_mill scope:eafp_industry_state = { can_construct_building = building_paper_mill } }
							AND = { is_building_type = building_chemical_plant scope:eafp_industry_state = { can_construct_building = building_chemical_plant } }
							AND = { is_building_type = building_explosives_factory scope:eafp_industry_state = { can_construct_building = building_explosives_factory } }
							AND = { is_building_type = building_synthetics_plant scope:eafp_industry_state = { can_construct_building = building_synthetics_plant } }
							AND = { is_building_type = building_steel_mill scope:eafp_industry_state = { can_construct_building = building_steel_mill } }
							AND = { is_building_type = building_motor_industry scope:eafp_industry_state = { can_construct_building = building_motor_industry } }
							AND = { is_building_type = building_automotive_industry scope:eafp_industry_state = { can_construct_building = building_automotive_industry } }
							AND = { is_building_type = building_electrics_industry scope:eafp_industry_state = { can_construct_building = building_electrics_industry } }
							AND = { is_building_type = building_shipyard scope:eafp_industry_state = { can_construct_building = building_shipyard } }
							AND = { is_building_type = building_arms_industry scope:eafp_industry_state = { can_construct_building = building_arms_industry } }
							AND = { is_building_type = building_artillery_foundry scope:eafp_industry_state = { can_construct_building = building_artillery_foundry } }
							AND = { is_building_type = building_munition_plant scope:eafp_industry_state = { can_construct_building = building_munition_plant } }
							AND = { is_building_type = building_art_academy scope:eafp_industry_state = { can_construct_building = building_art_academy } }
							AND = { is_building_type = building_port scope:eafp_industry_state = { can_construct_building = building_port } }
							AND = { is_building_type = building_trade_center scope:eafp_industry_state = { can_construct_building = building_trade_center } }
						}
					}
				}
			}
			order_by = eafp_concession_industry_score
			max = 1
			check_range_bounds = no
			# Concession growth: +1 level of the holder's concession company share.
			if = {
				limit = { is_building_type = building_eafp_concession }
				scope:eafp_industry_holder = { save_temporary_scope_as = concession_holder }
				scope:eafp_industry_state = { eafp_concession_add_company_level = { LEVELS = 1 } }
				if = {
					limit = {
						scope:eafp_industry_state = {
							any_scope_building = {
								is_building_type = building_eafp_concession
								levels_owned_by_country = { target = scope:eafp_industry_holder value >= scope:concession_expected_levels }
							}
						}
					}
					scope:eafp_industry_holder = { add_treasury = { value = eafp_concession_industry_level_cost multiply = -1 } }
				}
				else = { debug_log = "EAFP concession growth: concession share did not grow; no cost charged" }
			}
			# Local industry, port or trade center.
			else = {
				# The holder's current levels here (country total; normally only its company).
				eafp_concession_measure_country_levels = { COUNTRY = scope:eafp_industry_holder }
				if = {
					limit = { scope:concession_level_low < 100 }
					save_temporary_scope_value_as = {
						name = eafp_industry_new_levels
						value = { value = scope:concession_level_low add = 1 }
					}
					if = {
						limit = { is_building_type = building_food_industry }
						scope:eafp_industry_state = { eafp_concession_industry_create = { TYPE = building_food_industry } }
					}
					else_if = {
						limit = { is_building_type = building_textile_mill }
						scope:eafp_industry_state = { eafp_concession_industry_create = { TYPE = building_textile_mill } }
					}
					else_if = {
						limit = { is_building_type = building_furniture_manufactory }
						scope:eafp_industry_state = { eafp_concession_industry_create = { TYPE = building_furniture_manufactory } }
					}
					else_if = {
						limit = { is_building_type = building_glassworks }
						scope:eafp_industry_state = { eafp_concession_industry_create = { TYPE = building_glassworks } }
					}
					else_if = {
						limit = { is_building_type = building_tooling_workshop }
						scope:eafp_industry_state = { eafp_concession_industry_create = { TYPE = building_tooling_workshop } }
					}
					else_if = {
						limit = { is_building_type = building_paper_mill }
						scope:eafp_industry_state = { eafp_concession_industry_create = { TYPE = building_paper_mill } }
					}
					else_if = {
						limit = { is_building_type = building_chemical_plant }
						scope:eafp_industry_state = { eafp_concession_industry_create = { TYPE = building_chemical_plant } }
					}
					else_if = {
						limit = { is_building_type = building_explosives_factory }
						scope:eafp_industry_state = { eafp_concession_industry_create = { TYPE = building_explosives_factory } }
					}
					else_if = {
						limit = { is_building_type = building_synthetics_plant }
						scope:eafp_industry_state = { eafp_concession_industry_create = { TYPE = building_synthetics_plant } }
					}
					else_if = {
						limit = { is_building_type = building_steel_mill }
						scope:eafp_industry_state = { eafp_concession_industry_create = { TYPE = building_steel_mill } }
					}
					else_if = {
						limit = { is_building_type = building_motor_industry }
						scope:eafp_industry_state = { eafp_concession_industry_create = { TYPE = building_motor_industry } }
					}
					else_if = {
						limit = { is_building_type = building_automotive_industry }
						scope:eafp_industry_state = { eafp_concession_industry_create = { TYPE = building_automotive_industry } }
					}
					else_if = {
						limit = { is_building_type = building_electrics_industry }
						scope:eafp_industry_state = { eafp_concession_industry_create = { TYPE = building_electrics_industry } }
					}
					else_if = {
						limit = { is_building_type = building_shipyard }
						scope:eafp_industry_state = { eafp_concession_industry_create = { TYPE = building_shipyard } }
					}
					else_if = {
						limit = { is_building_type = building_arms_industry }
						scope:eafp_industry_state = { eafp_concession_industry_create = { TYPE = building_arms_industry } }
					}
					else_if = {
						limit = { is_building_type = building_artillery_foundry }
						scope:eafp_industry_state = { eafp_concession_industry_create = { TYPE = building_artillery_foundry } }
					}
					else_if = {
						limit = { is_building_type = building_munition_plant }
						scope:eafp_industry_state = { eafp_concession_industry_create = { TYPE = building_munition_plant } }
					}
					else_if = {
						limit = { is_building_type = building_art_academy }
						scope:eafp_industry_state = { eafp_concession_industry_create = { TYPE = building_art_academy } }
					}
					else_if = {
						limit = { is_building_type = building_port }
						scope:eafp_industry_state = { eafp_concession_industry_create = { TYPE = building_port } }
					}
					else_if = {
						limit = { is_building_type = building_trade_center }
						scope:eafp_industry_state = { eafp_concession_industry_create = { TYPE = building_trade_center } }
					}
					if = {
						limit = { levels_owned_by_country = { target = scope:eafp_industry_holder value >= scope:eafp_industry_new_levels } }
						scope:eafp_industry_holder = { add_treasury = { value = eafp_concession_industry_level_cost multiply = -1 } }
					}
					else = { debug_log = "EAFP concession industry: company share did not grow; no cost charged" }
				}
			}
		}
	}

# State scope, after an ownership change (once rights are reconciled). Every remaining
# concession holder of this state gets a concession treaty with the new owner for this
# state, so the right is backed by a treaty with the country that now owns the state.
# (eafp_concession_create_treaty_effect). Skipped when such an article already exists. The article's grant does nothing here: the holder already has the right,
# so no levels are added. The previous owner's article ends through the article's own
# non_fulfillment rule (its source no longer owns the state).
eafp_concession_transfer_treaties_effect = {
	if = {
		limit = { exists = owner }
		save_temporary_scope_as = eafp_transfer_state
		owner = { save_temporary_scope_as = eafp_transfer_owner }
		every_in_list = {
			variable = concession_owner_varlist
			limit = {
				is_country_alive = yes
				NOT = { this = scope:eafp_transfer_owner }
			}
			save_temporary_scope_as = eafp_transfer_holder
			if = {
				limit = {
					NOT = {
						any_scope_treaty = {
							any_scope_article = {
								has_type = eafp_grant_concession
								source_country = scope:eafp_transfer_owner
								input_state = scope:eafp_transfer_state
							}
						}
					}
				}
				eafp_concession_create_treaty_effect = {
					HOST = scope:eafp_transfer_owner
					HOLDER = scope:eafp_transfer_holder
					STATE = scope:eafp_transfer_state
				}
			}
		}
	}
}

# A treaty holding only HOST's concession article for STATE to HOLDER, in force at once with
# binding period 0 days (the shortest create_treaty allows). Used when an existing right
# needs a new treaty; the article's grant does nothing then, since HOLDER already has it.
eafp_concession_create_treaty_effect = {
	$HOST$ = {
		create_treaty = {
			name = eafp_concession_treaty_name
			first_country = $HOST$
			second_country = $HOLDER$
			is_draft = no
			binding_period = { days = 0 }
			articles_to_create = {
				{
					article = eafp_grant_concession
					source_country = $HOST$
					target_country = $HOLDER$
					inputs = {
						{ state = $STATE$ }
					}
				}
			}
		}
	}
}

# FORCED is explicit at every caller; no dependence on treaty callback ordering.
eafp_concession_return_effect = {
	if = {
		limit = {
			owner = scope:concession_host
			eafp_concession_has_holder = { HOLDER = scope:concession_holder }
		}
		# Derive the holder's actual levels through comparisons, not a persistent ledger.
		# Binary search needs at most 32 steps for engine integer building levels.
		save_temporary_scope_value_as = { name = concession_loss_levels value = 0 }
		every_scope_building = {
			limit = { is_building_type = building_eafp_concession }
			eafp_concession_measure_country_levels = { COUNTRY = scope:concession_holder }
			save_temporary_scope_value_as = { name = concession_loss_levels value = scope:concession_level_low }
		}
		# Snapshot before removing the right/building or dissolving the company.
		save_temporary_scope_value_as = { name = concession_loss_relation value = eafp_concession_loss_relations }
		save_temporary_scope_value_as = { name = concession_loss_prestige value = eafp_concession_loss_prestige }
		remove_list_variable = { name = concession_owner_varlist target = scope:concession_holder }
		if = {
			limit = {
				has_building = building_eafp_concession
				NOT = { any_in_list = { variable = concession_owner_varlist always = yes } }
			}
			remove_building = building_eafp_concession
		}
		else_if = {
			limit = { has_building = building_eafp_concession }
			# Other rights remain: only the returning holder's share disappears.
			eafp_concession_remove_holder_share = yes
		}
		if = {
			limit = { always = $FORCED$ eafp_concession_loss_prestige > 0 }
			scope:concession_host = {
				change_relations = { country = scope:concession_holder value = { value = scope:concession_loss_relation multiply = -1 } }
			}
			scope:concession_holder = {
				# The previous loss cannot be reconstructed after rights/buildings are removed.
				# Retain the strongest unexpired shock; a smaller loss must not erase it.
				if = {
					limit = { NOT = { has_variable = concession_loss_prestige_snapshot } }
					set_variable = { name = concession_loss_prestige_snapshot value = 0 days = 730 }
				}
				if = {
					limit = {
						OR = {
							NOT = { has_modifier = eafp_concession_forced_loss }
							var:concession_loss_prestige_snapshot < scope:concession_loss_prestige
						}
					}
					set_variable = { name = concession_loss_prestige_snapshot value = scope:concession_loss_prestige days = 730 }
					remove_modifier = eafp_concession_forced_loss
					add_modifier = { name = eafp_concession_forced_loss multiplier = scope:concession_loss_prestige days = 730 }
				}
			}
		}
		eafp_concession_refresh_effect = yes
	}
}

# One global refresh, used by the international JE and immediate right/ownership changes.
# No money is moved here and this effect never schedules a timer or creates country journal entries.
# Treaty-backed rights that lost their company or building share are restored before the
# invalid rights are dropped.
eafp_concession_refresh_effect = {
	clear_global_variable_list = concession_candidate_states_gvarlist
	every_state = {
		limit = { OR = { has_variable_list = concession_owner_varlist has_building = building_eafp_concession } }
		save_temporary_scope_as = eafp_refresh_state
		owner = { save_temporary_scope_as = eafp_refresh_owner }
		every_in_list = {
			variable = concession_owner_varlist
			save_temporary_scope_as = eafp_refresh_holder
			# A right still backed by an in-force concession treaty with the state's owner is
			# restored when its concession company or its share of the concession building has
			# disappeared: eafp_concession_add_company_level gives the holder the 5 levels of a
			# new grant, founding the company again first (eafp_concession_rebuild_create ->
			# eafp_concession_ensure_company). Rights ended on purpose (return by incident, war
			# or withdrawal) have already left the list, so they are never restored here.
			# add_company_level reads scope:concession_holder; the caller's one is put back.
			if = {
				limit = {
					is_country_alive = yes
					NOT = { this = scope:eafp_refresh_owner }
					OR = {
						NOT = { has_company = company_type:company_eafp_concession }
						scope:eafp_refresh_state = {
							NOT = {
								any_scope_building = {
									is_building_type = building_eafp_concession
									levels_owned_by_country = { target = scope:eafp_refresh_holder value >= 1 }
								}
							}
						}
					}
					scope:eafp_refresh_owner = {
						any_scope_treaty = {
							any_scope_article = {
								has_type = eafp_grant_concession
								source_country = scope:eafp_refresh_owner
								target_country = scope:eafp_refresh_holder
								input_state = scope:eafp_refresh_state
							}
						}
					}
				}
				scope:concession_holder ?= { save_temporary_scope_as = eafp_refresh_caller_holder }
				save_temporary_scope_as = concession_holder
				scope:eafp_refresh_state = { eafp_concession_add_company_level = { LEVELS = 5 } }
				scope:eafp_refresh_caller_holder ?= { save_temporary_scope_as = concession_holder }
			}
			if = {
				limit = {
					OR = {
						is_country_alive = no
						this = scope:eafp_refresh_state.owner
						NOT = { has_company = company_type:company_eafp_concession }
						scope:eafp_refresh_state = {
							NOT = {
								any_scope_building = {
									is_building_type = building_eafp_concession
									levels_owned_by_country = { target = scope:eafp_refresh_holder value >= 1 }
								}
							}
						}
					}
				}
				scope:eafp_refresh_state = {
					remove_list_variable = { name = concession_owner_varlist target = prev }
				}
			}
		}
		if = {
			limit = { any_in_list = { variable = concession_owner_varlist always = yes } }
			add_to_global_variable_list = { name = concession_candidate_states_gvarlist target = this }
			eafp_concession_urban_usage_refresh = yes
		}
		else_if = {
			limit = { has_building = building_eafp_concession }
			remove_building = building_eafp_concession
		}
	}
	every_country = {
		# Rights are already reconciled above. Removing a company never removes ordinary factories.
		if = {
			limit = {
				has_company = company_type:company_eafp_concession
				NOT = { eafp_concession_holder_has_rights = yes }
			}
			remove_company = company_type:company_eafp_concession
		}
		if = {
			limit = { has_variable = concession_loss_prestige_snapshot NOT = { has_modifier = eafp_concession_forced_loss } }
			remove_variable = concession_loss_prestige_snapshot
		}
		# The modifier follows the actual company, not prosperity or the right count.
		if = {
			limit = { has_company = company_type:company_eafp_concession }
			if = {
				limit = { NOT = { has_modifier = eafp_concession_company_slot } }
				add_modifier = { name = eafp_concession_company_slot }
			}
		}
		else_if = {
			limit = { has_modifier = eafp_concession_company_slot }
			remove_modifier = eafp_concession_company_slot
		}
	}
}
