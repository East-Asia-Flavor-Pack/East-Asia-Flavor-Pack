je_concessions_in_china = {
	activation_description = {
		always = yes
	}

	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	group = je_group_global_international_situations
	is_shown_when_inactive = { always = yes }
	# Activated by the concession treaty's on_entry_into_force effect.
	should_be_involved = { eafp_concession_involved_country = yes }
	should_show_when_not_involved = { always = yes }
	widget = {
		gui = "gui/journal_entry_widgets/eafp_concessions_in_china.gui"
		name = "widget_eafp_concessions"
		container = "custom_widget_container_1"
	}
	immediate = { eafp_concession_refresh_effect = yes }
	on_weekly_pulse = {
		effect = {
			eafp_concession_refresh_effect = yes
			every_country = {
				if = {
					limit = {
						eafp_concession_involved_country = yes
						NOT = { is_involved_in_journal_entry = je_concessions_in_china }
					}
					scope:journal_entry = { add_involved_country = prev }
				}
				else_if = {
					limit = {
						is_involved_in_journal_entry = je_concessions_in_china
						NOT = { eafp_concession_involved_country = yes }
					}
					scope:journal_entry = { remove_involved_country = prev }
				}
			}
		}
	}
	on_monthly_pulse = {
		effect = {
			# The holder is drawn from in-force concession articles for this state; the
			# active-holder value needs concession_state saved first.
			every_state = {
				limit = {
					has_building = building_eafp_concession
					eafp_concession_eligible_state = yes
				}
				save_scope_as = concession_state
				if = {
				limit = { eafp_concession_active_holder_count > 0 }
				owner = { save_scope_as = concession_host }
				random = {
					chance = 5
					random_country = {
						limit = {
							is_country_alive = yes
							NOT = { this = scope:concession_host }
							prestige >= scope:concession_host.prestige
							any_scope_treaty = {
								any_scope_article = {
									has_type = eafp_grant_concession
									input_state = scope:concession_state
								}
							}
						}
						save_scope_as = concession_holder
						random_list = {
							20 = { trigger_event = { id = concessions_in_china.200 } }
							20 = { scope:concession_host = { trigger_event = { id = concessions_in_china.201 } } }
							20 = { trigger_event = { id = concessions_in_china.202 } }
							15 = { trigger_event = { id = concessions_in_china.203 } }
							2 = {
								trigger = { "relations(scope:concession_host)" <= -50 }
								scope:concession_state = { eafp_concession_return_effect = { FORCED = yes } }
								scope:concession_host = { trigger_event = { id = concessions_in_china.204 } }
							}
							1 = {
								trigger = {
									"relations(scope:concession_host)" <= -50
									scope:concession_host = { eafp_concession_can_start_return_war = yes }
								}
								eafp_concession_start_return_war = yes
							}
							10 = {
								trigger = {
									scope:concession_host = {
										OR = { has_revolution = yes has_any_secessionists_broken_out = yes }
									}
								}
								trigger_event = { id = concessions_in_china.206 }
							}
							10 = {
								trigger = { any_state = { eafp_concession_can_grant = { HOLDER = prev } } }
								trigger_event = { id = concessions_in_china.207 }
							}
						}
					}
				}
				}
			}
			# Once per actual state, after incidents; never once per holder.
			# Requires a real concession company and building.
			# The active-holder count (treaty + company + prestige) needs concession_state saved.
			every_in_global_list = {
				variable = concession_candidate_states_gvarlist
				save_scope_as = concession_state
				if = {
				limit = {
					eafp_concession_active_holder_count > 0
					eafp_concession_total_levels > 0
					infrastructure_delta > 0
					state_unemployment_rate > 0
					owner = { investment_pool > 0 }
				}
				ordered_scope_building = {
					limit = {
						weekly_profit > 0
						level > 0
						level_after_queued_constructions = level
						is_under_construction = no
						can_queue_building_levels = 1
						OR = {
							AND = { is_building_type = building_food_industry state = { can_construct_building = building_food_industry } }
							AND = { is_building_type = building_textile_mill state = { can_construct_building = building_textile_mill } }
							AND = { is_building_type = building_furniture_manufactory state = { can_construct_building = building_furniture_manufactory } }
							AND = { is_building_type = building_glassworks state = { can_construct_building = building_glassworks } }
							AND = { is_building_type = building_tooling_workshop state = { can_construct_building = building_tooling_workshop } }
							AND = { is_building_type = building_paper_mill state = { can_construct_building = building_paper_mill } }
							AND = { is_building_type = building_chemical_plant state = { can_construct_building = building_chemical_plant } }
							AND = { is_building_type = building_explosives_factory state = { can_construct_building = building_explosives_factory } }
							AND = { is_building_type = building_synthetics_plant state = { can_construct_building = building_synthetics_plant } }
							AND = { is_building_type = building_steel_mill state = { can_construct_building = building_steel_mill } }
							AND = { is_building_type = building_motor_industry state = { can_construct_building = building_motor_industry } }
							AND = { is_building_type = building_automotive_industry state = { can_construct_building = building_automotive_industry } }
							AND = { is_building_type = building_electrics_industry state = { can_construct_building = building_electrics_industry } }
							AND = { is_building_type = building_shipyard state = { can_construct_building = building_shipyard } }
							AND = { is_building_type = building_arms_industry state = { can_construct_building = building_arms_industry } }
							AND = { is_building_type = building_artillery_foundry state = { can_construct_building = building_artillery_foundry } }
							AND = { is_building_type = building_munition_plant state = { can_construct_building = building_munition_plant } }
							AND = { is_building_type = building_art_academy state = { can_construct_building = building_art_academy } }
						}
					}
					order_by = eafp_concession_industry_score
					max = 1
					check_range_bounds = no
					# State-scoped effect does not specify a company. Actual ownership must be tested.
					if = {
						limit = { is_building_type = building_food_industry }
						state = { start_privately_funded_building_construction = building_food_industry }
					}
					else_if = {
						limit = { is_building_type = building_textile_mill }
						state = { start_privately_funded_building_construction = building_textile_mill }
					}
					else_if = {
						limit = { is_building_type = building_furniture_manufactory }
						state = { start_privately_funded_building_construction = building_furniture_manufactory }
					}
					else_if = {
						limit = { is_building_type = building_glassworks }
						state = { start_privately_funded_building_construction = building_glassworks }
					}
					else_if = {
						limit = { is_building_type = building_tooling_workshop }
						state = { start_privately_funded_building_construction = building_tooling_workshop }
					}
					else_if = {
						limit = { is_building_type = building_paper_mill }
						state = { start_privately_funded_building_construction = building_paper_mill }
					}
					else_if = {
						limit = { is_building_type = building_chemical_plant }
						state = { start_privately_funded_building_construction = building_chemical_plant }
					}
					else_if = {
						limit = { is_building_type = building_explosives_factory }
						state = { start_privately_funded_building_construction = building_explosives_factory }
					}
					else_if = {
						limit = { is_building_type = building_synthetics_plant }
						state = { start_privately_funded_building_construction = building_synthetics_plant }
					}
					else_if = {
						limit = { is_building_type = building_steel_mill }
						state = { start_privately_funded_building_construction = building_steel_mill }
					}
					else_if = {
						limit = { is_building_type = building_motor_industry }
						state = { start_privately_funded_building_construction = building_motor_industry }
					}
					else_if = {
						limit = { is_building_type = building_automotive_industry }
						state = { start_privately_funded_building_construction = building_automotive_industry }
					}
					else_if = {
						limit = { is_building_type = building_electrics_industry }
						state = { start_privately_funded_building_construction = building_electrics_industry }
					}
					else_if = {
						limit = { is_building_type = building_shipyard }
						state = { start_privately_funded_building_construction = building_shipyard }
					}
					else_if = {
						limit = { is_building_type = building_arms_industry }
						state = { start_privately_funded_building_construction = building_arms_industry }
					}
					else_if = {
						limit = { is_building_type = building_artillery_foundry }
						state = { start_privately_funded_building_construction = building_artillery_foundry }
					}
					else_if = {
						limit = { is_building_type = building_munition_plant }
						state = { start_privately_funded_building_construction = building_munition_plant }
					}
					else_if = {
						limit = { is_building_type = building_art_academy }
						state = { start_privately_funded_building_construction = building_art_academy }
					}
				}
				}
			}

			# One paid expansion per AI holder/month, using the same effect as the GUI.
			# Treaty checks need saved scopes, so the most profitable eligible state is
			# tracked with a running maximum instead of an ordered list iterator.
			every_country = {
				limit = {
					is_ai = yes
					is_country_alive = yes
					has_company = company_type:company_eafp_concession
				}
				save_temporary_scope_as = concession_holder
				if = {
					limit = { eafp_concession_holder_has_rights = { HOLDER = scope:concession_holder } }
					# Reset per holder; a stale best-state scope is never used while this is 0.
					save_temporary_scope_value_as = { name = concession_best_profit value = 0 }
					every_in_global_list = {
						variable = concession_candidate_states_gvarlist
						save_temporary_scope_as = concession_state
						if = {
							limit = {
								eafp_concession_state_profit > scope:concession_best_profit
								eafp_concession_can_expand = { HOLDER = scope:concession_holder STATE = scope:concession_state LEVELS = 1 }
							}
							save_temporary_scope_value_as = { name = concession_best_profit value = eafp_concession_state_profit }
							save_temporary_scope_as = concession_best_state
						}
					}
					if = {
						limit = { scope:concession_best_profit > 0 }
						scope:concession_best_state = {
							save_temporary_scope_as = concession_state
							eafp_concession_expand_effect = { LEVELS = 1 }
						}
					}
				}
			}

		}
	}
	weight = 600
}
